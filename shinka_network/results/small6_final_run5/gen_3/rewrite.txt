"""Surrogate QUBO terms for QOBLIB 08 network design. ShinkaEvolve edits the EVOLVE block."""

import time

# EVOLVE-BLOCK-START
import numpy as np


def surrogate_terms(demand):
    """Build a sparse, congestion-aware direct-link and two-hop surrogate."""
    matrix = np.asarray(demand, dtype=float)
    n = len(matrix)

    traffic = np.maximum(matrix, 0.0).copy()
    np.fill_diagonal(traffic, 0.0)
    maximum = float(np.max(traffic)) if traffic.size else 0.0
    if maximum <= 0.0:
        maximum = 1.0
    normalized = traffic / maximum

    row = traffic.sum(axis=1)
    col = traffic.sum(axis=0)
    total = float(row.sum())
    if total <= 0.0:
        total = 1.0

    # A direct link is valuable for its own traffic. The mild concave component
    # keeps medium-sized demands relevant instead of concentrating everything
    # on only the largest entries.
    roots = np.sqrt(normalized)
    linear = {}
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            direct = 0.72 * normalized[i, j] + 0.28 * roots[i, j]

            # Roughly estimate traffic that could use i->j as a transit link:
            # traffic arriving at i and traffic leaving j. Keep this modest so
            # it complements, rather than overwhelms, direct-demand evidence.
            transit = (col[i] * row[j]) / total
            transit_norm = transit / maximum
            transit_value = np.sqrt(max(0.0, transit_norm))
            linear[(i, j)] = -float(direct + 0.08 * transit_value)

    # Congestion proxies for edges and transit nodes. Edge proxy combines its
    # direct load with a small estimate of through-traffic potential.
    edge_transit = np.outer(col, row) / total
    edge_load = normalized + 0.16 * (edge_transit / maximum)
    edge_load = np.maximum(edge_load, 0.0)
    np.fill_diagonal(edge_load, 0.0)

    node_load = row + col
    node_scale = float(np.max(node_load)) if node_load.size else 0.0
    if node_scale <= 0.0:
        node_scale = 1.0
    node_pressure = node_load / node_scale

    quadratic = {}
    positive = normalized[normalized > 0.0]
    cutoff = float(np.quantile(positive, 0.70)) if positive.size else 0.0

    for i in range(n):
        for k in range(n):
            if i == k or normalized[i, k] <= 0.0:
                continue

            demand_level = float(normalized[i, k])
            support = 0.68 * demand_level + 0.32 * float(roots[i, k])

            candidates = []
            for j in range(n):
                if j == i or j == k:
                    continue

                first_load = float(edge_load[i, j])
                second_load = float(edge_load[j, k])
                middle_pressure = float(node_pressure[j])

                # Penalize routes that rely on already busy links or transit
                # nodes, while retaining a small preference for useful edges.
                congestion = (
                    0.20
                    + first_load
                    + second_load
                    + 0.45 * middle_pressure
                )
                edge_affinity = 0.12 * (
                    float(normalized[i, j]) + float(normalized[j, k])
                )
                route_score = (1.0 + edge_affinity) / (congestion ** 0.70)
                candidates.append((route_score, j, congestion))

            candidates.sort(key=lambda item: (-item[0], item[1]))
            if not candidates:
                continue

            # A few good alternatives provide a useful sampling signal without
            # creating a dense coupling graph.
            keep = 3 if demand_level >= cutoff else 2
            chosen = candidates[:min(keep, len(candidates))]
            best_score = max(float(chosen[0][0]), 1e-12)

            for rank, (route_score, j, congestion) in enumerate(chosen):
                relative_quality = min(1.0, float(route_score) / best_score)
                rank_discount = 1.0 - 0.12 * rank
                congestion_discount = 1.0 / (1.0 + 0.20 * float(congestion))
                weight = (
                    -0.50
                    * support
                    * (0.55 + 0.45 * relative_quality)
                    * rank_discount
                    * congestion_discount
                )
                first, second = (i, j), (j, k)
                key = (first, second) if first < second else (second, first)
                quadratic[key] = float(weight)

    # The evaluator applies degree penalties separately.
    return linear, quadratic, 1.15


# EVOLVE-BLOCK-END


def run_surrogate(demand):
    """Fixed entry point called by the evaluator, once per demand matrix."""
    demand = np.array(demand, dtype=float)
    start = time.perf_counter()
    linear, quadratic, penalty_weight = surrogate_terms(demand.copy())
    return {
        "n": len(demand),
        "linear": linear,
        "quadratic": quadratic,
        "penalty_weight": penalty_weight,
        "seconds": time.perf_counter() - start,
    }
