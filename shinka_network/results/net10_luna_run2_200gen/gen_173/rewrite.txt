"""Surrogate QUBO terms for QOBLIB 08 network design."""

import time

# EVOLVE-BLOCK-START
import numpy as np


def surrogate_terms(demand):
    """Build direct-link rewards and a normalized portfolio of two-hop routes."""
    matrix = np.asarray(demand, dtype=float)
    if matrix.ndim != 2 or matrix.shape[0] != matrix.shape[1]:
        raise ValueError("demand must be a square matrix")
    n = int(matrix.shape[0])
    if n == 0:
        return {}, {}, 1.0

    traffic = np.nan_to_num(
        matrix, nan=0.0, posinf=0.0, neginf=0.0
    ).copy()
    np.maximum(traffic, 0.0, out=traffic)
    np.fill_diagonal(traffic, 0.0)

    scale = float(np.max(traffic))
    if scale <= 0.0:
        scale = 1.0
    d = traffic / scale

    # Keep direct demand as the primary signal, while retaining some
    # sensitivity to moderate demands.
    linear = {}
    for i in range(n):
        for j in range(n):
            if i != j:
                value = float(d[i, j])
                linear[(i, j)] = -(0.75 * value + 0.25 * np.sqrt(value))

    # Estimate the relative burden of using each node as a transit point.
    row_load = np.sum(d, axis=1)
    col_load = np.sum(d, axis=0)
    incident = row_load + col_load
    mean_incident = float(np.mean(incident))
    if mean_incident > 1e-12:
        pressure = np.clip(incident / mean_incident, 0.35, 3.0)
    else:
        pressure = np.ones(n, dtype=float)

    quadratic = {}
    portfolio_size = min(4, max(1, n - 2))
    path_strength = 1.05
    temperature = 0.48

    for i in range(n):
        for k in range(n):
            if i == k or d[i, k] <= 0.0:
                continue

            value = float(d[i, k])
            support = 0.70 * value + 0.30 * np.sqrt(value)

            # Rank intermediary nodes by a proxy for the bottleneck cost
            # of the two edges and the transit burden at the intermediary.
            candidates = []
            for j in range(n):
                if j == i or j == k:
                    continue
                edge_exposure = float(d[i, j] + d[j, k])
                cost = 0.58 * edge_exposure + 0.42 * float(pressure[j])
                candidates.append((cost, j))

            candidates.sort()
            selected = candidates[:portfolio_size]

            # A normalized softmax shares this OD pair's route budget among
            # a few relatively low-exposure alternatives.
            min_cost = selected[0][0]
            scores = np.array(
                [np.exp(-(cost - min_cost) / temperature) for cost, _ in selected],
                dtype=float,
            )
            total = float(np.sum(scores))
            if total <= 0.0:
                scores.fill(1.0 / len(scores))
            else:
                scores /= total

            for probability, (_, j) in zip(scores, selected):
                first = (i, j)
                second = (j, k)
                key = (first, second) if first < second else (second, first)
                quadratic[key] = quadratic.get(key, 0.0) - (
                    path_strength * support * float(probability)
                )

    # Degree penalties are added by the evaluator.
    return linear, quadratic, 1.0


# EVOLVE-BLOCK-END


def run_surrogate(demand):
    """Fixed entry point called by the evaluator, once per demand matrix."""
    demand = np.array(demand, dtype=float)
    start = time.perf_counter()
    linear, quadratic, penalty_weight = surrogate_terms(demand.copy())
    return {
        "n": len(demand),
        "linear": linear,
        "quadratic": quadratic,
        "penalty_weight": penalty_weight,
        "seconds": time.perf_counter() - start,
    }
