"""Surrogate QUBO terms for QOBLIB 08 network design."""

import time

# EVOLVE-BLOCK-START
import numpy as np


def surrogate_terms(demand):
    """Build direct-link and aggregate-capped two-hop cost terms."""
    matrix = np.asarray(demand, dtype=float)
    if matrix.ndim != 2 or matrix.shape[0] != matrix.shape[1]:
        raise ValueError("demand must be a square matrix")
    n = int(matrix.shape[0])

    # Ignore invalid, negative, and self traffic.
    traffic = np.nan_to_num(
        matrix, nan=0.0, posinf=0.0, neginf=0.0
    ).copy()
    np.maximum(traffic, 0.0, out=traffic)
    np.fill_diagonal(traffic, 0.0)

    scale = float(np.max(traffic)) if traffic.size else 0.0
    if scale <= 0.0:
        scale = 1.0
    normalized = traffic / scale

    # Direct-link reward: preserve strong demand preference while retaining
    # modest support for smaller OD flows.
    linear_fraction = 0.82
    linear = {}
    direct_reward = np.zeros((n, n), dtype=float)
    for i in range(n):
        for k in range(n):
            if i == k:
                continue
            value = float(normalized[i, k])
            reward = linear_fraction * value + (1.0 - linear_fraction) * np.sqrt(value)
            direct_reward[i, k] = reward
            linear[(i, k)] = -float(reward)

    # Approximate relay congestion using total incident traffic. A modest
    # floor avoids making any node categorically unusable as a relay.
    incident = np.sum(traffic, axis=0) + np.sum(traffic, axis=1)
    mean_incident = float(np.mean(incident)) if n else 0.0
    if mean_incident <= 0.0:
        mean_incident = 1.0
    node_pressure = np.maximum(0.70, incident / mean_incident)

    quadratic = {}
    path_strength = 0.52
    relay_exponent = 0.70
    edge_pressure_factor = 0.65
    aggregate_path_cap = 0.75

    for i in range(n):
        for k in range(n):
            if i == k or traffic[i, k] <= 0.0:
                continue

            od = float(normalized[i, k])
            # Concavity retains useful signal for moderate-demand OD pairs.
            path_support = 0.78 * od + 0.22 * np.sqrt(od)

            candidates = []
            total_raw = 0.0
            for j in range(n):
                if j == i or j == k:
                    continue

                edge_pressure = float(normalized[i, j] + normalized[j, k])
                relay_factor = float(node_pressure[j]) ** relay_exponent
                raw = (
                    path_strength * path_support
                    / (relay_factor * (1.0 + edge_pressure_factor * edge_pressure))
                )
                if raw > 0.0:
                    candidates.append((j, raw))
                    total_raw += raw

            if total_raw <= 0.0:
                continue

            # Limit all two-hop support for this OD together, rather than
            # allowing each possible relay to contribute an unrestricted bonus.
            cap = aggregate_path_cap * float(direct_reward[i, k])
            multiplier = min(1.0, cap / total_raw)

            for j, raw in candidates:
                first, second = (i, j), (j, k)
                key = (first, second) if first < second else (second, first)
                quadratic[key] = quadratic.get(key, 0.0) - float(raw * multiplier)

    # Degree penalties are supplied by the evaluator.
    penalty_weight = 1.0
    return linear, quadratic, penalty_weight


# EVOLVE-BLOCK-END


def run_surrogate(demand):
    """Fixed entry point called by the evaluator, once per demand matrix."""
    demand = np.array(demand, dtype=float)
    start = time.perf_counter()
    linear, quadratic, penalty_weight = surrogate_terms(demand.copy())
    return {
        "n": len(demand),
        "linear": linear,
        "quadratic": quadratic,
        "penalty_weight": penalty_weight,
        "seconds": time.perf_counter() - start,
    }
