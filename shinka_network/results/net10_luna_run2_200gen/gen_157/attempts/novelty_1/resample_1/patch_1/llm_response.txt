<DESCRIPTION>
This version keeps the direct-link reward, but limits the total two-hop reward for each origin–destination pair to 75% of its direct-link reward. Candidate routes are weighted by transit-node pressure and the direct demand on their component links, so indirect support favors less congested paths. The cap prevents a pair with many possible intermediates from accumulating excessive reward. A moderately stronger degree penalty is intended to encourage more feasible samples before repair.
</DESCRIPTION>

"""Surrogate QUBO terms for QOBLIB 08 network design. ShinkaEvolve edits the EVOLVE block."""

import time

# EVOLVE-BLOCK-START
import numpy as np


def surrogate_terms(demand):
    """Build direct-link and capped two-hop surrogate cost terms."""
    traffic = np.asarray(demand, dtype=float)
    n = len(traffic)

    # Ignore self-traffic and invalid negative/non-finite values.
    traffic = np.nan_to_num(traffic, nan=0.0, posinf=0.0, neginf=0.0)
    traffic = np.maximum(traffic, 0.0).copy()
    np.fill_diagonal(traffic, 0.0)

    scale = float(np.max(traffic)) if traffic.size else 0.0
    if scale <= 0.0:
        scale = 1.0
    normalized = traffic / scale

    # Direct service remains the main signal. A small concave component helps
    # moderate demands influence link selection without dominating large ones.
    direct_value = 0.88 * normalized + 0.12 * np.sqrt(normalized)
    linear = {
        (i, j): -float(direct_value[i, j])
        for i in range(n)
        for j in range(n)
        if i != j
    }

    # Estimate how attractive each node is as a transit point. Using incident
    # demand penalizes transit through nodes already responsible for substantial
    # traffic, while the floor avoids eliminating any node entirely.
    incident = np.sum(normalized, axis=0) + np.sum(normalized, axis=1)
    mean_incident = float(np.mean(incident)) if incident.size else 0.0
    if mean_incident > 0.0:
        node_pressure = np.maximum(0.60, incident / mean_incident)
    else:
        node_pressure = np.ones(n, dtype=float)

    quadratic = {}
    support_per_route = 0.24
    support_cap = 0.75

    # For each OD pair, form candidate two-hop rewards, then proportionally
    # cap their aggregate. This prevents many alternative intermediates from
    # making one OD pair disproportionately influential.
    for i in range(n):
        for k in range(n):
            if i == k or traffic[i, k] <= 0.0:
                continue

            od_reward = float(direct_value[i, k])
            candidates = []
            total_support = 0.0

            for j in range(n):
                if j == i or j == k:
                    continue

                edge_pressure = float(normalized[i, j] + normalized[j, k])
                route_quality = (
                    1.0
                    / np.sqrt(float(node_pressure[j]))
                    / (1.0 + 0.35 * edge_pressure)
                )
                weight = support_per_route * od_reward * route_quality
                if weight > 0.0:
                    candidates.append((j, weight))
                    total_support += weight

            limit = support_cap * od_reward
            multiplier = min(1.0, limit / total_support) if total_support > 0.0 else 1.0

            for j, weight in candidates:
                first, second = (i, j), (j, k)
                key = (first, second) if first < second else (second, first)
                quadratic[key] = -float(weight * multiplier)

    # The evaluator supplies the degree penalties separately.
    penalty_weight = 1.35
    return linear, quadratic, penalty_weight


# EVOLVE-BLOCK-END


def run_surrogate(demand):
    """Fixed entry point called by the evaluator, once per demand matrix."""
    demand = np.array(demand, dtype=float)
    start = time.perf_counter()
    linear, quadratic, penalty_weight = surrogate_terms(demand.copy())
    return {
        "n": len(demand),
        "linear": linear,
        "quadratic": quadratic,
        "penalty_weight": penalty_weight,
        "seconds": time.perf_counter() - start,
    }
