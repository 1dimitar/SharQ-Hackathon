"""Surrogate QUBO terms for QOBLIB 08 network design."""

import time

# EVOLVE-BLOCK-START
import numpy as np


def surrogate_terms(demand):
    """Build direct-link rewards and discounted two-hop path rewards."""
    matrix = np.asarray(demand, dtype=float)
    n = len(matrix)

    traffic = np.maximum(matrix, 0.0).copy()
    np.fill_diagonal(traffic, 0.0)

    scale = float(np.max(traffic)) if traffic.size else 0.0
    if scale <= 0.0:
        scale = 1.0

    # Intermediate concavity: retain emphasis on large direct demands,
    # while allowing moderate demands to contribute to link selection.
    direct_linear_share = 0.80
    direct_sqrt_share = 0.20
    linear = {}
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            value = float(traffic[i, j]) / scale
            linear[(i, j)] = -(
                direct_linear_share * value
                + direct_sqrt_share * np.sqrt(value)
            )

    incident = np.sum(traffic, axis=0) + np.sum(traffic, axis=1)
    mean_incident = float(np.mean(incident)) if incident.size else 0.0
    if mean_incident <= 0.0:
        mean_incident = 1.0
    node_pressure = np.maximum(0.75, incident / mean_incident)

    quadratic = {}
    path_strength = 0.53
    path_linear_share = 0.70
    path_sqrt_share = 0.30
    edge_pressure_strength = 0.50

    for i in range(n):
        for k in range(n):
            if i == k:
                continue

            through_demand = float(traffic[i, k])
            if through_demand <= 0.0:
                continue

            normalized = through_demand / scale
            path_support = (
                path_linear_share * normalized
                + path_sqrt_share * np.sqrt(normalized)
            )
            # Bound the concavity adjustment to avoid disproportionately
            # rewarding tiny demands compared with substantial traffic.
            path_support = normalized * min(
                1.25, max(0.90, path_support / normalized)
            )

            for j in range(n):
                if j == i or j == k:
                    continue

                edge_pressure = (
                    float(traffic[i, j]) + float(traffic[j, k])
                ) / scale
                weight = -(
                    path_strength * path_support
                    / (
                        float(node_pressure[j])
                        * (1.0 + edge_pressure_strength * edge_pressure)
                    )
                )

                first, second = (i, j), (j, k)
                key = (first, second) if first < second else (second, first)
                quadratic[key] = weight

    # Degree penalties are supplied by the evaluator.
    penalty_weight = 1.0
    return linear, quadratic, penalty_weight


# EVOLVE-BLOCK-END


def run_surrogate(demand):
    """Fixed entry point called by the evaluator, once per demand matrix."""
    demand = np.array(demand, dtype=float)
    start = time.perf_counter()
    linear, quadratic, penalty_weight = surrogate_terms(demand.copy())
    return {
        "n": len(demand),
        "linear": linear,
        "quadratic": quadratic,
        "penalty_weight": penalty_weight,
        "seconds": time.perf_counter() - start,
    }
