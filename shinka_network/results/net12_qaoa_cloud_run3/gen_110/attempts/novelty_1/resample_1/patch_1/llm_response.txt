def _relay_candidates(i, k, normalized, pressure, limit=2):
    """Rank relays by affinity and pressure, retaining reward quality separately."""
    n = normalized.shape[0]
    candidates = []

    for j in range(n):
        if j == i or j == k:
            continue

        # Use pressure sensitivity only to select relays. The reward quality
        # remains the previous pressure discount, so this isolates ranking.
        leg_a = float(normalized[i, j])
        leg_b = float(normalized[j, k])
        leg_affinity = np.sqrt((0.05 + leg_a) * (0.05 + leg_b))
        reward_quality = (0.75 + 0.25 * leg_affinity) / (0.7 + float(pressure[j]))
        rank_quality = leg_affinity / ((0.75 + float(pressure[j])) ** 0.75)
        candidates.append((rank_quality, reward_quality, j))

    candidates.sort(key=lambda item: (-item[0], item[2]))
    return candidates[:limit]
