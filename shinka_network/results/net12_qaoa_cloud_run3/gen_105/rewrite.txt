"""Surrogate QUBO terms for QOBLIB 08 network design."""

import time

# EVOLVE-BLOCK-START
import numpy as np


def surrogate_terms(demand):
    """Build direct-link rewards and congestion-aware two-hop rewards."""
    traffic = np.asarray(demand, dtype=float)
    n = len(traffic)

    traffic = np.nan_to_num(
        traffic, nan=0.0, posinf=0.0, neginf=0.0
    )
    traffic = np.maximum(traffic, 0.0).copy()
    np.fill_diagonal(traffic, 0.0)

    scale = float(np.max(traffic)) if traffic.size else 0.0
    if scale <= 0.0:
        scale = 1.0
    q = traffic / scale

    # Retain a strong direct-service incentive while giving moderate demands
    # some influence over the scarce two outgoing and incoming link slots.
    linear = {}
    for i in range(n):
        for j in range(n):
            if i != j:
                d = float(q[i, j])
                linear[(i, j)] = -(0.78 * d + 0.22 * np.sqrt(d))

    # Approximate congestion from both the direct traffic on an edge and the
    # aggregate demand incident to its endpoints. These are routing-cost
    # proxies, not additional degree penalties.
    row = np.sum(q, axis=1)
    col = np.sum(q, axis=0)
    row_max = max(float(np.max(row)) if n else 0.0, 1.0)
    col_max = max(float(np.max(col)) if n else 0.0, 1.0)
    row_pressure = row / row_max
    col_pressure = col / col_max
    node_pressure = np.maximum(row_pressure, col_pressure)

    edge_burden = np.zeros((n, n), dtype=float)
    for i in range(n):
        for j in range(n):
            if i != j:
                edge_burden[i, j] = (
                    0.50 * float(q[i, j])
                    + 0.25 * float(row_pressure[i])
                    + 0.25 * float(col_pressure[j])
                )

    quadratic = {}
    path_strength = 0.42
    relay_budget = min(n - 2, max(2, int(np.sqrt(n)) + 1))

    for i in range(n):
        for k in range(n):
            if i == k:
                continue

            d = float(q[i, k])
            if d <= 0.0:
                continue

            support = 0.72 * d + 0.28 * np.sqrt(d)
            candidates = []

            for j in range(n):
                if j == i or j == k:
                    continue

                # Favor routes whose two links and intermediate node have
                # spare estimated capacity. The floor avoids unstable scores
                # when all estimated burdens are small.
                burden_a = 0.15 + float(edge_burden[i, j])
                burden_b = 0.15 + float(edge_burden[j, k])
                relay_cost = 0.65 + float(node_pressure[j])
                utility = 1.0 / (
                    np.sqrt(burden_a * burden_b) * relay_cost
                )
                candidates.append((utility, j))

            if not candidates:
                continue

            # Allocate a bounded amount of credit across good alternatives.
            # Shares are computed over all relays, then only the best few are
            # represented in the QUBO to keep its interaction graph compact.
            candidates.sort(key=lambda item: (-item[0], item[1]))
            utility_sum = sum(item[0] for item in candidates)
            if utility_sum <= 0.0:
                continue

            for utility, j in candidates[:relay_budget]:
                share = utility / utility_sum
                weight = -path_strength * support * share
                first, second = (i, j), (j, k)
                key = (first, second) if first < second else (second, first)
                quadratic[key] = quadratic.get(key, 0.0) + weight

    # Degree constraints are applied by the evaluator.
    penalty_weight = 5.0
    return linear, quadratic, penalty_weight


# EVOLVE-BLOCK-END


def run_surrogate(demand):
    """Fixed entry point called by the evaluator, once per demand matrix."""
    demand = np.array(demand, dtype=float)
    start = time.perf_counter()
    linear, quadratic, penalty_weight = surrogate_terms(demand.copy())
    return {
        "n": len(demand),
        "linear": linear,
        "quadratic": quadratic,
        "penalty_weight": penalty_weight,
        "seconds": time.perf_counter() - start,
    }
