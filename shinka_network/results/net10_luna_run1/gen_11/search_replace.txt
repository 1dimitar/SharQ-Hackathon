<DESCRIPTION>
Keep the direct-demand linear reward, and add only demand-conditioned two-hop interactions. A path \(i \to j \to k\) is rewarded only when \(t[i,k] > 0\), with its reward scaled by how strongly demand supports the incoming and outgoing legs at the intermediate node. The normalized inbound/outbound shares make the profile comparable across nodes and prevent generic two-hop paths from receiving the same reward. The interaction set is at most \(n(n-1)(n-2)\), within the term limit, and avoids the broader shared-endpoint and reciprocal-link penalties that performed worse.

def surrogate_terms(demand):
    """Build the surrogate cost terms of the topology QUBO from the demand matrix.

    Args:
        demand: (n, n) float array; demand[i, j] is the traffic from node i to node j
            (zero diagonal).

    Returns:
        linear: dict {(i, j): weight}. Weight on the binary variable x_ij
            (x_ij = 1 if the directed link i -> j is built), i != j.
        quadratic: dict {((i, j), (k, l)): weight}. Weight on the product x_ij * x_kl.
        penalty_weight: float. Strength of the degree penalty that the evaluator adds
            after rescaling these terms so the largest |weight| is 1.

    Lower energy must mean a better topology. Baseline: reward each link by the demand
    it carries directly.
    """
    n = len(demand)
    scale = float(demand.max()) or 1.0
    linear = {(i, j): -float(demand[i, j]) / scale for i in range(n) for j in range(n) if i != j}
    quadratic = {}
    penalty_weight = 1.0
    return linear, quadratic, penalty_weight

def surrogate_terms(demand):
    """Build direct-link rewards and demand-conditioned two-hop path rewards."""
    t = np.maximum(np.asarray(demand, dtype=float), 0.0).copy()
    n = t.shape[0]
    np.fill_diagonal(t, 0.0)

    scale = float(t.max()) if t.size else 0.0
    if scale > 0.0:
        t /= scale

    linear = {
        (i, j): -float(t[i, j])
        for i in range(n)
        for j in range(n)
        if i != j
    }

    # A path through j is more credible when both its incoming and outgoing
    # legs are supported by traffic at j. Normalize each leg by the total
    # traffic into/out of j, then rescale shares so uniform profiles are O(1).
    rows = t.sum(axis=1)
    cols = t.sum(axis=0)
    inbound = np.divide(
        t, cols[None, :], out=np.zeros_like(t), where=cols[None, :] > 0.0
    )
    outbound = np.divide(
        t, rows[:, None], out=np.zeros_like(t), where=rows[:, None] > 0.0
    )

    quadratic = {}
    for i in range(n):
        for j in range(n):
            if j == i:
                continue
            in_support = float(inbound[i, j])
            if in_support <= 0.0:
                continue
            for k in range(n):
                if k == i or k == j or t[i, k] <= 0.0:
                    continue
                out_support = float(outbound[j, k])
                if out_support <= 0.0:
                    continue
                profile = min(1.0, n * (in_support * out_support) ** 0.5)
                value = 0.12 * float(t[i, k]) * profile
                if value > 1e-14:
                    quadratic[((i, j), (j, k))] = -value

    penalty_weight = 1.0
    return linear, quadratic, penalty_weight
