<DESCRIPTION>
This replaces the direct-demand-plus-generic-path-reward model with a demand-conditioned transit-fit model. For each origin–destination pair, it scores candidate two-hop routes according to how strongly the intermediate node is connected to that particular origin and destination, rather than using a broad node-traffic statistic. It also retains a modest direct-link preference and discounts transit choices at nodes with high aggregate traffic. The resulting interactions remain restricted to two-hop paths, keeping the term count within the limit while aiming to make useful routes more prominent in both the best and typical low-energy samples.

import numpy as np


def surrogate_terms(demand):
    """Build demand-conditioned two-hop QUBO surrogate terms."""
    t = np.asarray(demand, dtype=float)
    n = len(t)
    if n == 0:
        return {}, {}, 1.0

    # Ignore negative entries if supplied; normalize for scale invariance.
    t = np.maximum(t, 0.0)
    scale = float(np.max(t))
    d = t / scale if scale > 0.0 else np.zeros_like(t)

    row = d.sum(axis=1)
    col = d.sum(axis=0)
    total = float(d.sum())

    linear = {}
    quadratic = {}

    # Prefer direct service for significant OD pairs, while mildly reflecting
    # how much traffic originates at i and terminates at j.
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            endpoint = 0.5 * (
                row[i] / (float(row.max()) + 1e-12)
                + col[j] / (float(col.max()) + 1e-12)
            )
            linear[(i, j)] = -0.72 * float(d[i, j]) - 0.06 * endpoint

    if total > 0.0:
        # Local link evidence estimates whether j is a plausible bridge for
        # this specific OD pair: i's traffic to j and j's traffic to k.
        # Normalizing each side avoids allowing a single hub to dominate.
        for i in range(n):
            for k in range(n):
                if i == k:
                    continue
                od = float(d[i, k])
                if od <= 0.0:
                    continue

                for j in range(n):
                    if j == i or j == k:
                        continue

                    in_fit = float(d[i, j]) / (float(row[i]) + 1e-12)
                    out_fit = float(d[j, k]) / (float(row[j]) + 1e-12)

                    # A small background term permits useful transit routes
                    # even when the exact adjacent OD traffic is absent.
                    fit = 0.15 + 0.85 * np.sqrt(in_fit * out_fit)

                    # Discount transit through nodes with high total
                    # throughput, which are more likely to become bottlenecks.
                    transit_load = (float(row[j]) + float(col[j])) / (
                        2.0 * total / max(1, n) + 1e-12
                    )
                    congestion_discount = 1.0 / (1.0 + 0.35 * transit_load)

                    key = ((i, j), (j, k))
                    quadratic[key] = -0.34 * od * fit * congestion_discount

    # Keep the returned coefficients numerically well scaled; the evaluator
    # performs its own rescaling as well.
    largest = max(
        [abs(v) for v in linear.values()]
        + [abs(v) for v in quadratic.values()]
        + [0.0]
    )
    if largest > 0.0:
        linear = {key: value / largest for key, value in linear.items()}
        quadratic = {key: value / largest for key, value in quadratic.items()}

    return linear, quadratic, 1.5


# EVOLVE-BLOCK-END
