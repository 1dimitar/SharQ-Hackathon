<DESCRIPTION>
Add bounded, pressure-sensitive couplings between links sharing a sender or receiver. A pair of links aimed at high-inbound-pressure destinations, or originating from high-outbound-pressure nodes, receives a stronger congestion penalty. The pressure is normalized from the demand matrix, so the adjustment remains bounded and adapts across instances. This adds at most \(2n\binom{n-1}{2}\) quadratic terms, within the limit, while preserving the direct-demand linear reward. Increase the degree penalty modestly to encourage valid topologies before repair.
</DESCRIPTION>

    t = np.asarray(demand, dtype=float)
    n = len(t)
    if n < 2:
        return {}, {}, 2.0

    t = np.maximum(t, 0.0).copy()
    np.fill_diagonal(t, 0.0)
    scale = float(t.max())
    if scale <= 0.0:
        return ({(i, j): 0.0 for i in range(n) for j in range(n) if i != j},
                {}, 2.0)

    d = t / scale
    links = [(i, j) for i in range(n) for j in range(n) if i != j]
    linear = {(i, j): -float(d[i, j]) for i, j in links}
    quadratic = {}

    # Normalize node traffic pressure to [0, 1] for scale-independent weights.
    row = d.sum(axis=1)
    col = d.sum(axis=0)
    row_pressure = row / (float(row.max()) + 1e-12)
    col_pressure = col / (float(col.max()) + 1e-12)

    pair_scale = 0.12
    for i in range(n):
        outgoing = [(i, j) for j in range(n) if j != i]
        for a in range(len(outgoing)):
            u = outgoing[a]
            for b in range(a + 1, len(outgoing)):
                v = outgoing[b]
                # Avoid concentrating two links from a busy sender on busy receivers.
                w = pair_scale * row_pressure[i] * np.sqrt(
                    col_pressure[u[1]] * col_pressure[v[1]]
                )
                if w > 1e-10:
                    quadratic[(u, v)] = float(w)

    for j in range(n):
        incoming = [(i, j) for i in range(n) if i != j]
        for a in range(len(incoming)):
            u = incoming[a]
            for b in range(a + 1, len(incoming)):
                v = incoming[b]
                # Busy origins sharing a busy receiver can amplify transit congestion.
                w = pair_scale * col_pressure[j] * np.sqrt(
                    row_pressure[u[0]] * row_pressure[v[0]]
                )
                if w > 1e-10:
                    key = (u, v) if u < v else (v, u)
                    quadratic[key] = quadratic.get(key, 0.0) + float(w)

    penalty_weight = 2.0
    return linear, quadratic, penalty_weight
