    def archive():
        """Reserve archive capacity for LP quality, energy, and topology novelty."""
        ranked = sorted(seen, key=seen.get)
        if not ranked:
            return []

        limit = min(len(ranked), max(2 * n, 24))
        out = []
        selected = set()

        # Preserve a reliable core of the best certified networks.
        elite_count = max(4, n // 2)
        for net in ranked[:elite_count]:
            out.append(net)
            selected.add(net)

        # Energy contributes a separate signal, but only among reasonably
        # competitive LP-scored networks.
        cutoff = best_score * 1.10
        eligible = [net for net in ranked if seen[net] <= cutoff]
        energy_count = max(4, limit // 3)
        energy_order = sorted(eligible, key=energy)
        for net in energy_order:
            if len(out) >= min(limit, elite_count + energy_count):
                break
            if net in selected:
                continue
            if all(distance(net, old) >= 4 for old in out):
                out.append(net)
                selected.add(net)

        # Fill remaining places by farthest-point selection. This explicitly
        # protects distinct search regions instead of letting LP rank alone
        # determine which regions remain available as parents.
        novelty_pool = eligible[:min(len(eligible), max(8 * n, 96))]
        while len(out) < limit:
            remaining = [net for net in novelty_pool if net not in selected]
            if not remaining:
                remaining = [net for net in ranked if net not in selected]
            if not remaining:
                break
            net = max(
                remaining,
                key=lambda x: (
                    min((distance(x, old) for old in out), default=2 * n),
                    -seen[x],
                    -energy(x),
                ),
            )
            out.append(net)
            selected.add(net)

        # If the eligible set was small, retain additional strong networks.
        for net in ranked:
            if len(out) >= limit:
                break
            if net not in selected:
                out.append(net)
                selected.add(net)
        return out
