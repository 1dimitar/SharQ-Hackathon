<DESCRIPTION>
Replace the link-pressure ranking with a genuine exploration reserve. Removing heavily loaded links does not reliably improve routing, and the current fill loop consumes every remaining slot before exploration can run.

Keep most LP calls for low-energy proposals. Reserve the remaining calls for structurally novel proposals and a randomized move-scale stratum, so numerous rotations cannot crowd out two-switches or longer kicks. Increase this reserve modestly when a parent stalls. Novelty is measured against good archived networks as well as the current batch, encouraging exploration of different basins rather than merely different siblings. All topology comparisons use precomputed edge sets to keep screening inexpensive.
</DESCRIPTION>

    def screen_batch(moves, parent, detail=None, batch_size=9):
        """Separate energy exploitation from novelty and move-scale exploration."""
        fresh = [
            m for m in moves
            if m not in seen and not tools.is_cached(m)
        ]
        if not fresh:
            return []

        ranked = sorted(fresh, key=energy)
        target = min(batch_size, len(ranked))
        stalls = parent_stalls.get(parent, 0)
        reserve = min(target - 1, max(2, batch_size // 3) + int(stalls >= 3))
        selected = ranked[:target - reserve]
        selected_set = set(selected)

        edge_sets = {net: set(net) for net in ranked}
        parent_set = set(parent)

        # Include the archive, not just siblings: novelty should identify
        # unexplored regions rather than variants of another good incumbent.
        references = [
            set(net) for net in archive()[:max(8, n)]
        ]
        references.extend(edge_sets[net] for net in selected)

        # Restrict max-min selection to a broad energy shortlist; leave the
        # final slot unrestricted so a miscalibrated surrogate cannot veto it.
        shortlist = ranked[:max(target, (3 * len(ranked)) // 4)]
        while len(selected) < target - 1:
            remaining = [net for net in shortlist if net not in selected_set]
            if not remaining:
                remaining = [net for net in ranked if net not in selected_set]
            if not remaining:
                break

            # Equal-sized networks: edge difference is half symmetric distance.
            # Iteration in energy order breaks novelty ties toward lower energy.
            net = max(
                remaining,
                key=lambda x: min(
                    (len(edge_sets[x] - ref) for ref in references),
                    default=2 * n,
                ),
            )
            selected.append(net)
            selected_set.add(net)
            references.append(edge_sets[net])

        # Sample a scale first, then a network. Otherwise the much larger
        # rotation neighborhood effectively monopolizes random exploration.
        strata = {}
        for net in ranked:
            if net in selected_set:
                continue
            removed = len(parent_set - edge_sets[net])
            scale = 0 if removed <= 2 else (1 if removed == 3 else 2)
            strata.setdefault(scale, []).append(net)
        if strata and len(selected) < target:
            scales = sorted(strata)
            scale = scales[int(rng.integers(len(scales)))]
            pool = strata[scale]
            selected.append(pool[int(rng.integers(len(pool)))])

        return selected[:target]
