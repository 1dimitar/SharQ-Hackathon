    def select_parent(pop):
        """Favor elites while directing exploration toward promising, underused regions."""
        if not pop:
            return best_net

        elite_n = min(6, len(pop))
        if rng.random() < 0.65 or elite_n == len(pop):
            parent = pop[int(rng.integers(elite_n))]
        else:
            # The exploration portion samples non-elites using quality, isolation,
            # visit rarity, and evidence that the region has yielded improvements.
            candidates_for_exploration = pop[elite_n:]
            weights = []
            for rank, net in enumerate(pop[elite_n:], start=elite_n):
                nearest = min(
                    (distance(net, other) for other in pop if other != net),
                    default=0,
                )
                isolation = 0.75 + 0.5 * min(1.0, nearest / max(1, 4 * n))
                rarity = 1.0 / ((1.0 + parent_visits.get(net, 0)) ** 0.5)
                promise = 1.0 + 0.5 * parent_gains.get(net, 0)
                quality = 1.0 / (1.0 + 0.08 * rank)
                weights.append(isolation * rarity * promise * quality)

            total = sum(weights)
            if total <= 0:
                parent = candidates_for_exploration[int(rng.integers(len(candidates_for_exploration)))]
            else:
                parent = candidates_for_exploration[
                    int(rng.choice(len(candidates_for_exploration), p=[w / total for w in weights]))
                ]

        parent_visits[parent] = parent_visits.get(parent, 0) + 1
        return parent
