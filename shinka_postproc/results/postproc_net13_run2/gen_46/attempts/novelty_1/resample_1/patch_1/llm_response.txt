<DESCRIPTION>
This design shifts the search from repeated small neighborhoods toward an adaptive mixture of guided switches, multi-switch perturbations, and repaired edge-set recombination. Recombination uses the repaired QAOA pool as a reservoir of alternate structures, while an LP-scored archive supplies stronger donors. Parent selection balances solution quality with under-exploration, and operator probabilities respond to which proposal types have recently produced improvements. LP calls are spent on a small, varied batch screened by the available surrogate rather than on a large neighborhood.

import json
import time

import numpy as np

def improve(demand, candidates, tools, rng):
    """Adaptive recombination and variable-neighborhood search for valid networks."""
    n = tools.n
    scores = {}
    best_net = None
    best_score = float("inf")
    visits = {}
    parent_gain = {}
    parent_stall = {}
    op_success = {"switch": 1.0, "burst": 1.0, "cross": 1.0}
    op_trials = {"switch": 1.0, "burst": 1.0, "cross": 1.0}

    def canon(arcs):
        return tuple(sorted(tuple(map(int, a)) for a in arcs))

    def dist(a, b):
        return len(set(a) ^ set(b))

    def energy(net):
        try:
            return float(tools.energy(net))
        except Exception:
            return 0.0

    def evaluate(net):
        nonlocal best_net, best_score
        net = canon(net)
        if net not in scores:
            scores[net] = tools.lp(net)
        value = scores[net]
        if value < best_score:
            best_score, best_net = value, net
        return value

    # Retain all repaired samples as recombination material; score a selective,
    # structurally varied subset rather than spending the initial budget on clones.
    reservoir = set()
    for candidate in candidates:
        try:
            net = canon(tools.repair(candidate))
            if tools.is_valid(net):
                reservoir.add(net)
        except Exception:
            pass

    if not reservoir:
        try:
            net = canon(tools.repair([]))
            if tools.is_valid(net):
                reservoir.add(net)
        except Exception:
            return []

    by_energy = sorted(reservoir, key=energy)
    seeds = []
    for net in by_energy:
        if all(dist(net, old) >= 4 for old in seeds):
            seeds.append(net)
        if len(seeds) >= 18:
            break
    for net in by_energy:
        if len(seeds) >= 25:
            break
        if net not in seeds:
            seeds.append(net)

    for net in seeds:
        if tools.lp_calls_left <= 0:
            break
        try:
            evaluate(net)
        except Exception as exc:
            if type(exc).__name__ == "BudgetExhausted":
                break

    if not scores:
        return list(seeds[0])

    def archive():
        ranked = sorted(scores, key=scores.get)
        chosen = []
        # First retain representatives from distinct topological regions.
        for net in ranked:
            if all(dist(net, old) >= 4 for old in chosen):
                chosen.append(net)
            if len(chosen) >= 22:
                break
        for net in ranked:
            if len(chosen) >= 36:
                break
            if net not in chosen:
                chosen.append(net)
        return chosen

    def choose_parent(pop):
        if len(pop) == 1:
            return pop[0]
        # Quality contributes, but seldom-used and productive parents remain eligible.
        weights = []
        for rank, net in enumerate(pop):
            q = 1.0 / (1.0 + 0.055 * rank)
            explore = 1.0 / ((1.0 + visits.get(net, 0)) ** 0.60)
            progress = 1.0 + min(2.0, parent_gain.get(net, 0) * 0.25)
            stale = 1.0 + min(1.5, parent_stall.get(net, 0) * 0.04)
            weights.append(q * (0.35 + 0.65 * explore) * progress * stale)
        probs = np.asarray(weights, dtype=float)
        probs /= probs.sum()
        parent = pop[int(rng.choice(len(pop), p=probs))]
        visits[parent] = visits.get(parent, 0) + 1
        return parent

    def switch(net, p, q):
        arcs = list(net)
        (a, b), (c, d) = arcs[p], arcs[q]
        if a == d or c == b:
            return None
        current = set(net)
        if (a, d) in current or (c, b) in current:
            return None
        current.remove((a, b))
        current.remove((c, d))
        current.add((a, d))
        current.add((c, b))
        return tuple(sorted(current))

    def get_hot(detail):
        if not detail:
            return set()
        hot = {
            tuple(map(int, arc))
            for arc, price in detail.get("prices", {}).items()
            if float(price) > 0
        }
        if not hot:
            loads = detail.get("loads", {})
            ranked = sorted(
                ((float(v), tuple(map(int, arc))) for arc, v in loads.items()),
                reverse=True,
            )
            hot = {arc for _, arc in ranked[:7]}
        return hot

    def switch_moves(parent, detail):
        arcs = list(parent)
        hot = get_hot(detail)
        moves = set()

        # Enumerate degree-preserving swaps touching a priced or heavily loaded arc.
        for p in range(len(arcs)):
            for q in range(p + 1, len(arcs)):
                if hot and arcs[p] not in hot and arcs[q] not in hot:
                    continue
                cand = switch(parent, p, q)
                if cand is not None:
                    moves.add(cand)

        # Unfocused samples protect against incomplete/degenerate dual guidance.
        for _ in range(45):
            p = int(rng.integers(len(arcs)))
            q = int(rng.integers(len(arcs) - 1))
            if q >= p:
                q += 1
            cand = switch(parent, p, q)
            if cand is not None:
                moves.add(cand)

        # A cyclic reassignment of three destinations is another valid local move.
        for _ in range(45):
            ids = rng.choice(len(arcs), size=3, replace=False)
            (a, b), (c, d), (e, f) = (arcs[int(i)] for i in ids)
            old = {(a, b), (c, d), (e, f)}
            if hot and not (old & hot):
                continue
            for new in (
                {(a, d), (c, f), (e, b)},
                {(a, f), (c, b), (e, d)},
            ):
                if any(x == y for x, y in new) or len(new) != 3:
                    continue
                if new & (set(parent) - old):
                    continue
                moves.add(tuple(sorted((set(parent) - old) | new)))

        return moves

    def burst_move(parent, detail):
        """Apply several valid switches in sequence to jump between basins."""
        cur = set(parent)
        steps = int(rng.integers(2, 7))
        hot = get_hot(detail)
        for _ in range(steps):
            arcs = list(cur)
            if len(arcs) < 2:
                break
            # Prefer touching a bottleneck, but sometimes make an unrestricted move.
            if hot and rng.random() < 0.70:
                eligible = [i for i, arc in enumerate(arcs) if arc in hot]
                p = int(rng.choice(eligible)) if eligible else int(rng.integers(len(arcs)))
            else:
                p = int(rng.integers(len(arcs)))
            q = int(rng.integers(len(arcs) - 1))
            if q >= p:
                q += 1
            a, b = arcs[p]
            c, d = arcs[q]
            if a == d or c == b or (a, d) in cur or (c, b) in cur:
                continue
            cur.remove((a, b))
            cur.remove((c, d))
            cur.add((a, d))
            cur.add((c, b))
        result = tuple(sorted(cur))
        return result if result != parent and tools.is_valid(result) else None

    def crossover(parent, pop):
        """Mix edge sets from two networks, then use the free repair operator."""
        if len(pop) < 2:
            donors = [x for x in reservoir if x != parent]
        else:
            donors = [x for x in pop if x != parent]
        if not donors:
            return None
        # Bias donors toward good archive members, with occasional pool diversity.
        if rng.random() < 0.70:
            donor = donors[int(rng.integers(min(8, len(donors))))]
        else:
            donor = donors[int(rng.integers(len(donors)))]

        pset, dset = set(parent), set(donor)
        common = pset & dset
        p_only = list(pset - common)
        d_only = list(dset - common)
        rng.shuffle(p_only)
        rng.shuffle(d_only)

        # Vary the inherited fractions to generate both conservative and novel offspring.
        keep_common = list(common)
        keep_p = p_only[:int(rng.integers(max(1, len(p_only) // 3), len(p_only) + 1))]
        keep_d = d_only[:int(rng.integers(max(1, len(d_only) // 3), len(d_only) + 1))]
        partial = keep_common + keep_p + keep_d

        try:
            child = canon(tools.repair(partial))
            if tools.is_valid(child) and child != parent:
                return child
        except Exception:
            pass
        return None

    def propose(parent, detail, pop):
        proposals = []
        moves = switch_moves(parent, detail)
        for net in moves:
            if net != parent and tools.is_valid(net):
                proposals.append(("switch", net))

        for _ in range(3):
            net = burst_move(parent, detail)
            if net is not None:
                proposals.append(("burst", net))

        for _ in range(2):
            net = crossover(parent, pop)
            if net is not None:
                proposals.append(("cross", net))

        # Deduplicate while keeping the originating operator for adaptive credit.
        unique = {}
        for op, net in proposals:
            if net not in scores and not tools.is_cached(net):
                unique.setdefault(net, op)
        return [(op, net) for net, op in unique.items()]

    def operator_choice():
        # Smoothed success rates, so every move family remains available.
        weights = []
        names = ("switch", "burst", "cross")
        for name in names:
            rate = (1.0 + op_success[name]) / (2.0 + op_trials[name])
            weights.append(0.25 + rate)
        p = np.asarray(weights, dtype=float)
        p /= p.sum()
        return names[int(rng.choice(len(names), p=p))]

    while tools.lp_calls_left > 0:
        pop = archive()
        parent = choose_parent(pop)
        try:
            detail = tools.lp_detail(parent)
        except Exception:
            detail = None

        proposed = propose(parent, detail, pop)
        if not proposed:
            # Recombination can still provide fresh material if a neighborhood is exhausted.
            for _ in range(12):
                child = crossover(parent, pop)
                if child is not None and child not in scores and not tools.is_cached(child):
                    proposed.append(("cross", child))
                    break

        if not proposed:
            child = burst_move(parent, detail)
            if child is not None and child not in scores and not tools.is_cached(child):
                proposed.append(("burst", child))

        if not proposed:
            break

        # Select a modest batch: surrogate-favored moves plus distinct exploratory moves.
        base_e = energy(parent)
        ranked = sorted(
            proposed,
            key=lambda item: (energy(item[1]) - base_e, rng.random()),
        )
        batch = []
        # Two screened proposals and one randomly selected tail proposal.
        for item in ranked:
            if all(dist(item[1], old[1]) >= 2 for old in batch):
                batch.append(item)
            if len(batch) >= 2:
                break
        tail = [item for item in proposed if item not in batch]
        if tail:
            op = operator_choice()
            preferred = [item for item in tail if item[0] == op]
            batch.append((preferred or tail)[int(rng.integers(len(preferred or tail)))])
        if len(batch) < 3:
            for item in ranked:
                if item not in batch:
                    batch.append(item)
                if len(batch) >= 3:
                    break

        before = best_score
        parent_score = scores.get(parent, float("inf"))
        for op, net in batch:
            if tools.lp_calls_left <= 0:
                break
            op_trials[op] += 1.0
            try:
                value = evaluate(net)
            except Exception as exc:
                if type(exc).__name__ == "BudgetExhausted":
                    return list(best_net) if best_net is not None else []
                continue
            if value < parent_score:
                op_success[op] += 1.0
                parent_gain[parent] = parent_gain.get(parent, 0) + 1
            else:
                parent_stall[parent] = parent_stall.get(parent, 0) + 1

        if best_score < before:
            parent_stall[parent] = 0

    return list(best_net) if best_net is not None else []

def run_postproc(pool_path, lp_budget, seed, control=False, time_limit=150):
    """Fixed entry point called by the evaluator: one run on one candidate pool."""
    import importlib.util
    import signal
    from pathlib import Path

    import oracle

    pool = json.load(open(pool_path))
    import qoblib_tools as qt

    if "hamiltonian_file" in pool:
        spec = importlib.util.spec_from_file_location(
            f"hamiltonian_{pool['hamiltonian_tag']}",
            Path(pool_path).resolve().parent / pool["hamiltonian_file"],
        )
        ham = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(ham)
        surrogate_terms = ham.surrogate_terms
    else:
        from hamiltonian import surrogate_terms

    n = int(pool["instance"][-2:])
    demand = qt.load_network_demand(n)
    linear, quadratic, _ = surrogate_terms(np.array(demand, dtype=float))
    tools = oracle.Tools(demand, lp_budget, (
        {tuple(map(int, a)): float(w) for a, w in linear.items()},
        {
            (tuple(map(int, a)), tuple(map(int, b))): float(w)
            for (a, b), w in quadratic.items()
        },
    ))
    candidates = [[tuple(map(int, a)) for a in c["arcs"]] for c in pool["candidates"]]
    if control:
        crng = np.random.default_rng(10_000 + seed)
        candidates = [qt.random_topology(n, crng) for _ in candidates]

    def _timeout(signum, frame):
        raise oracle.RunTimeout()

    start = time.perf_counter()
    returned, error, timed_out = None, None, False
    signal.signal(signal.SIGALRM, _timeout)
    signal.alarm(int(time_limit))
    try:
        returned = improve(demand.copy(), candidates, tools, np.random.default_rng(seed))
    except oracle.BudgetExhausted:
        pass
    except oracle.RunTimeout:
        timed_out = True
    except Exception as exc:
        error = f"{type(exc).__name__}: {exc}"
    finally:
        signal.alarm(0)

    best_score, best_arcs = tools.certified_best()
    if returned is not None:
        try:
            ret = [tuple(map(int, a)) for a in returned]
            if qt.is_valid_topology(n, ret):
                s = qt.route_min_congestion(demand, ret).objective
                if s < best_score:
                    best_score, best_arcs = s, tuple(sorted(ret))
        except Exception:
            pass

    return {
        "pool": pool_path,
        "control": control,
        "n": n,
        "error": error,
        "best": float(best_score),
        "arcs": [list(a) for a in best_arcs] if best_arcs else None,
        "lp_calls": tools.lp_calls_used,
        "seconds": time.perf_counter() - start,
        "timed_out": timed_out,
    }
