    def select_parent(pop):
        """Balance archive quality with novelty and under-explored parents."""
        if not pop:
            return best_net

        total_uses = sum(parent_uses.values())
        weights = []
        for rank, net in enumerate(pop):
            uses = parent_uses.get(net, 0)
            nearest = min(
                (distance(net, other) for other in pop if other != net),
                default=26,
            )
            novelty = min(1.0, nearest / 26.0)
            quality = 1.0 / (1.0 + 0.05 * rank)
            exploration = math.sqrt(
                math.log(total_uses + 2.0) / (uses + 1.0)
            )
            stall_factor = 1.0 / (1.0 + 0.06 * parent_stalls.get(net, 0))
            weights.append(
                quality * (0.65 + novelty) * (1.0 + 0.45 * exploration)
                * stall_factor
            )

        pick = float(rng.random()) * sum(weights)
        for net, weight in zip(pop, weights):
            pick -= weight
            if pick <= 0:
                return net
        return pop[-1]
