    def select_parent(pop):
        """Retain elite bias while adapting selection to parent productivity."""
        if not pop:
            return best_net

        # The original elite/uniform mixture is the baseline distribution.
        elite_n = min(6, len(pop))
        baseline = [
            (0.48 / elite_n if i < elite_n else 0.0)
            + 0.52 / len(pop)
            for i in range(len(pop))
        ]

        # Discount repeatedly visited parents with no improvement; reward
        # parents whose evaluations have improved their own incumbent.
        weights = []
        for net, base in zip(pop, baseline):
            visits = parent_visits.get(net, 0)
            stalls = parent_stalls.get(net, 0)
            successes = parent_successes.get(net, 0)
            productivity = (1.0 + 0.7 * successes) / (1.0 + 0.35 * visits + 0.65 * stalls)
            weights.append(base * productivity)

        total = sum(weights)
        if total <= 0:
            return pop[int(rng.integers(len(pop)))]
        return pop[int(rng.choice(len(pop), p=[w / total for w in weights]))]
