import math


def improve(demand, candidates, tools, rng):
    n = int(tools.n)
    m = 2 * n
    archive = {}                 # network -> LP score
    seen = set()
    best_net, best_score = None, float("inf")

    def canon(arcs):
        return tuple(sorted(tuple(map(int, e)) for e in arcs))

    def evaluate(net):
        nonlocal best_net, best_score
        net = canon(net)
        if net in seen or not tools.is_valid(net):
            return archive.get(net, float("inf"))
        seen.add(net)
        score = float(tools.lp(net))
        archive[net] = score
        if score < best_score:
            best_net, best_score = net, score
        return score

    # Seed on a compact, edge-diverse subset of the best-energy repaired samples.
    repaired = {}
    for cand in candidates:
        try:
            net = canon(tools.repair(cand))
            if tools.is_valid(net):
                repaired[net] = float(tools.energy(net))
        except Exception:
            continue

    ordered = sorted(repaired, key=repaired.get)
    seeds = []
    for net in ordered:
        if all(len(set(net) & set(old)) <= m - 3 for old in seeds):
            seeds.append(net)
        if len(seeds) >= 24:
            break
    for net in ordered:
        if len(seeds) >= 32:
            break
        if net not in seeds:
            seeds.append(net)
    if not seeds:
        seeds = [canon(tools.repair([]))]

    for net in seeds:
        if tools.lp_calls_left <= 0:
            break
        try:
            evaluate(net)
        except Exception:
            break
    if best_net is None:
        return list(seeds[0])

    def elite():
        return sorted(archive, key=archive.get)

    def trim():
        nonlocal archive
        ranked = elite()
        keep = []
        # A smaller archive reduces repeated expensive neighborhood generation.
        for net in ranked:
            if all(len(set(net) & set(old)) <= m - 3 for old in keep):
                keep.append(net)
            if len(keep) >= 20:
                break
        for net in ranked:
            if len(keep) >= 24:
                break
            if net not in keep:
                keep.append(net)
        archive = {x: archive[x] for x in keep}

    def proposals(net, use_hot=True, rotations=False):
        arcs = list(net)
        cur = set(net)
        try:
            detail = tools.lp_detail(net)
            prices = {tuple(k): float(v) for k, v in detail.get("prices", {}).items()}
            loads = {tuple(k): float(v) for k, v in detail.get("loads", {}).items()}
            score = max(1.0, float(detail.get("score", 1.0)))
        except Exception:
            prices, loads, score = {}, {}, 1.0

        hot = {e for e, p in prices.items() if p > 0.0}
        ranked = []

        # Two-edge destination switch.
        for p in range(m):
            a, b = arcs[p]
            for q in range(p + 1, m):
                c, d = arcs[q]
                e1, e2 = arcs[p], arcs[q]
                if use_hot and hot and e1 not in hot and e2 not in hot:
                    continue
                x, y = (a, d), (c, b)
                if a == d or c == b or x in cur or y in cur:
                    continue
                nxt = tuple(sorted((cur - {e1, e2}) | {x, y}))
                relief = sum(
                    max(0.0, prices.get(e, 0.0)) *
                    (0.5 + max(0.0, loads.get(e, 0.0)) / score)
                    for e in (e1, e2)
                )
                added = float(demand[a, d]) + float(demand[c, b])
                # Prices are the primary signal; demand guards against costly additions.
                rank = relief - added / max(1.0, float(demand.max()))
                ranked.append((rank, nxt))

        # Three-edge rotations selectively broaden the neighborhood on stagnation.
        if rotations:
            hot_idx = [i for i, e in enumerate(arcs) if e in hot]
            if len(hot_idx) < 2:
                hot_idx = list(range(m))
            # Bound the cubic neighborhood: prioritize triples touching bottlenecks.
            triples = []
            for p in hot_idx:
                for q in range(m):
                    if q == p:
                        continue
                    for r in range(q + 1, m):
                        if r == p:
                            continue
                        ids = tuple(sorted((p, q, r)))
                        if ids not in triples:
                            triples.append(ids)
                        if len(triples) >= 180:
                            break
                    if len(triples) >= 180:
                        break
                if len(triples) >= 180:
                    break
            for p, q, r in triples:
                (a, b), (c, d), (e, f) = arcs[p], arcs[q], arcs[r]
                # Rotate destinations: b->d, d->f, f->b.
                added = ((a, d), (c, f), (e, b))
                removed = (arcs[p], arcs[q], arcs[r])
                if any(u == v for u, v in added):
                    continue
                if len(set(added)) != 3 or any(x in cur for x in added):
                    continue
                nxt = tuple(sorted((cur - set(removed)) | set(added)))
                relief = sum(max(0.0, prices.get(e0, 0.0)) *
                             (0.5 + max(0.0, loads.get(e0, 0.0)) / score)
                             for e0 in removed)
                cost = sum(float(demand[u, v]) for u, v in added)
                ranked.append((relief - cost / max(1.0, float(demand.max())), nxt))

        # Cheap surrogate tie-break; keep only the best bounded proposal set.
        unique = {}
        for rank, nxt in ranked:
            if nxt == net or nxt in seen or tools.is_cached(nxt):
                continue
            if not tools.is_valid(nxt):
                continue
            if nxt not in unique or rank > unique[nxt]:
                unique[nxt] = rank
        result = list(unique)
        result.sort(key=lambda x: (
            -unique[x],
            float(tools.energy(x))
        ))
        return result

    iteration = 0
    stagnant = 0
    while tools.lp_calls_left > 0:
        trim()
        ranked = elite()
        if not ranked:
            break

        # Favor strong parents, but periodically sample deeper in the archive.
        if iteration % 5 == 4 and len(ranked) > 4:
            parent = ranked[int(rng.integers(min(12, len(ranked))))]
        else:
            parent = ranked[iteration % min(5, len(ranked))]

        # Rotation frequency ramps up only after repeated unsuccessful batches.
        rotate = stagnant >= 3
        moves = proposals(parent, use_hot=True, rotations=rotate)
        if len(moves) < 4:
            moves += proposals(parent, use_hot=False, rotations=rotate)

        # Small exploration component from another good archive member.
        if iteration % 4 == 0 and len(ranked) > 1:
            other = ranked[int(rng.integers(1, min(len(ranked), 10)))]
            moves += proposals(other, use_hot=True, rotations=rotate)

        # Deduplicate and score diverse top moves, with a small random tail.
        unique = list(dict.fromkeys(x for x in moves
                                    if x not in seen and not tools.is_cached(x)))
        if not unique:
            stagnant += 1
            iteration += 1
            continue

        batch_size = min(7, int(tools.lp_calls_left), len(unique))
        batch = unique[:batch_size]
        if len(unique) > batch_size and batch_size >= 4:
            tail = unique[batch_size:]
            count = min(1, len(tail))
            if count:
                batch.append(tail[int(rng.integers(len(tail)))])

        old_best = best_score
        for net in batch:
            if tools.lp_calls_left <= 0:
                break
            try:
                evaluate(net)
            except Exception:
                return list(best_net)

        if best_score < old_best:
            stagnant = 0
        else:
            stagnant += 1
        iteration += 1

    return list(best_net) if best_net is not None else []
# EVOLVE-BLOCK-END
