"""Classical post-processing of QAOA output for QOBLIB 08. ShinkaEvolve edits the EVOLVE block."""

import json
import time

import numpy as np

# EVOLVE-BLOCK-START
import math


def improve(demand, candidates, tools, rng):
    """Search diverse repaired seeds with dual-guided, surrogate-screened moves."""
    n = tools.n
    demand = np.asarray(demand, dtype=float)
    scores = {}
    details = {}
    archive = []
    best_net = None
    best_score = float("inf")
    stalls = {}
    global_stall = 0
    kick_size = 2

    def canon(arcs):
        return tuple(sorted(tuple(map(int, a)) for a in arcs))

    def distance(a, b):
        return len(set(a) ^ set(b))

    def evaluate(net):
        nonlocal best_net, best_score
        net = canon(net)
        if net not in scores:
            if tools.lp_calls_left <= 0:
                return float("inf")
            try:
                scores[net] = float(tools.lp(net))
            except Exception as exc:
                if type(exc).__name__ == "BudgetExhausted":
                    return float("inf")
                raise
        value = scores[net]
        if value < best_score:
            best_net, best_score = net, value
        return value

    def remember(net):
        """Maintain an elite set plus structurally distinct scored networks."""
        net = canon(net)
        if net not in scores or not math.isfinite(scores[net]) or net in archive:
            return
        archive.append(net)
        archive.sort(key=lambda x: scores[x])
        kept = archive[:5]
        for candidate in archive[5:]:
            if len(kept) >= 20:
                break
            if all(distance(candidate, old) >= 6 for old in kept):
                kept.append(candidate)
        for candidate in archive:
            if len(kept) >= 20:
                break
            if candidate not in kept:
                kept.append(candidate)
        archive[:] = kept

    # Repair all samples cheaply, then spend a small portion of the LP budget
    # on low-energy seeds while preserving topological diversity.
    repaired = set()
    for cand in candidates:
        try:
            net = canon(tools.repair(cand))
            if tools.is_valid(net):
                repaired.add(net)
        except Exception:
            continue
    if not repaired:
        try:
            net = canon(tools.repair([]))
            if tools.is_valid(net):
                repaired.add(net)
        except Exception:
            return []

    try:
        ordered = sorted(repaired, key=tools.energy)
    except Exception:
        ordered = list(repaired)

    seeds = []
    for net in ordered:
        if all(distance(net, old) >= 6 for old in seeds):
            seeds.append(net)
        if len(seeds) >= 14:
            break
    for net in ordered:
        if len(seeds) >= 26:
            break
        if net not in seeds:
            seeds.append(net)

    for net in seeds:
        if tools.lp_calls_left <= 0:
            break
        evaluate(net)
        remember(net)
    if best_net is None:
        return list(seeds[0]) if seeds else []

    def get_detail(net):
        if net not in details:
            try:
                details[net] = tools.lp_detail(net)
            except Exception:
                details[net] = {"prices": {}, "loads": {}}
        return details[net]

    def switch(arcs, current, p, q):
        (a, b), (c, d) = arcs[p], arcs[q]
        x, y = (a, d), (c, b)
        if a == d or c == b or x in current or y in current:
            return None
        return canon((current - {arcs[p], arcs[q]}) | {x, y})

    def make_moves(parent, detail, size):
        """Combine complete hot-link switches with sampled broad moves."""
        arcs = list(parent)
        current = set(parent)
        moves = set()
        raw_prices = detail.get("prices", {})
        prices = {tuple(map(int, a)): float(v) for a, v in raw_prices.items()}
        raw_loads = detail.get("loads", {})
        loads = {tuple(map(int, a)): float(v) for a, v in raw_loads.items()}

        hot = {a for a in arcs if prices.get(a, 0.0) > 0.0}
        if not hot:
            ranked_loads = sorted(arcs, key=lambda a: loads.get(a, 0.0), reverse=True)
            hot = set(ranked_loads[:max(5, n // 2)])
        else:
            hot.update(sorted(arcs, key=lambda a: (prices.get(a, 0.0), loads.get(a, 0.0)),
                              reverse=True)[:max(3, n // 4)])

        # Every legal switch touching a critical link is considered.
        for p in range(len(arcs)):
            for q in range(p + 1, len(arcs)):
                if arcs[p] in hot or arcs[q] in hot:
                    cand = switch(arcs, current, p, q)
                    if cand is not None and cand != parent:
                        moves.add(cand)

        # Sample non-hot switches to retain exploration without enumerating
        # the full O(m^2) neighborhood on every iteration.
        attempts = max(70, 24 * size)
        hot_ids = [i for i, a in enumerate(arcs) if a in hot]
        for _ in range(attempts):
            if len(arcs) < 2:
                break
            if hot_ids and rng.random() < 0.65:
                p = int(rng.choice(hot_ids))
            else:
                p = int(rng.integers(len(arcs)))
            q = int(rng.integers(len(arcs) - 1))
            if q >= p:
                q += 1
            cand = switch(arcs, current, p, q)
            if cand is not None and cand != parent:
                moves.add(cand)

        # Sample both orientations of three-edge rotations.
        rotation_trials = max(25, 16 * size)
        for _ in range(rotation_trials):
            if len(arcs) < 3:
                break
            ids = rng.choice(len(arcs), size=3, replace=False)
            old_arcs = [arcs[int(i)] for i in ids]
            old = set(old_arcs)
            if hot and not (old & hot) and rng.random() < 0.8:
                continue
            (a, b), (c, d), (e, f) = old_arcs
            outside = current - old
            for added in (((a, d), (c, f), (e, b)),
                          ((a, f), (c, b), (e, d))):
                new = set(added)
                if len(new) != 3 or any(x == y for x, y in new):
                    continue
                if new & outside:
                    continue
                cand = canon(outside | new)
                if cand != parent:
                    moves.add(cand)

        # A few larger kicks help leave a stalled basin.
        for _ in range(4):
            cur = set(parent)
            for _step in range(size):
                now = list(cur)
                if len(now) < 2:
                    break
                p, q = rng.choice(len(now), size=2, replace=False)
                (a, b), (c, d) = now[int(p)], now[int(q)]
                x, y = (a, d), (c, b)
                if a != d and c != b and x not in cur and y not in cur:
                    cur.remove((a, b))
                    cur.remove((c, d))
                    cur.add(x)
                    cur.add(y)
            cand = canon(cur)
            if cand != parent:
                moves.add(cand)

        fresh = [m for m in moves
                 if m not in scores and not tools.is_cached(m) and tools.is_valid(m)]
        return fresh, prices, hot

    def choose_parent():
        ranked = sorted(archive, key=lambda x: scores[x])
        if len(ranked) <= 1:
            return ranked[0] if ranked else best_net
        # Usually exploit the leading basin; periodically sample a diverse member.
        top = ranked[:min(7, len(ranked))]
        if len(ranked) > len(top) and rng.random() < 0.30:
            return ranked[int(rng.integers(len(top), len(ranked)))]
        weights = np.asarray([1.0 / ((i + 1) ** 0.75) for i in range(len(top))])
        return top[int(rng.choice(len(top), p=weights / weights.sum()))]

    def select_batch(moves, parent, prices, batch_size):
        if not moves:
            return []
        base = set(parent)
        base_energy = float(tools.energy(parent))
        rows = []
        for net in moves:
            nxt = set(net)
            removed, added = base - nxt, nxt - base
            dual = sum(prices.get(a, 0.0) for a in removed) - sum(
                prices.get(a, 0.0) for a in added
            )
            try:
                en = float(tools.energy(net))
            except Exception:
                en = base_energy
            rows.append((net, dual, en))

        dual_order = sorted(rows, key=lambda x: x[1], reverse=True)
        energy_order = sorted(rows, key=lambda x: x[2])
        dual_rank = {row[0]: i for i, row in enumerate(dual_order)}
        energy_rank = {row[0]: i for i, row in enumerate(energy_order)}
        blended = sorted(
            moves,
            key=lambda net: (0.55 * dual_rank[net] + 0.45 * energy_rank[net]
                             + 0.15 * float(rng.random())),
        )

        chosen = []
        used = set()
        for order, count in (([r[0] for r in dual_order], 2),
                             ([r[0] for r in energy_order], 3),
                             (blended, batch_size)):
            taken = 0
            for net in order:
                if net not in used:
                    chosen.append(net)
                    used.add(net)
                    taken += 1
                    if len(chosen) >= batch_size or taken >= count:
                        break

        # Add one random tail proposal to avoid over-trusting imperfect proxies.
        tail = [m for m in moves if m not in used]
        if tail and len(chosen) < batch_size and rng.random() < 0.75:
            chosen.append(tail[int(rng.integers(len(tail)))])
        return chosen[:batch_size]

    iteration = 0
    while tools.lp_calls_left > 0 and iteration < 10000:
        iteration += 1
        parent = choose_parent()
        if parent is None:
            break
        detail = get_detail(parent)
        moves, prices, hot = make_moves(parent, detail, kick_size)

        # If the focused neighborhood is exhausted, broaden the next attempt.
        if not moves:
            moves, prices, hot = make_moves(parent, {"prices": {}, "loads": {}},
                                             min(12, kick_size + 2))
        if not moves:
            global_stall += 1
            kick_size = min(12, kick_size + 1)
            continue

        batch = select_batch(moves, parent, prices, 8)
        before_best = best_score
        parent_score = scores.get(parent, best_score)
        evaluated = 0
        for net in batch:
            if tools.lp_calls_left <= 0:
                break
            value = evaluate(net)
            if math.isfinite(value):
                remember(net)
            evaluated += 1

        parent_stalls = stalls.get(parent, 0)
        batch_best = min((scores.get(net, float("inf")) for net in batch),
                         default=float("inf"))
        if batch_best < parent_score:
            stalls[parent] = 0
        else:
            stalls[parent] = parent_stalls + 1

        if best_score < before_best:
            global_stall = 0
            kick_size = 2
        else:
            global_stall += 1
            if global_stall % 4 == 0:
                kick_size = min(12, kick_size + 1)

    return list(best_net) if best_net is not None else []


# EVOLVE-BLOCK-END


def run_postproc(pool_path, lp_budget, seed, control=False, time_limit=150):
    """Fixed entry point called by the evaluator: one run on one candidate pool."""
    import signal

    import oracle
    from hamiltonian import surrogate_terms

    pool = json.load(open(pool_path))
    import qoblib_tools as qt

    n = int(pool["instance"][-2:])
    demand = qt.load_network_demand(n)
    linear, quadratic, _ = surrogate_terms(np.array(demand, dtype=float))
    tools = oracle.Tools(demand, lp_budget, (
        {tuple(map(int, a)): float(w) for a, w in linear.items()},
        {(tuple(map(int, a)), tuple(map(int, b))): float(w) for (a, b), w in quadratic.items()},
    ))
    candidates = [[tuple(map(int, a)) for a in c["arcs"]] for c in pool["candidates"]]
    if control:  # same number of random valid networks instead of QAOA output
        crng = np.random.default_rng(10_000 + seed)
        candidates = [qt.random_topology(n, crng) for _ in candidates]
    def _timeout(signum, frame):
        raise oracle.RunTimeout()

    start = time.perf_counter()
    returned, error, timed_out = None, None, False
    signal.signal(signal.SIGALRM, _timeout)
    signal.alarm(int(time_limit))
    try:
        returned = improve(demand.copy(), candidates, tools, np.random.default_rng(seed))
    except oracle.BudgetExhausted:
        pass
    except oracle.RunTimeout:
        timed_out = True
    except Exception as exc:  # noqa: BLE001
        error = f"{type(exc).__name__}: {exc}"
    finally:
        signal.alarm(0)
    best_score, best_arcs = tools.certified_best()
    if returned is not None:
        try:
            ret = [tuple(map(int, a)) for a in returned]
            if qt.is_valid_topology(n, ret):
                s = qt.route_min_congestion(demand, ret).objective
                if s < best_score:
                    best_score, best_arcs = s, tuple(sorted(ret))
        except Exception:  # noqa: BLE001
            pass
    return {
        "pool": pool_path, "control": control, "n": n, "error": error,
        "best": float(best_score), "arcs": [list(a) for a in best_arcs] if best_arcs else None,
        "lp_calls": tools.lp_calls_used, "seconds": time.perf_counter() - start, "timed_out": timed_out,
    }
