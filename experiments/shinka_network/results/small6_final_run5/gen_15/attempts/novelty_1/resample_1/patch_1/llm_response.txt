<DESCRIPTION>
This design uses a soft, demand-based routing proxy rather than rewarding a fixed number of two-hop paths in isolation. For each OD pair, it selects up to two low-pressure transit nodes, then estimates how much traffic those routes would expose each directed link to. That exposure informs both a modest link reward and sparse positive couplings between heavily exposed links sharing an endpoint, approximating local load balancing. Two-hop rewards remain, but are discounted by the estimated route pressure; a small reciprocal-link penalty discourages unproductive backtracking without forbidding useful mutual links.

"""Surrogate QUBO terms for QOBLIB 08 network design."""

import time

# EVOLVE-BLOCK-START
import numpy as np


def surrogate_terms(demand):
    """Build a sparse QUBO from soft route exposure and local load balance."""
    matrix = np.asarray(demand, dtype=float)
    n = int(matrix.shape[0])

    traffic = np.maximum(matrix, 0.0).copy()
    np.fill_diagonal(traffic, 0.0)
    max_demand = float(np.max(traffic)) if traffic.size else 0.0
    if max_demand <= 0.0:
        max_demand = 1.0
    q = traffic / max_demand
    root_q = np.sqrt(q)

    # A node with substantial total incident traffic is a less attractive
    # transit point. The imbalance term discourages routing through nodes
    # whose incoming and outgoing traffic profiles differ greatly.
    incoming = traffic.sum(axis=0)
    outgoing = traffic.sum(axis=1)
    incident = incoming + outgoing
    max_incident = float(np.max(incident)) if incident.size else 0.0
    if max_incident > 0.0:
        pressure = incident / max_incident
        imbalance = np.abs(incoming - outgoing) / max_incident
    else:
        pressure = np.zeros(n, dtype=float)
        imbalance = np.zeros(n, dtype=float)

    # Keep only the two least-loaded candidate intermediates per OD pair.
    # Store their route costs for a smooth allocation of proxy traffic.
    routes = {}
    exposure = traffic.copy()
    for i in range(n):
        for k in range(n):
            if i == k or traffic[i, k] <= 0.0:
                continue

            candidates = []
            for j in range(n):
                if j == i or j == k:
                    continue
                edge_pressure = q[i, j] + q[j, k]
                cost = (
                    0.58 * float(pressure[j])
                    + 0.24 * float(edge_pressure)
                    + 0.18 * float(imbalance[j])
                )
                candidates.append((cost, j, float(edge_pressure)))

            candidates.sort(key=lambda item: (item[0], item[1]))
            chosen = candidates[:2]
            if not chosen:
                continue
            routes[(i, k)] = chosen

            # Treat high-demand pairs as more likely to use a direct shortcut.
            # Remaining proxy flow is split over the two low-pressure routes.
            direct_share = 0.18 + 0.42 * float(root_q[i, k])
            indirect = float(traffic[i, k]) * (1.0 - direct_share)
            best_cost = chosen[0][0]
            weights = [
                np.exp(-4.0 * max(0.0, item[0] - best_cost))
                for item in chosen
            ]
            total_weight = float(sum(weights))
            if total_weight <= 0.0:
                weights = [1.0] * len(chosen)
                total_weight = float(len(chosen))
            for weight, (_, j, _) in zip(weights, chosen):
                share = indirect * float(weight) / total_weight
                exposure[i, j] += share
                exposure[j, k] += share

    max_exposure = float(np.max(exposure)) if exposure.size else 0.0
    if max_exposure <= 0.0:
        max_exposure = 1.0
    exposure_norm = exposure / max_exposure

    # Direct demand remains the principal link signal; projected route
    # exposure gives a smaller bonus to links useful as transit capacity.
    linear = {}
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            direct = 0.72 * float(q[i, j]) + 0.28 * float(root_q[i, j])
            transit = 0.16 * float(np.sqrt(exposure_norm[i, j]))
            linear[(i, j)] = -float(direct + transit)

    quadratic = {}

    # A pair of links forms a plausible two-hop route. Route rewards favor
    # useful OD demand but shrink when the intermediate or its links look
    # like likely congestion points.
    for (i, k), candidates in routes.items():
        support = 0.68 * float(q[i, k]) + 0.32 * float(root_q[i, k])
        best_cost = candidates[0][0]
        soft_weights = [
            np.exp(-4.0 * max(0.0, item[0] - best_cost))
            for item in candidates
        ]
        total_weight = float(sum(soft_weights))
        if total_weight <= 0.0:
            total_weight = 1.0

        for rank, ((cost, j, edge_pressure), soft_weight) in enumerate(
            zip(candidates, soft_weights)
        ):
            route_quality = 1.0 / (1.0 + 0.55 * cost + 0.20 * edge_pressure)
            share = float(soft_weight) / total_weight
            reward = 0.34 * support * route_quality * (0.72 + 0.28 * share)
            first, second = (i, j), (j, k)
            key = (first, second) if first < second else (second, first)
            quadratic[key] = quadratic.get(key, 0.0) - float(reward)

    # Local load-balance proxy: two highly exposed links leaving the same
    # node, or entering the same node, are mildly discouraged together.
    # This is sparse (O(n^3)) and acts only on pairs sharing an endpoint.
    balance_scale = 0.11
    for v in range(n):
        outgoing_links = [(v, j) for j in range(n) if j != v]
        incoming_links = [(i, v) for i in range(n) if i != v]

        for links in (outgoing_links, incoming_links):
            for a in range(len(links)):
                e1 = links[a]
                load1 = float(np.sqrt(exposure_norm[e1[0], e1[1]]))
                for b in range(a + 1, len(links)):
                    e2 = links[b]
                    load2 = float(np.sqrt(exposure_norm[e2[0], e2[1]]))
                    key = (e1, e2) if e1 < e2 else (e2, e1)
                    quadratic[key] = quadratic.get(key, 0.0) + (
                        balance_scale * load1 * load2
                    )

    # Reciprocal links often spend degree capacity on backtracking. Preserve
    # a discount when both directions have meaningful direct demand.
    mean_direct = (
        float(sum(-value for value in linear.values()))
        / max(1, n * (n - 1))
    )
    reciprocal_scale = 0.12 * mean_direct
    for i in range(n):
        for j in range(i + 1, n):
            mutual = min(float(q[i, j]), float(q[j, i]))
            key = ((i, j), (j, i))
            quadratic[key] = quadratic.get(key, 0.0) + (
                reciprocal_scale * (1.0 - 0.55 * mutual)
            )

    penalty_weight = min(1.48, 1.28 + 0.012 * max(0, n - 10))
    return linear, quadratic, float(penalty_weight)


# EVOLVE-BLOCK-END


def run_surrogate(demand):
    """Fixed entry point called by the evaluator, once per demand matrix."""
    demand = np.array(demand, dtype=float)
    start = time.perf_counter()
    linear, quadratic, penalty_weight = surrogate_terms(demand.copy())
    return {
        "n": len(demand),
        "linear": linear,
        "quadratic": quadratic,
        "penalty_weight": penalty_weight,
        "seconds": time.perf_counter() - start,
    }
