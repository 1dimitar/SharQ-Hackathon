<DESCRIPTION>
This version keeps direct-link rewards and two-hop route support, but limits a third route to high-demand pairs whose third-best route is close to the second-best. It adds sparse pairwise terms for outgoing and incoming links that estimate whether two selected neighbors provide complementary service to the node’s demand. The pair costs are decomposed into linear preferences and residual quadratic interactions, retaining only the strongest residuals to keep the model compact. A two-node cut correction remains as a separate load-capacity proxy. Demand is normalized before totals are computed, and the degree penalty is increased modestly to support feasible sampling.

"""Surrogate QUBO terms for QOBLIB 08 network design. ShinkaEvolve edits the EVOLVE block."""

import time

# EVOLVE-BLOCK-START
import numpy as np


def surrogate_terms(demand):
    """Build direct-link, selective-route, neighbor-pair, and cut terms."""
    traffic = np.asarray(demand, dtype=float)
    if traffic.ndim != 2 or traffic.shape[0] != traffic.shape[1]:
        raise ValueError("demand must be a square matrix")

    n = int(traffic.shape[0])
    traffic = np.nan_to_num(
        traffic, copy=True, nan=0.0, posinf=0.0, neginf=0.0
    )
    np.maximum(traffic, 0.0, out=traffic)
    np.fill_diagonal(traffic, 0.0)

    maximum = float(np.max(traffic)) if traffic.size else 0.0
    z = traffic / maximum if maximum > 0.0 else traffic.copy()

    row_sum = z.sum(axis=1)
    col_sum = z.sum(axis=0)
    row_share = z / np.maximum(row_sum[:, None], 1e-15)
    col_share = z / np.maximum(col_sum[None, :], 1e-15)

    incident = np.maximum(row_sum, col_sum)
    max_incident = float(np.max(incident)) if incident.size else 0.0
    if max_incident > 0.0:
        pressure = 0.48 + 0.72 * incident / max_incident
    else:
        pressure = np.ones(n, dtype=float)

    sqrt_z = np.sqrt(z)
    direct_score = (
        0.62 * np.power(z, 0.78)
        + 0.24 * sqrt_z
        + 0.14 * np.sqrt(np.maximum(row_share * col_share, 0.0))
    )
    linear_array = -direct_score.copy()
    quadratic = {}

    def add_pair(first, second, value):
        value = float(value)
        if abs(value) < 1e-14 or first == second:
            return
        key = (first, second) if first < second else (second, first)
        quadratic[key] = quadratic.get(key, 0.0) + value

    positive = z[z > 0.0]
    cutoff = float(np.quantile(positive, 0.68)) if positive.size else 0.0

    # Support for plausible two-hop paths, with a third path retained only
    # for high-demand pairs where it is nearly as attractive as route two.
    affinity = 0.56 * np.sqrt(np.maximum(row_share, 0.0)) + 0.44 * sqrt_z
    rank_pressure = pressure ** 0.88
    reward_pressure = pressure ** 0.35

    for i in range(n):
        for k in range(n):
            amount = float(z[i, k])
            if i == k or amount <= 0.0:
                continue

            demand_score = 0.68 * amount ** 0.74 + 0.32 * sqrt_z[i, k]
            support = float(demand_score * (1.0 - 0.32 * sqrt_z[i, k]))

            candidates = []
            for j in range(n):
                if j == i or j == k:
                    continue
                balance = float(np.sqrt(max(
                    affinity[i, j] * affinity[j, k], 0.0
                )))
                evidence = 0.20 * (sqrt_z[i, j] + sqrt_z[j, k])
                rank = (0.18 + balance + evidence) / rank_pressure[j]
                candidates.append((float(rank), j, balance))

            candidates.sort(key=lambda item: (-item[0], item[1]))
            keep = min(2, len(candidates))
            if (
                amount >= cutoff
                and len(candidates) > 2
                and candidates[2][0] >= 0.92 * candidates[1][0]
            ):
                keep = 3

            for position, (_, j, path_affinity) in enumerate(candidates[:keep]):
                rank_factor = 1.12 - 0.10 * position
                route_scale = (
                    support
                    * rank_factor
                    * (0.45 + 0.85 * path_affinity)
                    / reward_pressure[j]
                )
                first = (i, j)
                second = (j, k)
                add_pair(first, second, -0.37 * route_scale)

                # A mild redundancy cost avoids overvaluing a direct link
                # and the same demand's indirect route simultaneously.
                redundancy = (
                    0.085
                    * support
                    * (0.45 + 0.55 * sqrt_z[i, k])
                    * (0.25 + 0.75 * path_affinity)
                )
                direct = (i, k)
                add_pair(direct, first, redundancy)
                add_pair(direct, second, redundancy)

    # A demand-conditioned service profile estimates which destinations
    # are reachable from each possible next-hop neighbor.
    def service_profile(matrix, shares):
        local_affinity = 0.62 * np.sqrt(np.maximum(shares, 0.0))
        local_affinity += 0.38 * np.sqrt(np.maximum(matrix, 0.0))
        np.fill_diagonal(local_affinity, 0.0)

        transition = np.zeros_like(local_affinity)
        totals = local_affinity.sum(axis=1)
        for node in range(n):
            if totals[node] > 1e-15:
                transition[node] = local_affinity[node] / totals[node]
            elif n > 1:
                transition[node] = 1.0 / (n - 1)
                transition[node, node] = 0.0

        return np.eye(n) + 1.4 * transition + 0.7 * (transition @ transition)

    # For each node, estimate the pair cost of selecting two outgoing
    # neighbors and, separately, two incoming neighbors. On a valid
    # degree-two topology each node selects exactly one such pair.
    mean_direct = float(direct_score.sum()) / max(1, n * (n - 1))
    depth_scale = max(
        0.75, float(np.log2(max(n, 2)) / np.log2(10.0))
    )
    pair_budget = max(2, n // 3)

    if n >= 5 and mean_direct > 0.0:
        for reverse in (False, True):
            matrix = z.T if reverse else z
            shares = row_share.T if reverse else row_share
            service = service_profile(matrix, shares)
            totals = matrix.sum(axis=1)
            mean_total = float(totals.mean())

            if mean_total <= 1e-15:
                continue

            for node in range(n):
                total = float(totals[node])
                if total <= 1e-15:
                    continue

                neighbors = np.flatnonzero(np.arange(n) != node)
                m = len(neighbors)
                left, right = np.triu_indices(m, 1)
                weights = matrix[node] / total

                first_service = service[neighbors[left], :]
                second_service = service[neighbors[right], :]
                combined = np.maximum(first_service + second_service, 1e-12)

                coverage_cost = -np.log(combined) @ weights
                imbalance_cost = (
                    np.abs(first_service - second_service) / combined
                ) @ weights
                costs = coverage_cost + 0.38 * imbalance_cost

                grand_mean = float(costs.mean())
                pair_row_sum = np.zeros(m, dtype=float)
                np.add.at(pair_row_sum, left, costs)
                np.add.at(pair_row_sum, right, costs)
                row_mean = pair_row_sum / max(m - 1, 1)

                main_effect = (
                    (m - 1.0) / max(m - 2.0, 1.0)
                ) * (row_mean - grand_mean)
                residual = (
                    costs
                    - grand_mean
                    - main_effect[left]
                    - main_effect[right]
                )
                spread = float(np.sqrt(np.mean((costs - grand_mean) ** 2)))
                if spread <= 1e-12:
                    continue

                importance = float(np.clip(total / mean_total, 0.60, 1.60))
                strength = (
                    0.30 * mean_direct * depth_scale * importance
                    / max(0.07, spread)
                )

                def node_link(neighbor):
                    return (neighbor, node) if reverse else (node, neighbor)

                for position, neighbor in enumerate(neighbors):
                    link = node_link(int(neighbor))
                    linear_array[link] += strength * float(main_effect[position])

                residual_rms = float(np.sqrt(np.mean(residual * residual)))
                if residual_rms <= 1e-12:
                    continue

                order = sorted(
                    range(len(residual)),
                    key=lambda q: (
                        -abs(float(residual[q])),
                        int(left[q]),
                        int(right[q]),
                    ),
                )
                kept = 0
                for q in order:
                    value = float(residual[q])
                    if abs(value) < 0.80 * residual_rms:
                        break
                    a = node_link(int(neighbors[int(left[q])]))
                    b = node_link(int(neighbors[int(right[q])]))
                    add_pair(a, b, strength * value)
                    kept += 1
                    if kept >= pair_budget:
                        break

    # Approximate the capacity penalty of two-node cuts. For a valid
    # degree-two topology, the cut's boundary capacity depends on its
    # reciprocal link pair; remove endpoint offsets that are constant
    # across valid topologies.
    cut_linear = np.zeros((n, n), dtype=float)
    incident_scale = max_incident if max_incident > 0.0 else 1.0
    cut_strength = 0.80 / 144.0

    for i in range(n):
        for j in range(i + 1, n):
            internal = float(z[i, j] + z[j, i])
            external = max(
                0.0,
                max(
                    float(row_sum[i] + row_sum[j]),
                    float(col_sum[i] + col_sum[j]),
                ) - internal,
            )
            squared_load = (external / incident_scale) ** 2
            if squared_load <= 0.0:
                continue

            coefficient = cut_strength * squared_load
            cut_linear[i, j] = cut_linear[j, i] = 7.0 * coefficient
            add_pair((i, j), (j, i), 13.0 * coefficient)

    node_offset = cut_linear.sum(axis=1) / max(n - 1, 1)
    edge_offset = float(cut_linear.sum()) / max(n * (n - 1), 1)
    for i in range(n):
        for j in range(n):
            if i != j:
                linear_array[i, j] += (
                    cut_linear[i, j]
                    - node_offset[i]
                    - node_offset[j]
                    + edge_offset
                )

    linear = {
        (i, j): float(linear_array[i, j])
        for i in range(n)
        for j in range(n)
        if i != j
    }
    quadratic = {
        key: float(value)
        for key, value in quadratic.items()
        if abs(value) > 1e-12
    }

    return linear, quadratic, 1.25


# EVOLVE-BLOCK-END


def run_surrogate(demand):
    """Fixed entry point called by the evaluator, once per demand matrix."""
    demand = np.array(demand, dtype=float)
    start = time.perf_counter()
    linear, quadratic, penalty_weight = surrogate_terms(demand.copy())
    return {
        "n": len(demand),
        "linear": linear,
        "quadratic": quadratic,
        "penalty_weight": penalty_weight,
        "seconds": time.perf_counter() - start,
    }
