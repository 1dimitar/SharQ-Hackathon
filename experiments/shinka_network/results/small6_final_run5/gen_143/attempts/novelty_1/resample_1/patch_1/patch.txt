 def surrogate_terms(demand):
     """Build demand-aware direct-link and sparse two-hop QUBO terms."""
     traffic = np.asarray(demand, dtype=float)
     n = int(traffic.shape[0])
 
     traffic = np.maximum(np.nan_to_num(traffic, nan=0.0, posinf=0.0, neginf=0.0), 0.0)
     traffic = traffic.copy()
     np.fill_diagonal(traffic, 0.0)
 
     scale = float(np.max(traffic)) if traffic.size else 0.0
     if scale <= 0.0:
         scale = 1.0
     z = traffic / scale
 
     row_sum = traffic.sum(axis=1)
     col_sum = traffic.sum(axis=0)
     row_share = traffic / np.maximum(row_sum[:, None], 1e-12)
     col_share = traffic / np.maximum(col_sum[None, :], 1e-12)
 
     # Transit pressure is high for nodes that already generate or receive much
     # traffic. A floor avoids making any node categorically unusable.
     incident = np.maximum(row_sum, col_sum)
     max_incident = float(np.max(incident)) if incident.size else 0.0
     if max_incident > 0.0:
         pressure = 0.48 + 0.72 * incident / max_incident
     else:
         pressure = np.ones(n, dtype=float)
 
     # Direct-link score: soften the largest demands while giving a modest
     # bonus to links that account for a substantial share of both endpoints.
     share_bonus = np.sqrt(np.maximum(row_share * col_share, 0.0))
     direct_score = 0.62 * np.power(z, 0.78) + 0.24 * np.sqrt(z) + 0.14 * share_bonus
 
     linear = {
         (i, j): -float(direct_score[i, j])
         for i in range(n)
         for j in range(n)
         if i != j
     }
 
     positive = traffic[traffic > 0.0]
     if positive.size:
         cutoff = float(np.quantile(positive, 0.68))
     else:
         cutoff = 0.0
 
     quadratic = {}
 
     def add_pair(a, b, value):
         # Products are symmetric, so store one canonical key and aggregate
         # contributions should any terms coincide.
         key = (a, b) if a < b else (b, a)
         quadratic[key] = quadratic.get(key, 0.0) + float(value)
 
     for i in range(n):
         for k in range(n):
             if i == k or traffic[i, k] <= 0.0:
                 continue
 
             demand_score = 0.68 * z[i, k] ** 0.74 + 0.32 * np.sqrt(z[i, k])
             # A strong direct demand is already represented in the linear
             # score, so reduce (but do not remove) its indirect-path reward.
             indirect_factor = 1.0 - 0.32 * np.sqrt(z[i, k])
             support = float(demand_score * indirect_factor)
 
             candidates = []
             for j in range(n):
                 if j == i or j == k:
                     continue
 
                 # Prefer a transit node for which both legs have endpoint
                 # affinity, but discount nodes with high aggregate pressure.
                 first_affinity = 0.56 * np.sqrt(row_share[i, j]) + 0.44 * np.sqrt(z[i, j])
                 second_affinity = 0.56 * np.sqrt(col_share[j, k]) + 0.44 * np.sqrt(z[j, k])
                 balance = np.sqrt(max(first_affinity * second_affinity, 0.0))
                 edge_evidence = 0.20 * (np.sqrt(z[i, j]) + np.sqrt(z[j, k]))
                 endpoint_pressure = float(np.sqrt(pressure[i] * pressure[k]))
                 relative_pressure = float(np.clip(
                     endpoint_pressure / max(float(pressure[j]), 1e-12),
                     0.85,
                     1.15,
                 ))
                 rank = (
                     (0.18 + balance + edge_evidence)
                     / (pressure[j] ** 0.88)
                     * relative_pressure
                 )
                 candidates.append((float(rank), j))
 
             candidates.sort(key=lambda item: (-item[0], item[1]))
             # Two routes are the default. Add a third only when a substantial
             # demand has genuinely ambiguous alternatives.
             keep = min(2, len(candidates))
             if (
                 len(candidates) >= 3
                 and traffic[i, k] >= cutoff
                 and candidates[2][0] >= 0.93 * candidates[1][0]
             ):
                 keep = 3
 
             for position, (rank, j) in enumerate(candidates[:keep]):
                 rank_factor = 1.12 - 0.10 * position
                 route_scale = support * rank_factor / (pressure[j] ** 0.35)
                 first = (i, j)
                 second = (j, k)
 
                 # Negative coupling rewards completion of a plausible
                 # two-hop route.
                 route_reward = -0.37 * route_scale
                 add_pair(first, second, route_reward)
 
                 # A weak positive interaction models redundancy: when the
                 # direct link is present, both legs of this same alternative
                 # are less valuable specifically for this demand.
                 direct_redundancy = 0.085 * support * (0.45 + 0.55 * np.sqrt(z[i, k]))
                 add_pair((i, k), first, direct_redundancy)
                 add_pair((i, k), second, direct_redundancy)
 
     # Penalize reciprocal links that reduce boundary capacity for a
     # two-node set carrying substantial external traffic.
     cut_linear = np.zeros((n, n), dtype=float)
     incident_scale = max(max_incident, 1e-12)
     cut_strength = 0.80 / 144.0
 
     for i in range(n):
         for j in range(i + 1, n):
             internal = float(traffic[i, j] + traffic[j, i])
             external = max(
                 0.0,
                 max(
                     float(row_sum[i] + row_sum[j]),
                     float(col_sum[i] + col_sum[j]),
                 ) - internal,
             )
             if external <= 0.0:
                 continue
 
             # Scale conservatively according to how exposed each endpoint
             # is to external traffic, clipping outlier ratios.
             endpoint_ratio = 0.5 * (
                 external / max(float(incident[i]), 1e-12)
                 + external / max(float(incident[j]), 1e-12)
             )
             vulnerability = float(np.clip(
                 0.85 + 0.30 * endpoint_ratio, 0.85, 1.15
             ))
             squared_load = (external / incident_scale) ** 2
             coefficient = cut_strength * squared_load * vulnerability
 
             # 1/(4-x-y)^2 = 1/16 + 7(x+y)/144 + 13xy/144.
             cut_linear[i, j] = cut_linear[j, i] = 7.0 * coefficient
             add_pair((i, j), (j, i), 13.0 * coefficient)
 
     # Remove endpoint and global offsets: these are constant over valid
     # directed degree-two topologies and avoid adding a degree bias.
     if n > 1:
         node_offset = cut_linear.sum(axis=1) / (n - 1)
         edge_offset = float(cut_linear.sum()) / (n * (n - 1))
         for i, j in linear:
             linear[(i, j)] += float(
                 cut_linear[i, j]
                 - node_offset[i]
                 - node_offset[j]
                 + edge_offset
             )
 
+    # Reward pairs of neighbors whose onward service complements one another.
+    # Profiles estimate one-step service from each possible neighbor; the
+    # diagonal represents destinations already reached at that neighbor.
+    mean_direct = float(direct_score.sum()) / max(1, n * (n - 1))
+    service_strength = 0.16 * mean_direct
+    service_budget = max(1, n // 3)
+
+    for reverse in (False, True):
+        scores = direct_score.T if reverse else direct_score
+        demands = traffic.T if reverse else traffic
+        profiles = np.minimum(
+            1.0,
+            2.0 * scores
+            / np.maximum(scores.sum(axis=1, keepdims=True), 1e-12),
+        )
+        np.fill_diagonal(profiles, 1.0)
+
+        totals = demands.sum(axis=1)
+        mean_total = float(totals.mean())
+        if mean_total <= 1e-12:
+            continue
+
+        for node in range(n):
+            total = float(totals[node])
+            if total <= 1e-12:
+                continue
+
+            weights = demands[node] / total
+            neighbors = [a for a in range(n) if a != node]
+            local_profiles = profiles[neighbors]
+            candidates = []
+
+            for a_pos in range(len(neighbors)):
+                first_profile = local_profiles[a_pos]
+                for b_pos in range(a_pos + 1, len(neighbors)):
+                    second_profile = local_profiles[b_pos]
+                    union = (
+                        first_profile + second_profile
+                        - first_profile * second_profile
+                    )
+                    gain = float(
+                        weights @ (union - np.maximum(first_profile, second_profile))
+                    )
+                    if gain > 1e-12:
+                        candidates.append((
+                            gain, neighbors[a_pos], neighbors[b_pos]
+                        ))
+
+            if not candidates:
+                continue
+
+            candidates.sort(key=lambda item: (-item[0], item[1], item[2]))
+            best_gain = candidates[0][0]
+            importance = float(np.clip(total / mean_total, 0.60, 1.40))
+            strength = service_strength * importance
+
+            for gain, a, b in candidates[:service_budget]:
+                first = (a, node) if reverse else (node, a)
+                second = (b, node) if reverse else (node, b)
+                add_pair(
+                    first,
+                    second,
+                    -strength * min(1.0, gain / max(best_gain, 1e-12)),
+                )

     return linear, quadratic, 1.15

 def run_surrogate(demand):
     """Fixed entry point called by the evaluator, once per demand matrix."""
     demand = np.array(demand, dtype=float)
     start = time.perf_counter()
     linear, quadratic, penalty_weight = surrogate_terms(demand.copy())
     return {
         "n": len(demand),
         "linear": linear,
         "quadratic": quadratic,
         "penalty_weight": penalty_weight,
         "seconds": time.perf_counter() - start,
     }
