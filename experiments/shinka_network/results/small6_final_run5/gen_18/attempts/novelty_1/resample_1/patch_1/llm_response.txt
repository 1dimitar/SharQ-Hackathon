<DESCRIPTION>
This replaces route-by-route two-hop rewards with a static soft-routing load estimate. Demand is softly assigned across possible intermediate nodes to estimate which links may become heavily loaded; linear costs and sparse pairwise penalties then discourage selecting multiple high-load links with the same source or destination. The resulting couplings are compact and organized around node-local link groups, aiming to give the sampler and post-processor a more useful load-balancing signal without flooding the QUBO with path terms.
</DESCRIPTION>

"""Surrogate QUBO terms for QOBLIB 08 network design."""

import time

# EVOLVE-BLOCK-START
import numpy as np


def surrogate_terms(demand):
    """Build a link-load proxy and node-local load-balancing couplings."""
    traffic = np.asarray(demand, dtype=float)
    n = len(traffic)

    traffic = np.maximum(traffic, 0.0).copy()
    np.fill_diagonal(traffic, 0.0)

    maximum = float(traffic.max()) if traffic.size else 0.0
    if maximum <= 0.0:
        linear = {
            (i, j): 0.0
            for i in range(n)
            for j in range(n)
            if i != j
        }
        return linear, {}, 1.30

    t = traffic / maximum
    root = np.sqrt(t)

    row = traffic.sum(axis=1)
    col = traffic.sum(axis=0)
    row_max = max(float(row.max()), 1e-12)
    col_max = max(float(col.max()), 1e-12)

    # Relative node pressure is used only to spread the soft routing proxy
    # away from already busy transit nodes.
    node_load = np.maximum(row, col)
    node_load_max = max(float(node_load.max()), 1e-12)
    pressure = np.maximum(0.55, node_load / node_load_max)

    # Begin with direct traffic, then add a modest estimate of traffic that
    # could use each edge as one leg of a two-hop route. The route assignment
    # is soft and normalized per OD pair; it does not presume a particular
    # topology or select a single intermediate node.
    estimated_load = t.copy()
    floor = 0.025

    for i in range(n):
        for k in range(n):
            amount = float(t[i, k])
            if i == k or amount <= 0.0:
                continue

            intermediates = []
            total = 0.0
            for j in range(n):
                if j == i or j == k:
                    continue

                a = float(t[i, j])
                b = float(t[j, k])
                evidence = (
                    0.10
                    + np.sqrt((a + floor) * (b + floor))
                    + 0.20 * (a + b)
                )
                score = float(evidence / (pressure[j] ** 0.55))
                intermediates.append((j, score))
                total += score

            if total > 0.0:
                transit_amount = 0.30 * amount
                for j, score in intermediates:
                    share = transit_amount * score / total
                    estimated_load[i, j] += share
                    estimated_load[j, k] += share

    load_max = max(float(estimated_load.max()), 1e-12)
    load = np.minimum(1.0, estimated_load / load_max)

    # Favor useful direct links, while lightly charging links whose predicted
    # direct-plus-transit load is high.
    linear = {}
    for i in range(n):
        for j in range(n):
            if i == j:
                continue

            endpoint_need = 0.5 * (
                row[i] / row_max + col[j] / col_max
            )
            direct_reward = 0.78 * t[i, j] + 0.22 * root[i, j]
            load_charge = 0.075 * float(load[i, j] ** 0.8)
            linear[(i, j)] = float(
                -direct_reward * (0.96 + 0.04 * endpoint_need)
                + load_charge
            )

    quadratic = {}

    def add_pair(a, b, weight):
        if a == b or abs(weight) < 1e-14:
            return
        key = (a, b) if a < b else (b, a)
        quadratic[key] = quadratic.get(key, 0.0) + float(weight)

    # In a valid topology each node chooses exactly two links from each
    # group. Penalize pairs that make either its outgoing or incoming pair
    # unusually concentrated in predicted load.
    for v in range(n):
        outgoing = [(v, w) for w in range(n) if w != v]
        incoming = [(u, v) for u in range(n) if u != v]

        source_factor = 0.65 + 0.35 * float(row[v] / row_max)
        target_factor = 0.65 + 0.35 * float(col[v] / col_max)

        for a in range(len(outgoing)):
            e1 = outgoing[a]
            l1 = float(load[e1[0], e1[1]]) ** 0.65
            for b in range(a + 1, len(outgoing)):
                e2 = outgoing[b]
                l2 = float(load[e2[0], e2[1]]) ** 0.65
                add_pair(e1, e2, 0.11 * source_factor * l1 * l2)

        for a in range(len(incoming)):
            e1 = incoming[a]
            l1 = float(load[e1[0], e1[1]]) ** 0.65
            for b in range(a + 1, len(incoming)):
                e2 = incoming[b]
                l2 = float(load[e2[0], e2[1]]) ** 0.65
                add_pair(e1, e2, 0.11 * target_factor * l1 * l2)

    # Number of couplings is at most n*(n-1)*(n-2), well below the limit.
    return linear, quadratic, 1.30


# EVOLVE-BLOCK-END


def run_surrogate(demand):
    """Fixed entry point called by the evaluator, once per demand matrix."""
    demand = np.array(demand, dtype=float)
    start = time.perf_counter()
    linear, quadratic, penalty_weight = surrogate_terms(demand.copy())
    return {
        "n": len(demand),
        "linear": linear,
        "quadratic": quadratic,
        "penalty_weight": penalty_weight,
        "seconds": time.perf_counter() - start,
    }
