<DESCRIPTION>
Separates demand preparation, route selection, and QUBO assembly into small stages. It retains the direct-link reward and sparse two-hop route terms, while selecting alternatives using both endpoint affinity and transit-node pressure. The bounded route selection keeps the coupling count below the required limit and makes the construction easier to tune without changing the interface.
</DESCRIPTION>

"""Surrogate QUBO terms for QOBLIB 08 network design."""

import time

# EVOLVE-BLOCK-START
import numpy as np


def _prepare_demand(demand):
    """Return nonnegative, zero-diagonal demand and normalized features."""
    traffic = np.maximum(np.asarray(demand, dtype=float), 0.0).copy()
    n = traffic.shape[0]
    np.fill_diagonal(traffic, 0.0)

    scale = float(np.max(traffic)) if traffic.size else 0.0
    if scale <= 0.0:
        scale = 1.0

    normalized = traffic / scale
    roots = np.sqrt(normalized)

    # Transit pressure reflects both traffic entering and leaving each node.
    incident = np.maximum(traffic.sum(axis=0), traffic.sum(axis=1))
    max_incident = float(np.max(incident)) if incident.size else 0.0
    if max_incident <= 0.0:
        max_incident = 1.0
    pressure = np.maximum(0.65, incident / max_incident)

    positive = traffic[traffic > 0.0]
    cutoff = float(np.quantile(positive, 0.75)) if positive.size else 0.0
    return traffic, normalized, roots, pressure, cutoff


def _add_route(quadratic, i, j, k, weight):
    """Accumulate a two-edge route interaction under a canonical key."""
    first, second = (i, j), (j, k)
    key = (first, second) if first < second else (second, first)
    quadratic[key] = quadratic.get(key, 0.0) + float(weight)


def _build_route_terms(traffic, normalized, roots, pressure, cutoff):
    """Choose a small set of pressure-aware two-hop alternatives per demand."""
    n = traffic.shape[0]
    quadratic = {}

    for i in range(n):
        for k in range(n):
            amount = traffic[i, k]
            if i == k or amount <= 0.0:
                continue

            support = 0.70 * normalized[i, k] + 0.30 * roots[i, k]
            candidates = []

            for j in range(n):
                if j == i or j == k:
                    continue

                edge_pressure = normalized[i, j] + normalized[j, k]
                transit_pressure = float(pressure[j])
                affinity = 0.25 + edge_pressure
                rank = affinity / (transit_pressure ** 0.75)
                discount = transit_pressure * (1.0 + 0.5 * edge_pressure)
                candidates.append((float(rank), j, float(discount)))

            # Strong demands get a wider menu; weak demands keep the model sparse.
            candidates.sort(key=lambda item: (-item[0], item[1]))
            keep = min(len(candidates), 4 if amount >= cutoff else 2)
            selected = set()

            for position, (_, j, discount) in enumerate(candidates[:keep]):
                selected.add(j)
                rank_factor = 1.0 + 0.08 * (keep - position)
                weight = -0.46 * float(support) * rank_factor / discount
                _add_route(quadratic, i, j, k, weight)

            # Preserve one distinct alternative with a favorable pressure discount.
            remaining = [item for item in candidates if item[1] not in selected]
            if remaining:
                _, j, discount = min(remaining, key=lambda item: (item[2], item[1]))
                _add_route(
                    quadratic,
                    i,
                    j,
                    k,
                    -0.30 * float(support) / discount,
                )

    return quadratic


def surrogate_terms(demand):
    """Build direct-link rewards and sparse, pressure-aware route couplings."""
    traffic, normalized, roots, pressure, cutoff = _prepare_demand(demand)
    n = traffic.shape[0]

    # Blend linear and concave rewards so medium-sized demands remain relevant.
    linear = {
        (i, j): -float(0.75 * normalized[i, j] + 0.25 * roots[i, j])
        for i in range(n)
        for j in range(n)
        if i != j
    }

    quadratic = _build_route_terms(
        traffic, normalized, roots, pressure, cutoff
    )
    return linear, quadratic, 1.25


# EVOLVE-BLOCK-END


def run_surrogate(demand):
    """Fixed entry point called by the evaluator."""
    demand = np.array(demand, dtype=float)
    start = time.perf_counter()
    linear, quadratic, penalty_weight = surrogate_terms(demand.copy())
    return {
        "n": len(demand),
        "linear": linear,
        "quadratic": quadratic,
        "penalty_weight": penalty_weight,
        "seconds": time.perf_counter() - start,
    }
