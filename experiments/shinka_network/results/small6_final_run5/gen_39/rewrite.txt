"""Surrogate QUBO terms for QOBLIB 08 network design."""

import time

# EVOLVE-BLOCK-START
import numpy as np


def _prepare_demand(demand):
    """Return sanitized, normalized demand and node pressure estimates."""
    traffic = np.asarray(demand, dtype=float)
    traffic = np.nan_to_num(traffic, nan=0.0, posinf=0.0, neginf=0.0)
    traffic = np.maximum(traffic, 0.0).copy()
    n = traffic.shape[0]
    np.fill_diagonal(traffic, 0.0)

    scale = float(np.max(traffic)) if traffic.size else 0.0
    if scale <= 0.0:
        scale = 1.0
    normalized = traffic / scale
    roots = np.sqrt(normalized)

    incident = np.maximum(traffic.sum(axis=0), traffic.sum(axis=1))
    largest_incident = float(np.max(incident)) if incident.size else 0.0
    if largest_incident > 0.0:
        pressure = np.maximum(0.65, incident / largest_incident)
    else:
        pressure = np.full(n, 0.65, dtype=float)

    return traffic, normalized, roots, pressure


def _direct_link_terms(normalized, roots):
    """Reward links that directly serve substantial demand."""
    n = normalized.shape[0]
    return {
        (i, j): -float(0.75 * normalized[i, j] + 0.25 * roots[i, j])
        for i in range(n)
        for j in range(n)
        if i != j
    }


def _pair_key(first, second):
    """Canonicalize an undirected QUBO coupling key."""
    return (first, second) if first < second else (second, first)


def _route_terms(traffic, normalized, roots, pressure):
    """Build sparse two-hop rewards with bounded constituent-link support."""
    n = normalized.shape[0]
    positive = traffic[traffic > 0.0]
    cutoff = float(np.quantile(positive, 0.75)) if positive.size else 0.0
    quadratic = {}

    for source in range(n):
        for target in range(n):
            endpoint_demand = float(traffic[source, target])
            if source == target or endpoint_demand <= 0.0:
                continue

            endpoint_support = (
                0.70 * normalized[source, target]
                + 0.30 * roots[source, target]
            )
            candidates = []

            for transit in range(n):
                if transit == source or transit == target:
                    continue

                first_support = float(normalized[source, transit])
                second_support = float(normalized[transit, target])
                leg_sum = first_support + second_support

                # A floor preserves some value for routes whose links are not
                # themselves high-demand, while the geometric mean discounts
                # paths with one or two weakly supported legs.
                leg_confidence = 0.30 + 0.70 * np.sqrt(
                    first_support * second_support
                )
                node_pressure = float(pressure[transit])
                affinity = (
                    0.25
                    + 0.55 * np.sqrt(first_support * second_support)
                    + 0.15 * leg_sum
                )
                rank = affinity / (node_pressure ** 0.75)
                discount = node_pressure * (1.0 + 0.35 * leg_sum)
                candidates.append(
                    (
                        float(rank),
                        transit,
                        float(discount),
                        float(leg_confidence),
                    )
                )

            candidates.sort(reverse=True)
            if not candidates:
                continue

            base_keep = min(
                len(candidates),
                4 if endpoint_demand >= cutoff else 2,
            )
            keep = base_keep
            leading_rank = candidates[0][0]

            # Prune clearly inferior routes, and only expand a close boundary.
            while (
                keep > 1
                and candidates[keep - 1][0] < 0.70 * leading_rank
            ):
                keep -= 1

            if keep == base_keep and keep < len(candidates):
                if (
                    candidates[keep][0] >= 0.92 * candidates[keep - 1][0]
                    and candidates[keep][0] >= 0.78 * leading_rank
                ):
                    keep += 1

            reward_scale = min(1.0, (float(base_keep) / keep) ** 0.35)
            selected = set()

            for position, (_, transit, discount, leg_confidence) in enumerate(
                candidates[:keep]
            ):
                selected.add(transit)
                rank_factor = 1.0 + 0.08 * max(1, base_keep - position)
                weight = (
                    -0.46
                    * float(endpoint_support)
                    * leg_confidence
                    * rank_factor
                    * reward_scale
                    / discount
                )
                first = (source, transit)
                second = (transit, target)
                key = _pair_key(first, second)
                quadratic[key] = quadratic.get(key, 0.0) + float(weight)

            # Add one low-pressure alternative, but apply the same weak-leg
            # discount so an unsupported fallback cannot dominate the energy.
            remaining = [
                candidate
                for candidate in candidates
                if candidate[1] not in selected
            ]
            if remaining:
                _, transit, discount, leg_confidence = min(
                    remaining,
                    key=lambda item: (item[2], item[1]),
                )
                first = (source, transit)
                second = (transit, target)
                key = _pair_key(first, second)
                fallback = (
                    -0.30
                    * float(endpoint_support)
                    * leg_confidence
                    / discount
                )
                quadratic[key] = quadratic.get(key, 0.0) + float(fallback)

    return quadratic


def _structural_terms(normalized, linear, quadratic):
    """Add compact reciprocal-link and demand-overlap adjustments."""
    n = normalized.shape[0]
    mean_direct = -float(sum(linear.values())) / max(1, n * (n - 1))
    depth_scale = max(
        0.75,
        float(np.log2(max(n, 2))) / float(np.log2(10.0)),
    )

    # Discourage reciprocal links unless mutual demand makes them useful.
    backtrack_weight = 0.65 * mean_direct * depth_scale
    if backtrack_weight > 0.0:
        for i in range(n):
            for j in range(i + 1, n):
                mutual = min(
                    float(normalized[i, j]),
                    float(normalized[j, i]),
                )
                key = ((i, j), (j, i))
                quadratic[key] = quadratic.get(key, 0.0) + float(
                    backtrack_weight * (1.0 - 0.45 * mutual)
                )

    # Penalize a small set of highly redundant neighbors.  Keeping this
    # budget modest limits coupling growth while retaining a diversity signal.
    diversity_scale = 0.40 * mean_direct * depth_scale
    pair_budget = max(1, n // 4)

    if diversity_scale > 0.0:
        for reverse in (False, True):
            profiles = normalized.T if reverse else normalized
            for node in range(n):
                weights = (
                    normalized[:, node].copy()
                    if reverse
                    else normalized[node, :].copy()
                )
                total = float(weights.sum())
                if total <= 0.0:
                    continue
                weights /= total

                coverage = profiles @ weights
                neighbors = [v for v in range(n) if v != node]
                best_coverage = max(
                    (float(coverage[v]) for v in neighbors),
                    default=0.0,
                )
                if best_coverage <= 1e-15:
                    continue

                overlaps = []
                for pos, first_neighbor in enumerate(neighbors):
                    first_profile = profiles[first_neighbor]
                    for second_neighbor in neighbors[pos + 1:]:
                        second_profile = profiles[second_neighbor]
                        overlap = float(
                            np.dot(
                                np.minimum(first_profile, second_profile),
                                weights,
                            )
                        ) / best_coverage
                        if overlap > 0.40:
                            overlaps.append(
                                (overlap, first_neighbor, second_neighbor)
                            )

                overlaps.sort(reverse=True)
                for overlap, first_neighbor, second_neighbor in overlaps[
                    :pair_budget
                ]:
                    if reverse:
                        first = (first_neighbor, node)
                        second = (second_neighbor, node)
                    else:
                        first = (node, first_neighbor)
                        second = (node, second_neighbor)

                    key = _pair_key(first, second)
                    quadratic[key] = quadratic.get(key, 0.0) + float(
                        diversity_scale * overlap
                    )


def surrogate_terms(demand):
    """Return direct costs, sparse route couplings, and degree penalty."""
    traffic, normalized, roots, pressure = _prepare_demand(demand)
    linear = _direct_link_terms(normalized, roots)
    quadratic = _route_terms(traffic, normalized, roots, pressure)
    _structural_terms(normalized, linear, quadratic)

    n = normalized.shape[0]
    penalty_weight = min(1.45, 1.30 + 0.008 * max(0, n - 10))
    return linear, quadratic, float(penalty_weight)


# EVOLVE-BLOCK-END


def run_surrogate(demand):
    """Fixed entry point called by the evaluator, once per demand matrix."""
    demand = np.array(demand, dtype=float)
    start = time.perf_counter()
    linear, quadratic, penalty_weight = surrogate_terms(demand.copy())
    return {
        "n": len(demand),
        "linear": linear,
        "quadratic": quadratic,
        "penalty_weight": penalty_weight,
        "seconds": time.perf_counter() - start,
    }
