+def _prepare(demand):
+    """Normalize nonnegative demand and compute reusable node/link statistics."""
+    traffic = np.asarray(demand, dtype=float)
+    traffic = np.nan_to_num(traffic, nan=0.0, posinf=0.0, neginf=0.0)
+    traffic = np.maximum(traffic, 0.0).copy()
+    np.fill_diagonal(traffic, 0.0)

+    n = traffic.shape[0]
+    maximum = float(traffic.max()) if traffic.size else 0.0
+    if maximum > 0.0:
+        normalized = traffic / maximum

+        normalized = traffic.copy()

+    cols = traffic.sum(axis=0)

+def _estimate_transit_pressure(normalized, pressure):
+    """Add a modest transit-load estimate to incident-demand pressure."""
+    n = normalized.shape[0]
+    transit = np.zeros(n, dtype=float)
+
+    for source in range(n):
+        for target in range(n):
+            demand = float(normalized[source, target])
+            if source == target or demand <= 0.0:
+                continue
+
+            choices = []
+            for middle in range(n):
+                if middle == source or middle == target:
+                    continue
+                first = float(normalized[source, middle])
+                second = float(normalized[middle, target])
+                evidence = first + second
+                if evidence <= 0.02:
+                    continue
+                balanced = np.sqrt(first * second)
+                rank = (0.15 + 0.55 * balanced + 0.15 * evidence)
+                rank /= float(pressure[middle]) ** 0.7
+                choices.append((rank, middle, evidence))
+
+            if not choices:
+                continue
+
+            choices.sort(key=lambda item: (-item[0], item[1]))
+            selected = choices[:2]
+            strengths = [
+                rank * (0.15 + evidence)
+                for rank, _, evidence in selected
+            ]
+            total = sum(strengths)
+            if total <= 0.0:
+                continue
+
+            amount = demand * (0.35 + 0.65 * (1.0 - demand))
+            for (_, middle, _), strength in zip(selected, strengths):
+                transit[middle] += amount * strength / total
+
+    projected = pressure + 0.30 * transit / max(float(transit.max()), 1.0)
+    return np.clip(projected, 0.65, 1.8)

+    quadratic[key] = quadratic.get(key, 0.0) + float(weight)
