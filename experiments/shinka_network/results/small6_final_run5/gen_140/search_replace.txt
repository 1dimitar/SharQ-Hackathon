<DESCRIPTION>
Add sparse pairwise factors for the two outgoing and two incoming links at each node. These factors estimate whether a pair of neighbors provides complementary service to the node’s traffic, while also discouraging an uneven split of that traffic. Decomposing the estimated pair cost into linear endpoint effects and a sparse residual lets the QUBO capture this signal without adding a dense set of couplings. The budget of `n // 4` pairs per node and direction keeps the interactions compact; the strength is scaled by average direct-link score and node traffic.

def _add_service_factors(z, direct_score, linear, quadratic):
    """Add sparse pairwise costs for complementary outgoing/incoming service."""
    n = z.shape[0]
    mean_direct = float(direct_score.sum()) / max(1, n * (n - 1))
    if mean_direct <= 0.0 or n < 4:
        return

    affinity = direct_score + 0.08
    np.fill_diagonal(affinity, 0.0)
    transition = affinity.copy()
    for _ in range(24):
        transition /= np.maximum(
            transition.sum(axis=1, keepdims=True), 1e-15
        )
        transition /= np.maximum(
            transition.sum(axis=0, keepdims=True), 1e-15
        )

    profiles = np.eye(n) + 2.0 * transition + 1.4 * (transition @ transition)
    depth_scale = max(
        0.75, float(np.log2(max(n, 2)) / np.log2(10.0))
    )
    base_strength = 0.18 * mean_direct * depth_scale
    pair_budget = max(1, n // 4)

    def add_pair(first, second, value):
        if first == second or abs(value) < 1e-14:
            return
        key = (
            (first, second) if first < second else (second, first)
        )
        quadratic[key] = quadratic.get(key, 0.0) + float(value)

    for reverse in (False, True):
        service = profiles.T if reverse else profiles
        traffic = z.T if reverse else z
        totals = traffic.sum(axis=1)
        mean_total = float(totals.mean())
        if mean_total <= 1e-15:
            continue

        for node in range(n):
            total = float(totals[node])
            if total <= 1e-15:
                continue

            neighbors = [a for a in range(n) if a != node]
            m = len(neighbors)
            left, right = np.triu_indices(m, 1)
            weights = traffic[node] / total
            neighbor_service = service[neighbors, :]
            first_service = neighbor_service[left]
            second_service = neighbor_service[right]
            combined = np.maximum(first_service + second_service, 1e-12)

            coverage_cost = -np.log(combined) @ weights
            load_difference = np.abs(
                ((first_service - second_service) / combined) @ weights
            )
            costs = coverage_cost + 0.55 * load_difference

            grand_mean = float(costs.mean())
            spread = float(np.sqrt(np.mean((costs - grand_mean) ** 2)))
            if spread <= 1e-12:
                continue

            dense_cost = np.zeros((m, m), dtype=float)
            dense_cost[left, right] = costs
            dense_cost[right, left] = costs
            row_mean = dense_cost.sum(axis=1) / (m - 1)
            main_effect = (
                (m - 1.0) / (m - 2.0)
            ) * (row_mean - grand_mean)
            residual = (
                costs - grand_mean - main_effect[left] - main_effect[right]
            )

            node_importance = float(np.clip(
                total / mean_total, 0.65, 1.50
            ))
            strength = (
                base_strength * node_importance / max(0.06, spread)
            )

            def service_link(neighbor):
                return (
                    (neighbor, node) if reverse else (node, neighbor)
                )

            for position, neighbor in enumerate(neighbors):
                link = service_link(neighbor)
                linear[link] += strength * float(main_effect[position])

            residual_rms = float(np.sqrt(np.mean(residual * residual)))
            if residual_rms <= 1e-12:
                continue

            ranked = sorted(
                range(len(residual)),
                key=lambda q: (
                    -abs(float(residual[q])),
                    int(left[q]),
                    int(right[q]),
                ),
            )
            for q in ranked[:pair_budget]:
                value = float(residual[q])
                if abs(value) < residual_rms:
                    break
                add_pair(
                    service_link(neighbors[int(left[q])]),
                    service_link(neighbors[int(right[q])]),
                    strength * value,
                )
