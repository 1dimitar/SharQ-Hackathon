"""Surrogate QUBO terms for QOBLIB 08 network design. ShinkaEvolve edits the EVOLVE block."""

import time

# EVOLVE-BLOCK-START
import numpy as np


def surrogate_terms(demand):
    """Build direct-link rewards and sparse, component-supported route terms."""
    traffic = np.asarray(demand, dtype=float)
    n = int(traffic.shape[0])

    traffic = np.maximum(
        np.nan_to_num(traffic, nan=0.0, posinf=0.0, neginf=0.0), 0.0
    ).copy()
    np.fill_diagonal(traffic, 0.0)

    scale = float(np.max(traffic)) if traffic.size else 0.0
    if scale <= 0.0:
        scale = 1.0
    z = traffic / scale

    row_sum = traffic.sum(axis=1)
    col_sum = traffic.sum(axis=0)
    row_share = traffic / np.maximum(row_sum[:, None], 1e-12)
    col_share = traffic / np.maximum(col_sum[None, :], 1e-12)

    incident = np.maximum(row_sum, col_sum)
    max_incident = float(np.max(incident)) if incident.size else 0.0
    if max_incident > 0.0:
        pressure = 0.50 + 0.65 * incident / max_incident
    else:
        pressure = np.ones(n, dtype=float)

    # A concave direct reward recognizes useful demand without letting the
    # largest individual OD entries dominate all other link choices.
    share = np.sqrt(np.maximum(row_share * col_share, 0.0))
    direct_score = (
        0.68 * np.power(z, 0.78)
        + 0.22 * np.sqrt(z)
        + 0.10 * share
    )
    linear = {
        (i, j): -float(direct_score[i, j])
        for i in range(n)
        for j in range(n)
        if i != j
    }

    positive = traffic[traffic > 0.0]
    cutoff = float(np.quantile(positive, 0.70)) if positive.size else 0.0

    quadratic = {}

    def add_pair(a, b, value):
        key = (a, b) if a < b else (b, a)
        quadratic[key] = quadratic.get(key, 0.0) + float(value)

    for i in range(n):
        for k in range(n):
            if i == k or traffic[i, k] <= 0.0:
                continue

            od = float(0.68 * z[i, k] ** 0.74 + 0.32 * np.sqrt(z[i, k]))
            # Strong direct ODs still benefit from short alternate routes,
            # but their indirect reward is mildly tempered.
            od_support = od * (1.0 - 0.25 * np.sqrt(z[i, k]))

            candidates = []
            for j in range(n):
                if j == i or j == k:
                    continue

                zij = float(z[i, j])
                zjk = float(z[j, k])

                # Explicitly require evidence on both legs. The floor retains
                # some value for links needed to connect sparse traffic.
                leg_support = 0.16 + 0.84 * (zij * zjk) ** 0.25

                first_affinity = 0.55 * np.sqrt(row_share[i, j]) + 0.45 * np.sqrt(zij)
                second_affinity = 0.55 * np.sqrt(col_share[j, k]) + 0.45 * np.sqrt(zjk)
                balance = np.sqrt(max(first_affinity * second_affinity, 0.0))

                rank = (
                    leg_support
                    * (0.20 + balance)
                    / (pressure[j] ** 0.75)
                )
                candidates.append((float(rank), j, leg_support, balance))

            candidates.sort(key=lambda item: (-item[0], item[1]))

            # One route for ordinary ODs, two for larger ones. This keeps the
            # coupling graph sparse and bounds construction by 4*n^3 terms.
            keep = 2 if traffic[i, k] >= cutoff else 1
            selected = candidates[:min(keep, len(candidates))]

            for position, (_, j, leg_support, balance) in enumerate(selected):
                first = (i, j)
                second = (j, k)

                # The two-hop reward is directly proportional to demand
                # support on both component links, with a small affinity and
                # rank adjustment.
                rank_factor = 1.0 if position == 0 else 0.78
                route_quality = 0.72 + 0.28 * balance
                route_reward = (
                    -0.31
                    * od_support
                    * leg_support
                    * route_quality
                    * rank_factor
                    / (pressure[j] ** 0.20)
                )
                add_pair(first, second, route_reward)

                # A weak positive coupling makes a direct link and one
                # supported component of its two-hop alternative less
                # attractive together, discouraging redundant routing.
                if z[i, j] >= z[j, k]:
                    component = first
                else:
                    component = second
                redundancy = (
                    0.045
                    * od_support
                    * leg_support
                    * (0.35 + 0.65 * np.sqrt(z[i, k]))
                )
                add_pair((i, k), component, redundancy)

    return linear, quadratic, 1.15


# EVOLVE-BLOCK-END


def run_surrogate(demand):
    """Fixed entry point called by the evaluator, once per demand matrix."""
    demand = np.array(demand, dtype=float)
    start = time.perf_counter()
    linear, quadratic, penalty_weight = surrogate_terms(demand.copy())
    return {
        "n": len(demand),
        "linear": linear,
        "quadratic": quadratic,
        "penalty_weight": penalty_weight,
        "seconds": time.perf_counter() - start,
    }
