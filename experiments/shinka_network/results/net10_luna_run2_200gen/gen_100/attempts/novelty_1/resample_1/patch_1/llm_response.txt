<DESCRIPTION>
This version keeps direct-demand rewards as the main signal and adds pressure-adjusted two-hop rewards. For each positive-demand OD pair, route quality determines how its reward is distributed among candidate intermediates; the distribution is not uniform. The total reward across those routes is capped at 28% of that pair’s direct-link reward, limiting the influence of indirect paths while still favoring less congested relays and links. Degree penalties remain the evaluator’s responsibility.

"""Surrogate QUBO terms for QOBLIB 08 network design. ShinkaEvolve edits the EVOLVE block."""

import time

# EVOLVE-BLOCK-START
import numpy as np


def surrogate_terms(demand):
    """Build linear, quadratic, and degree-penalty terms for the topology QUBO."""
    t = np.asarray(demand, dtype=float)
    if t.ndim != 2 or t.shape[0] != t.shape[1]:
        raise ValueError("demand must be a square matrix")

    t = np.nan_to_num(t, nan=0.0, posinf=0.0, neginf=0.0)
    t = np.maximum(t, 0.0).copy()
    np.fill_diagonal(t, 0.0)
    n = t.shape[0]

    if n == 0:
        return {}, {}, 1.05

    scale = float(np.max(t))
    if scale <= 0.0:
        scale = 1.0
    d = t / scale

    # Direct traffic remains the dominant and reliable link-selection signal.
    # A small concave component prevents moderate demands from being ignored.
    direct_reward = 0.90 * d + 0.10 * np.sqrt(d)
    linear = {
        (i, j): -float(direct_reward[i, j])
        for i in range(n)
        for j in range(n)
        if i != j
    }

    # Approximate relay congestion from total incident traffic. Normalize by
    # the mean so pressure remains meaningful across differently scaled inputs.
    incident = np.sum(d, axis=0) + np.sum(d, axis=1)
    mean_incident = float(np.mean(incident))
    if mean_incident > 0.0:
        pressure = np.maximum(0.65, incident / mean_incident)
    else:
        pressure = np.ones(n, dtype=float)

    quadratic = {}
    route_cap = 0.28

    # For each OD pair, share a bounded reward among its two-hop routes.
    # Route qualities differ by relay pressure and estimated direct load on
    # the two constituent links; squaring quality sharpens this preference.
    for i in range(n):
        for k in range(n):
            if i == k or d[i, k] <= 0.0:
                continue

            intermediates = []
            qualities = []
            for j in range(n):
                if j == i or j == k:
                    continue

                edge_load = float(d[i, j] + d[j, k])
                quality = 1.0 / (
                    np.sqrt(float(pressure[j])) * (1.0 + 0.75 * edge_load)
                )
                intermediates.append(j)
                qualities.append(quality * quality)

            total_quality = float(sum(qualities))
            if total_quality <= 0.0:
                continue

            # The aggregate indirect reward for this OD pair is capped relative
            # to its direct-link reward, rather than growing with route count.
            pair_budget = route_cap * float(direct_reward[i, k])
            for j, quality in zip(intermediates, qualities):
                share = quality / total_quality
                weight = -pair_budget * share
                quadratic[((i, j), (j, k))] = float(weight)

    return linear, quadratic, 1.05


# EVOLVE-BLOCK-END


def run_surrogate(demand):
    """Fixed entry point called by the evaluator, once per demand matrix."""
    demand = np.array(demand, dtype=float)
    start = time.perf_counter()
    linear, quadratic, penalty_weight = surrogate_terms(demand.copy())
    return {
        "n": len(demand),
        "linear": linear,
        "quadratic": quadratic,
        "penalty_weight": penalty_weight,
        "seconds": time.perf_counter() - start,
    }
