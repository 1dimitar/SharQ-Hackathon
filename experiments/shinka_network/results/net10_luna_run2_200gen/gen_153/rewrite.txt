"""Surrogate QUBO terms for QOBLIB 08 network design."""

import time

# EVOLVE-BLOCK-START
import numpy as np


def surrogate_terms(demand):
    """Build direct-link, route, and diversity costs from a demand matrix."""
    d = np.asarray(demand, dtype=float)
    d = np.nan_to_num(d, nan=0.0, posinf=0.0, neginf=0.0)
    d = np.maximum(d, 0.0).copy()
    n = int(d.shape[0])
    np.fill_diagonal(d, 0.0)

    scale = float(np.max(d)) if d.size else 0.0
    if scale <= 0.0:
        scale = 1.0
    z = d / scale

    rows = np.sum(z, axis=1)
    cols = np.sum(z, axis=0)
    incident = rows + cols
    max_incident = max(float(np.max(incident)) if n else 0.0, 1.0)
    mean_incident = max(float(np.mean(incident)) if n else 0.0, 1e-12)
    pressure = np.maximum(0.75, incident / mean_incident)

    # Favor direct service, with a small endpoint-capacity signal.
    linear = {}
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            direct = float(z[i, j])
            endpoint = (float(rows[i]) + float(cols[j])) / (2.0 * max_incident)
            value = 0.76 * direct + 0.16 * np.sqrt(direct) + 0.08 * endpoint
            linear[(i, j)] = -float(value)

    # Redundant outgoing/incoming choices can concentrate traffic. Similarity
    # is measured on normalized destination and origin profiles respectively.
    row_norm = np.sqrt(np.sum(z * z, axis=1))
    col_norm = np.sqrt(np.sum(z * z, axis=0))
    row_profile = np.divide(
        z, row_norm[:, None], out=np.zeros_like(z),
        where=row_norm[:, None] > 0.0,
    )
    col_profile = np.divide(
        z, col_norm[None, :], out=np.zeros_like(z),
        where=col_norm[None, :] > 0.0,
    )

    quadratic = {}
    diversity_strength = 0.055
    for node in range(n):
        neighbors = [v for v in range(n) if v != node]
        for a, j in enumerate(neighbors):
            for k in neighbors[a + 1:]:
                sim_out = float(np.dot(row_profile[j], row_profile[k]))
                if sim_out > 1e-10:
                    quadratic[((node, j), (node, k))] = diversity_strength * sim_out
                sim_in = float(np.dot(col_profile[j], col_profile[k]))
                if sim_in > 1e-10:
                    quadratic[((j, node), (k, node))] = diversity_strength * sim_in

    # Allocate a bounded indirect-service reward among plausible two-hop paths.
    # Restrict each OD pair to its best-supported intermediates to keep the
    # interaction graph sparse and avoid rewarding arbitrary paths equally.
    route_limit = min(max(n // 2, 2), 8)
    for i in range(n):
        for k in range(n):
            traffic = float(z[i, k])
            if i == k or traffic <= 0.0:
                continue

            candidates = []
            for j in range(n):
                if j == i or j == k:
                    continue
                first = float(z[i, j])
                second = float(z[j, k])
                support = np.sqrt(first * second)
                support_factor = 0.12 + 0.88 * support / (1.0 + support)
                transit_factor = 1.0 / (0.65 + float(pressure[j]))
                congestion_factor = 1.0 / np.sqrt(
                    (1.0 + 0.6 * first) * (1.0 + 0.6 * second)
                )
                candidates.append((
                    support_factor * transit_factor * congestion_factor,
                    j,
                ))

            candidates.sort(reverse=True)
            chosen = candidates[:route_limit]
            denominator = sum(score for score, _ in chosen)
            if denominator <= 0.0:
                continue

            # Blend the current program's concave demand signal with a modest
            # total route budget, avoiding excessive preference for detours.
            route_signal = 0.8 * traffic + 0.2 * np.sqrt(traffic)
            budget = 0.48 * route_signal
            for score, j in chosen:
                key = ((i, j), (j, k))
                quadratic[key] = quadratic.get(key, 0.0) - (
                    budget * score / denominator
                )

    # Keep the pair count below the prescribed O(n^3) interaction budget.
    return linear, quadratic, 1.0


# EVOLVE-BLOCK-END


def run_surrogate(demand):
    """Fixed entry point called by the evaluator, once per demand matrix."""
    demand = np.array(demand, dtype=float)
    start = time.perf_counter()
    linear, quadratic, penalty_weight = surrogate_terms(demand.copy())
    return {
        "n": len(demand),
        "linear": linear,
        "quadratic": quadratic,
        "penalty_weight": penalty_weight,
        "seconds": time.perf_counter() - start,
    }
