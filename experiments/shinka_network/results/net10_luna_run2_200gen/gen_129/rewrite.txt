"""Surrogate QUBO terms for QOBLIB 08 network design."""

import time

# EVOLVE-BLOCK-START
import numpy as np


def surrogate_terms(demand):
    """Build direct-link and congestion-discounted two-hop cost terms."""
    matrix = np.asarray(demand, dtype=float)
    n = len(matrix)

    traffic = np.maximum(matrix, 0.0).copy()
    np.fill_diagonal(traffic, 0.0)
    scale = float(np.max(traffic)) if traffic.size else 0.0
    if scale <= 0.0:
        scale = 1.0

    # Preserve a strong preference for direct service while giving moderate
    # OD demands some influence through a concave component.
    linear = {}
    for i in range(n):
        for j in range(n):
            if i != j:
                value = float(traffic[i, j]) / scale
                linear[(i, j)] = -(0.85 * value + 0.15 * np.sqrt(value))

    # High incident demand makes a node less attractive as a transit point.
    incident = np.sum(traffic, axis=0) + np.sum(traffic, axis=1)
    mean_incident = float(np.mean(incident)) if incident.size else 0.0
    if mean_incident <= 0.0:
        mean_incident = 1.0
    node_pressure = np.maximum(0.75, incident / mean_incident)

    # Reward useful two-hop paths, but sharply discount paths whose component
    # edges already have high direct demand.
    quadratic = {}
    path_strength = 0.40
    for i in range(n):
        for k in range(n):
            if i == k:
                continue

            normalized_demand = float(traffic[i, k]) / scale
            if normalized_demand <= 0.0:
                continue
            path_support = (
                0.70 * normalized_demand
                + 0.30 * np.sqrt(normalized_demand)
            )

            for j in range(n):
                if j == i or j == k:
                    continue

                edge_pressure = (
                    float(traffic[i, j]) + float(traffic[j, k])
                ) / scale
                congestion_discount = (1.0 + edge_pressure) ** 1.5
                weight = (
                    -path_strength * path_support
                    / (float(node_pressure[j]) * congestion_discount)
                )

                first, second = (i, j), (j, k)
                key = (first, second) if first < second else (second, first)
                quadratic[key] = weight

    # The evaluator supplies the degree penalty; use a modestly stronger
    # setting to favor feasible samples before its repair step.
    penalty_weight = 1.25
    return linear, quadratic, penalty_weight


# EVOLVE-BLOCK-END


def run_surrogate(demand):
    """Fixed entry point called by the evaluator, once per demand matrix."""
    demand = np.array(demand, dtype=float)
    start = time.perf_counter()
    linear, quadratic, penalty_weight = surrogate_terms(demand.copy())
    return {
        "n": len(demand),
        "linear": linear,
        "quadratic": quadratic,
        "penalty_weight": penalty_weight,
        "seconds": time.perf_counter() - start,
    }
