"""Surrogate QUBO terms for QOBLIB 08 network design."""

import time

# EVOLVE-BLOCK-START
import numpy as np


def _prepare_demand(demand):
    """Return nonnegative, off-diagonal demand normalized to [0, 1]."""
    d = np.asarray(demand, dtype=float).copy()
    if d.ndim != 2 or d.shape[0] != d.shape[1]:
        raise ValueError("demand must be a square matrix")
    d = np.maximum(d, 0.0)
    np.fill_diagonal(d, 0.0)

    scale = float(np.max(d)) if d.size else 0.0
    if scale > 0.0:
        d /= scale
    return d


def _direct_terms(d):
    """Score direct links, retaining a modest signal for busy endpoints."""
    n = d.shape[0]
    row = d.sum(axis=1)
    col = d.sum(axis=0)
    node_scale = max(float(np.max(row)), float(np.max(col)), 1.0)

    linear = {}
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            endpoint_pressure = (float(row[i]) + float(col[j])) / (
                2.0 * node_scale
            )
            value = float(d[i, j])
            linear[(i, j)] = -(0.90 * value + 0.10 * endpoint_pressure)
    return linear


def _path_terms(d):
    """Build sparse, OD-capped rewards for selected two-hop paths."""
    n = d.shape[0]
    if n < 3:
        return {}

    # Nodes with larger aggregate traffic are less desirable as transit nodes.
    node_load = d.sum(axis=0) + d.sum(axis=1)
    mean_load = float(np.mean(node_load))
    if mean_load > 0.0:
        transit = 2.0 * mean_load / (mean_load + node_load)
    else:
        transit = np.ones(n, dtype=float)

    # Limiting candidate intermediates reduces diffuse rewards and term count.
    # At n=24 this emits at most 4*n*(n-2) terms, below the required bound.
    candidate_count = min(4, n - 2)
    quadratic = {}

    for i in range(n):
        for k in range(n):
            traffic = float(d[i, k])
            if i == k or traffic <= 0.0:
                continue

            candidates = [j for j in range(n) if j != i and j != k]

            # Prefer transit nodes with spare aggregate capacity. A mild
            # compatibility factor favors paths whose constituent links have
            # some direct-demand support, without making that support dominant.
            def path_score(j):
                support = float(d[i, j]) + float(d[j, k])
                return float(transit[j]) * (0.80 + 0.20 * support)

            candidates.sort(key=path_score, reverse=True)
            chosen = candidates[:candidate_count]
            factors = [path_score(j) for j in chosen]
            factor_sum = sum(factors)
            if factor_sum <= 0.0:
                continue

            # Keep the total indirect reward for this OD pair bounded relative
            # to its direct reward; distribute only among the selected paths.
            total_reward = min(0.30, 0.75 * traffic)
            for j, factor in zip(chosen, factors):
                weight = -total_reward * factor / factor_sum
                quadratic[((i, j), (j, k))] = float(weight)

    return quadratic


def surrogate_terms(demand):
    """Return linear terms, quadratic terms, and evaluator penalty weight."""
    d = _prepare_demand(demand)
    linear = _direct_terms(d)
    quadratic = _path_terms(d)
    return linear, quadratic, 1.0


# EVOLVE-BLOCK-END


def run_surrogate(demand):
    """Fixed entry point called by the evaluator, once per demand matrix."""
    demand = np.array(demand, dtype=float)
    start = time.perf_counter()
    linear, quadratic, penalty_weight = surrogate_terms(demand.copy())
    return {
        "n": len(demand),
        "linear": linear,
        "quadratic": quadratic,
        "penalty_weight": penalty_weight,
        "seconds": time.perf_counter() - start,
    }
