"""Surrogate QUBO terms for QOBLIB 08 network design. ShinkaEvolve edits the EVOLVE block."""

import time

# EVOLVE-BLOCK-START
import numpy as np


def surrogate_terms(demand):
    """Build direct-link and pressure-discounted two-hop cost terms."""
    matrix = np.asarray(demand, dtype=float)
    n = len(matrix)

    # Ignore negative demand and self-traffic; keep the zero-demand case safe.
    traffic = np.maximum(matrix, 0.0).copy()
    np.fill_diagonal(traffic, 0.0)
    scale = float(np.max(traffic)) if traffic.size else 0.0
    if scale <= 0.0:
        scale = 1.0
    normalized = traffic / scale

    # Mostly reward direct demand, with a small concave component to preserve
    # influence for moderate-demand links.
    linear = {}
    for i in range(n):
        for j in range(n):
            if i != j:
                value = float(normalized[i, j])
                linear[(i, j)] = -(0.94 * value + 0.06 * np.sqrt(value))

    # Incident demand estimates how costly it may be to use a node as transit.
    incident = np.sum(traffic, axis=0) + np.sum(traffic, axis=1)
    mean_incident = float(np.mean(incident)) if incident.size else 0.0
    if mean_incident <= 0.0:
        mean_incident = 1.0
    node_pressure = np.maximum(0.75, incident / mean_incident)

    # Reward useful two-hop paths. Use primarily linear OD demand, retain a
    # modest concave contribution, and discount busy transit nodes and
    # component links with substantial direct demand.
    quadratic = {}
    path_strength = 0.52
    for i in range(n):
        for k in range(n):
            if i == k:
                continue
            flow = float(normalized[i, k])
            if flow <= 0.0:
                continue

            path_support = 0.90 * flow + 0.10 * np.sqrt(flow)
            for j in range(n):
                if j == i or j == k:
                    continue

                edge_pressure = float(normalized[i, j] + normalized[j, k])
                weight = (
                    -path_strength * path_support
                    / (float(node_pressure[j]) * (1.0 + 0.65 * edge_pressure))
                )

                first, second = (i, j), (j, k)
                key = (first, second) if first < second else (second, first)
                quadratic[key] = weight

    # The evaluator supplies the degree penalties; this modest increase helps
    # counter the many attractive path terms without overwhelming cost signals.
    penalty_weight = 1.25
    return linear, quadratic, penalty_weight


# EVOLVE-BLOCK-END


def run_surrogate(demand):
    """Fixed entry point called by the evaluator, once per demand matrix."""
    demand = np.array(demand, dtype=float)
    start = time.perf_counter()
    linear, quadratic, penalty_weight = surrogate_terms(demand.copy())
    return {
        "n": len(demand),
        "linear": linear,
        "quadratic": quadratic,
        "penalty_weight": penalty_weight,
        "seconds": time.perf_counter() - start,
    }
