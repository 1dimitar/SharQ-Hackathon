"""Surrogate QUBO terms for QOBLIB 08 network design."""

import time

# EVOLVE-BLOCK-START
import numpy as np


def surrogate_terms(demand):
    """Build direct-link and pressure-discounted two-hop cost terms."""
    matrix = np.asarray(demand, dtype=float)
    n = len(matrix)

    # Use nonnegative off-diagonal traffic and handle zero demand robustly.
    traffic = np.maximum(matrix, 0.0).copy()
    np.fill_diagonal(traffic, 0.0)
    scale = float(np.max(traffic)) if traffic.size else 0.0
    if scale <= 0.0:
        scale = 1.0

    normalized = traffic / scale

    # Concave support retains a direct-link preference while giving
    # moderate-demand links some influence.
    linear = {}
    for i in range(n):
        for j in range(n):
            if i != j:
                value = float(normalized[i, j])
                linear[(i, j)] = -(0.85 * value + 0.15 * np.sqrt(value))

    # Approximate transit burden using total incident demand.
    incident = np.sum(traffic, axis=0) + np.sum(traffic, axis=1)
    mean_incident = float(np.mean(incident)) if incident.size else 0.0
    if mean_incident <= 0.0:
        mean_incident = 1.0
    node_pressure = np.maximum(0.75, incident / mean_incident)

    # Precompute normalized edge pressure for efficient path construction.
    edge_pressure = normalized
    quadratic = {}
    path_strength = 0.56

    for i in range(n):
        for k in range(n):
            if i == k:
                continue

            through_demand = float(normalized[i, k])
            if through_demand <= 0.0:
                continue

            # Blend linear and concave support, with a bounded boost for
            # smaller OD demands.
            raw_support = (
                0.70 * through_demand
                + 0.30 * np.sqrt(through_demand)
            )
            relative_quality = raw_support / through_demand
            relative_quality = min(1.25, max(0.9, relative_quality))
            path_support = through_demand * relative_quality

            for j in range(n):
                if j == i or j == k:
                    continue

                pressure = float(edge_pressure[i, j] + edge_pressure[j, k])
                weight = (
                    -path_strength * path_support
                    / (float(node_pressure[j]) * (1.0 + 0.5 * pressure))
                )

                first, second = (i, j), (j, k)
                key = (first, second) if first < second else (second, first)
                quadratic[key] = weight

    # The evaluator adds the degree constraints.
    penalty_weight = 1.0
    return linear, quadratic, penalty_weight


# EVOLVE-BLOCK-END


def run_surrogate(demand):
    """Fixed entry point called by the evaluator, once per demand matrix."""
    demand = np.array(demand, dtype=float)
    start = time.perf_counter()
    linear, quadratic, penalty_weight = surrogate_terms(demand.copy())
    return {
        "n": len(demand),
        "linear": linear,
        "quadratic": quadratic,
        "penalty_weight": penalty_weight,
        "seconds": time.perf_counter() - start,
    }
