"""Surrogate QUBO terms for QOBLIB 08 network design."""

import time

# EVOLVE-BLOCK-START
import numpy as np


def _demand_statistics(demand):
    """Return a safe demand scale and per-node incident traffic."""
    matrix = np.asarray(demand, dtype=float)
    n = matrix.shape[0]
    scale = float(np.max(np.abs(matrix))) if matrix.size else 0.0
    if scale <= 0.0:
        scale = 1.0

    outgoing = np.sum(matrix, axis=1)
    incoming = np.sum(matrix, axis=0)
    incident = outgoing + incoming
    max_incident = float(np.max(incident)) if incident.size else 0.0
    return matrix, n, scale, outgoing, incoming, incident, max_incident


def _build_linear(matrix, n, scale, outgoing, incoming):
    """Favor direct traffic while retaining a small endpoint-capacity signal."""
    row_scale = float(np.max(outgoing)) if outgoing.size else 0.0
    col_scale = float(np.max(incoming)) if incoming.size else 0.0
    endpoint_scale = max(row_scale, col_scale, 1.0)

    linear = {}
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            endpoint_pressure = (float(outgoing[i]) + float(incoming[j])) / (
                2.0 * endpoint_scale
            )
            linear[(i, j)] = -(
                0.90 * float(matrix[i, j]) / scale
                + 0.10 * endpoint_pressure
            )
    return linear


def _build_two_hop_terms(matrix, n, scale, incident, max_incident):
    """Reward useful two-hop routes, with bounded pressure on transit nodes."""
    quadratic = {}
    if max_incident <= 0.0:
        return quadratic

    for i in range(n):
        for j in range(n):
            if i == j:
                continue

            # Keep the discount bounded: even the busiest node can still be
            # selected as a transit point when the route is otherwise useful.
            load_ratio = float(incident[j]) / max_incident
            transit_factor = max(0.75, 1.0 - 0.25 * load_ratio)

            for k in range(n):
                if k == i or k == j:
                    continue
                traffic = float(matrix[i, k])
                if traffic <= 0.0:
                    continue

                # The ordered links form i -> j -> k and jointly serve i -> k.
                quadratic[((i, j), (j, k))] = (
                    -0.12 * (traffic / scale) * transit_factor
                )
    return quadratic


def surrogate_terms(demand):
    """Build linear and quadratic surrogate costs from a demand matrix."""
    stats = _demand_statistics(demand)
    matrix, n, scale, outgoing, incoming, incident, max_incident = stats

    linear = _build_linear(matrix, n, scale, outgoing, incoming)
    quadratic = _build_two_hop_terms(
        matrix, n, scale, incident, max_incident
    )

    # Degree constraints are added by the evaluator, not included here.
    penalty_weight = 1.0
    return linear, quadratic, penalty_weight


# EVOLVE-BLOCK-END


def run_surrogate(demand):
    """Fixed entry point called by the evaluator, once per demand matrix."""
    demand = np.array(demand, dtype=float)
    start = time.perf_counter()
    linear, quadratic, penalty_weight = surrogate_terms(demand.copy())
    return {
        "n": len(demand),
        "linear": linear,
        "quadratic": quadratic,
        "penalty_weight": penalty_weight,
        "seconds": time.perf_counter() - start,
    }
