"""Surrogate QUBO terms for QOBLIB 08 network design. ShinkaEvolve edits the EVOLVE block."""

import time

# EVOLVE-BLOCK-START
import numpy as np


def surrogate_terms(demand):
    """Build direct-link and pressure-discounted two-hop cost terms."""
    traffic = np.asarray(demand, dtype=float)
    if traffic.ndim != 2 or traffic.shape[0] != traffic.shape[1]:
        raise ValueError("demand must be a square matrix")

    n = traffic.shape[0]
    if n == 0:
        return {}, {}, 1.0

    # Work with finite, nonnegative off-diagonal traffic only.
    traffic = np.nan_to_num(
        traffic, nan=0.0, posinf=0.0, neginf=0.0
    ).copy()
    np.maximum(traffic, 0.0, out=traffic)
    np.fill_diagonal(traffic, 0.0)

    scale = float(np.max(traffic))
    if scale <= 0.0:
        scale = 1.0
    normalized = traffic / scale

    # Keep a direct-demand signal, while allowing moderate demands to matter.
    linear = {}
    for i in range(n):
        for j in range(n):
            if i == j:
                continue
            value = float(normalized[i, j])
            linear[(i, j)] = -(0.85 * value + 0.15 * float(np.sqrt(value)))

    # Mean-normalized incident traffic estimates transit pressure.
    incident = np.sum(traffic, axis=0) + np.sum(traffic, axis=1)
    mean_incident = float(np.mean(incident))
    if mean_incident <= 0.0:
        node_pressure = np.ones(n, dtype=float)
    else:
        node_pressure = np.maximum(0.75, incident / mean_incident)

    # Reward useful two-hop paths, tempered by relay pressure and existing
    # direct load on the two links in the path.
    quadratic = {}
    path_strength = 0.46

    for i in range(n):
        for k in range(n):
            if i == k:
                continue

            through = float(normalized[i, k])
            if through <= 0.0:
                continue

            path_support = 0.70 * through + 0.30 * float(np.sqrt(through))

            for j in range(n):
                if j == i or j == k:
                    continue

                edge_pressure = float(normalized[i, j] + normalized[j, k])
                weight = (
                    -path_strength
                    * path_support
                    / (
                        float(node_pressure[j])
                        * (1.0 + 0.5 * edge_pressure)
                    )
                )

                first, second = (i, j), (j, k)
                key = (first, second) if first < second else (second, first)
                quadratic[key] = float(weight)

    # Degree penalties are added by the evaluator.
    return linear, quadratic, 1.0


# EVOLVE-BLOCK-END


def run_surrogate(demand):
    """Fixed entry point called by the evaluator, once per demand matrix."""
    demand = np.array(demand, dtype=float)
    start = time.perf_counter()
    linear, quadratic, penalty_weight = surrogate_terms(demand.copy())
    return {
        "n": len(demand),
        "linear": linear,
        "quadratic": quadratic,
        "penalty_weight": penalty_weight,
        "seconds": time.perf_counter() - start,
    }
