<DESCRIPTION>
This crossover keeps the direct-link rewards, relay budget, demand support, and 0.80 high-demand cutoff from the stronger-performing implementation. It blends that implementation’s geometric-mean relay ranking with the inspiration program’s preference for relays with two useful legs, and adds a mild edge-pressure adjustment when ranking and weighting paths. The result remains sparse, deterministic, and within the quadratic-term limit.

"""Surrogate QUBO terms for QOBLIB 08 network design. ShinkaEvolve edits the EVOLVE block."""

import time

# EVOLVE-BLOCK-START
import math
import numpy as np


def surrogate_terms(demand):
    """Build direct-link rewards and sparse, pressure-aware two-hop rewards."""
    traffic = np.maximum(np.asarray(demand, dtype=float), 0.0).copy()
    n = len(traffic)
    np.fill_diagonal(traffic, 0.0)

    scale = float(np.max(traffic)) if traffic.size else 0.0
    if scale <= 0.0:
        scale = 1.0
    traffic /= scale

    # Direct-link rewards retain a concave component for moderate demands.
    linear = {}
    for i in range(n):
        for j in range(n):
            if i != j:
                value = float(traffic[i, j])
                linear[(i, j)] = -(0.75 * value + 0.25 * math.sqrt(value))

    # A node with heavy incoming or outgoing demand is a less attractive relay.
    incoming = np.sum(traffic, axis=0)
    outgoing = np.sum(traffic, axis=1)
    incident = np.maximum(incoming, outgoing)
    max_incident = float(np.max(incident)) if incident.size else 0.0
    if max_incident <= 0.0:
        max_incident = 1.0
    node_pressure = np.maximum(0.65, incident / max_incident)

    # Keep the proven sparse relay budget, with a few additional alternatives
    # for the highest-demand OD pairs.
    relay_count = min(n - 2, max(2, n // 4))
    extra_relays = min(2, max(0, n - 2 - relay_count))
    positive_od = traffic[traffic > 0.0]
    high_cutoff = (
        float(np.quantile(positive_od, 0.80))
        if positive_od.size else float("inf")
    )

    quadratic = {}
    path_strength = 0.28

    for i in range(n):
        for k in range(n):
            if i == k:
                continue

            od = float(traffic[i, k])
            if od <= 0.0:
                continue

            support = 0.75 * od + 0.25 * math.sqrt(od)
            pair_relay_count = relay_count
            if od >= high_cutoff:
                pair_relay_count = min(n - 2, relay_count + extra_relays)

            candidates = []
            for j in range(n):
                if j == i or j == k:
                    continue

                a = float(traffic[i, j])
                b = float(traffic[j, k])

                # Blend geometric leg affinity with a bottleneck-aware score:
                # the latter favors relays for which both legs are useful.
                geometric = math.sqrt((0.1 + a) * (0.1 + b))
                bottleneck = min(a, b) + 0.25 * max(a, b) + 0.01
                edge_pressure = 1.0 + 0.25 * (a + b)
                pressure = float(node_pressure[j]) * edge_pressure
                affinity = 0.8 * geometric + 0.2 * bottleneck
                rank = affinity / (0.75 + pressure)
                candidates.append((rank, j, pressure))

            candidates.sort(key=lambda item: (-item[0], item[1]))

            for rank, j, pressure in candidates[:pair_relay_count]:
                rank_factor = 0.8 + 0.2 * rank / (1.0 + rank)
                weight = (
                    -path_strength * support * rank_factor
                    / (0.75 + pressure)
                )

                first = (i, j)
                second = (j, k)
                key = (first, second) if first < second else (second, first)
                quadratic[key] = quadratic.get(key, 0.0) + weight

    # Degree penalties are supplied by the evaluator.
    penalty_weight = 1.5
    return linear, quadratic, penalty_weight


# EVOLVE-BLOCK-END


def run_surrogate(demand):
    """Fixed entry point called by the evaluator, once per demand matrix."""
    demand = np.array(demand, dtype=float)
    start = time.perf_counter()
    linear, quadratic, penalty_weight = surrogate_terms(demand.copy())
    return {
        "n": len(demand),
        "linear": linear,
        "quadratic": quadratic,
        "penalty_weight": penalty_weight,
        "seconds": time.perf_counter() - start,
    }
