<DESCRIPTION>
Reorganizes term construction into separate traffic normalization, direct-link scoring, and sparse relay-selection stages. Relay candidates are ranked by their weakest leg and adjusted for node pressure, so the limited interaction budget favors plausible paths without creating a dense QUBO. The returned terms remain deterministic and within the quadratic-term limit.
</DESCRIPTION>

"""Surrogate QUBO terms for directed network design."""

import time

# EVOLVE-BLOCK-START
import numpy as np


def _normalized_traffic(demand):
    """Return nonnegative, zero-diagonal traffic scaled to [0, 1]."""
    traffic = np.asarray(demand, dtype=float).copy()
    traffic = np.maximum(traffic, 0.0)
    np.fill_diagonal(traffic, 0.0)
    scale = float(np.max(traffic)) if traffic.size else 0.0
    if scale > 0.0:
        traffic /= scale
    return traffic


def _direct_link_terms(traffic):
    """Reward direct service, preserving signal for moderate demands."""
    n = traffic.shape[0]
    linear = {}
    for i in range(n):
        for j in range(n):
            if i != j:
                value = float(traffic[i, j])
                linear[(i, j)] = -(0.75 * value + 0.25 * np.sqrt(value))
    return linear


def _relay_pressure(traffic):
    """Estimate the relative burden of using each node as a relay."""
    incoming = np.sum(traffic, axis=0)
    outgoing = np.sum(traffic, axis=1)
    incident = np.maximum(incoming, outgoing)
    largest = float(np.max(incident)) if incident.size else 0.0
    if largest <= 0.0:
        return np.ones(traffic.shape[0], dtype=float)
    return np.maximum(0.65, incident / largest)


def _relay_terms(traffic, pressure):
    """Add sparse rewards for selected two-hop routes."""
    n = traffic.shape[0]
    quadratic = {}
    relay_budget = min(n - 2, max(1, n // 5))
    path_strength = 0.46

    for source in range(n):
        for target in range(n):
            if source == target:
                continue

            od = float(traffic[source, target])
            if od <= 0.0:
                continue
            support = 0.70 * od + 0.30 * np.sqrt(od)

            # Rank candidates before computing coefficients. This keeps the
            # interaction graph sparse and gives the budget to useful relays.
            candidates = []
            for relay in range(n):
                if relay == source or relay == target:
                    continue
                first = float(traffic[source, relay])
                second = float(traffic[relay, target])
                bottleneck_affinity = min(first, second) + 0.25 * max(first, second)
                rank = bottleneck_affinity / (0.75 + float(pressure[relay]))
                candidates.append((rank, relay, first + second))

            candidates.sort(key=lambda item: (-item[0], item[1]))
            for rank, relay, leg_load in candidates[:relay_budget]:
                rank_factor = 0.8 + 0.2 * rank / (1.0 + rank)
                weight = (
                    -path_strength
                    * support
                    * rank_factor
                    / (float(pressure[relay]) * (1.0 + 0.5 * leg_load))
                )
                first_link = (source, relay)
                second_link = (relay, target)
                key = (
                    (first_link, second_link)
                    if first_link < second_link
                    else (second_link, first_link)
                )
                quadratic[key] = quadratic.get(key, 0.0) + weight

    return quadratic


def surrogate_terms(demand):
    """Build direct-link and sparse relay-support cost terms."""
    traffic = _normalized_traffic(demand)
    linear = _direct_link_terms(traffic)
    pressure = _relay_pressure(traffic)
    quadratic = _relay_terms(traffic, pressure)

    # Degree penalties are added by the evaluator.
    return linear, quadratic, 5.0


# EVOLVE-BLOCK-END


def run_surrogate(demand):
    """Fixed entry point called by the evaluator, once per demand matrix."""
    demand = np.array(demand, dtype=float)
    start = time.perf_counter()
    linear, quadratic, penalty_weight = surrogate_terms(demand.copy())
    return {
        "n": len(demand),
        "linear": linear,
        "quadratic": quadratic,
        "penalty_weight": penalty_weight,
        "seconds": time.perf_counter() - start,
    }
