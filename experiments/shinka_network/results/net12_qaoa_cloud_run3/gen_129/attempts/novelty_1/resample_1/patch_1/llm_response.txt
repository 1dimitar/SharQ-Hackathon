<DESCRIPTION>
The earlier programs reward direct demand and selected two-hop paths. This design keeps those signals but treats relay choice as a small, normalized allocation: each OD pair distributes its route reward across only its strongest relays, with pressure-adjusted path plausibility. It also aggregates those allocations into a modest link-level exposure signal, so links useful to many plausible routes receive some credit without letting transit centrality overwhelm direct demand. A stronger degree penalty is intended to reduce invalid samples before repair.
</DESCRIPTION>

"""Surrogate QUBO terms for QOBLIB 08 network design."""

import time

# EVOLVE-BLOCK-START
import math
import numpy as np


def surrogate_terms(demand):
    """Return direct-link rewards and sparse, exposure-weighted route terms."""
    traffic = np.asarray(demand, dtype=float)
    n = len(traffic)

    traffic = np.nan_to_num(
        traffic, nan=0.0, posinf=0.0, neginf=0.0
    )
    traffic = np.maximum(traffic, 0.0).copy()
    np.fill_diagonal(traffic, 0.0)

    maximum = float(np.max(traffic)) if traffic.size else 0.0
    if maximum <= 0.0:
        maximum = 1.0
    demand_norm = traffic / maximum

    # Pressure measures how much traffic is incident to each possible relay.
    row_load = np.sum(demand_norm, axis=1)
    col_load = np.sum(demand_norm, axis=0)
    node_load = np.maximum(row_load, col_load)
    load_scale = float(np.max(node_load)) if node_load.size else 0.0
    if load_scale > 0.0:
        pressure = np.maximum(0.55, node_load / load_scale)
    else:
        pressure = np.ones(n, dtype=float)

    # Direct links remain valuable, with a concave component retaining
    # influence for moderate demands.
    linear = {}
    for i in range(n):
        for j in range(n):
            if i != j:
                d = float(demand_norm[i, j])
                linear[(i, j)] = -(0.72 * d + 0.28 * math.sqrt(d))

    if n < 3 or not np.any(demand_norm):
        return linear, {}, 3.0

    positive = demand_norm[demand_norm > 0.0]
    high_cutoff = (
        float(np.quantile(positive, 0.88))
        if positive.size else float("inf")
    )

    # For each OD pair, choose a few plausible relays and split that OD's
    # route credit among them. This prevents many weak alternatives from
    # accumulating an excessive reward.
    quadratic = {}
    exposure = np.zeros((n, n), dtype=float)

    for i in range(n):
        for k in range(n):
            if i == k:
                continue

            d = float(demand_norm[i, k])
            if d <= 0.0:
                continue

            support = 0.68 * d + 0.32 * math.sqrt(d)
            keep = 3 if d >= high_cutoff else 2
            candidates = []

            for j in range(n):
                if j == i or j == k:
                    continue

                a = float(demand_norm[i, j])
                b = float(demand_norm[j, k])

                # Harmonic leg affinity emphasizes paths whose weaker leg is
                # also useful; the small floor allows unseen legs to compete.
                leg_a = 0.035 + math.sqrt(a)
                leg_b = 0.035 + math.sqrt(b)
                affinity = 2.0 * leg_a * leg_b / (leg_a + leg_b)

                # Discount relays with substantial incident traffic.
                rank = affinity / (float(pressure[j]) ** 1.0)
                candidates.append((rank, j))

            candidates.sort(key=lambda item: (-item[0], item[1]))
            chosen = candidates[:keep]
            if not chosen:
                continue

            # Squared normalized scores make the best relay influential while
            # preserving alternatives for route splitting.
            denom = sum(rank * rank for rank, _ in chosen)
            if denom <= 0.0:
                probabilities = [1.0 / len(chosen)] * len(chosen)
            else:
                probabilities = [(rank * rank) / denom for rank, _ in chosen]

            for (rank, j), probability in zip(chosen, probabilities):
                first = (i, j)
                second = (j, k)
                key = (first, second) if first < second else (second, first)

                # Each OD contributes a bounded amount of pairwise route
                # reward, rather than the same full reward to every relay.
                weight = -0.40 * support * probability
                quadratic[key] = quadratic.get(key, 0.0) + float(weight)

                route_mass = support * probability
                exposure[i, j] += route_mass
                exposure[j, k] += route_mass

    # Small marginal credit for links appearing in many promising routes.
    # Keeping this weaker than direct demand avoids turning generic hubs into
    # the dominant signal.
    max_exposure = float(np.max(exposure)) if exposure.size else 0.0
    if max_exposure > 0.0:
        for i in range(n):
            for j in range(n):
                if i != j and exposure[i, j] > 0.0:
                    marginal = math.sqrt(float(exposure[i, j]) / max_exposure)
                    linear[(i, j)] -= 0.10 * marginal

    # Degree penalties are supplied by the evaluator.
    return linear, quadratic, 3.0


# EVOLVE-BLOCK-END


def run_surrogate(demand):
    """Fixed entry point called by the evaluator."""
    demand = np.array(demand, dtype=float)
    start = time.perf_counter()
    linear, quadratic, penalty_weight = surrogate_terms(demand.copy())
    return {
        "n": len(demand),
        "linear": linear,
        "quadratic": quadratic,
        "penalty_weight": penalty_weight,
        "seconds": time.perf_counter() - start,
    }
