"""Surrogate QUBO terms for QOBLIB 08 network design. ShinkaEvolve edits the EVOLVE block."""

import time

# EVOLVE-BLOCK-START
import numpy as np


def surrogate_terms(demand):
    """Build direct-link rewards and pressure-discounted relay-path terms."""
    matrix = np.asarray(demand, dtype=float)
    n = len(matrix)

    # Sanitize input and exclude self traffic.
    traffic = np.maximum(
        np.nan_to_num(matrix, nan=0.0, posinf=0.0, neginf=0.0), 0.0
    ).copy()
    np.fill_diagonal(traffic, 0.0)

    scale = float(np.max(traffic)) if traffic.size else 0.0
    if scale <= 0.0:
        scale = 1.0

    normalized = traffic / scale
    roots = np.sqrt(normalized)

    # Concave direct reward retains useful influence from moderate demands.
    linear = {
        (i, j): -float(0.75 * normalized[i, j] + 0.25 * roots[i, j])
        for i in range(n)
        for j in range(n)
        if i != j
    }

    incident = np.maximum(traffic.sum(axis=0), traffic.sum(axis=1))
    max_incident = float(np.max(incident)) if incident.size else 0.0
    if max_incident <= 0.0:
        max_incident = 1.0
    node_pressure = np.maximum(0.65, incident / max_incident)

    positive = traffic[traffic > 0.0]
    cutoff = float(np.quantile(positive, 0.75)) if positive.size else 0.0
    quadratic = {}

    for i in range(n):
        for k in range(n):
            if i == k or traffic[i, k] <= 0.0:
                continue

            support = 0.70 * normalized[i, k] + 0.30 * roots[i, k]
            candidates = []
            for j in range(n):
                if j == i or j == k:
                    continue

                left = float(normalized[i, j])
                right = float(normalized[j, k])
                edge_pressure = left + right

                # Balanced-leg affinity: favor relays where both constituent
                # links plausibly support the OD flow.
                weak = min(left, right)
                strong = max(left, right)
                affinity = weak + 0.25 * strong

                pressure = float(node_pressure[j])
                rank = (0.25 + affinity) / pressure
                discount = pressure * (1.0 + 0.5 * edge_pressure)
                candidates.append((float(rank), j, float(discount)))

            candidates.sort(key=lambda item: (-item[0], item[1]))
            keep = min(len(candidates), 4 if traffic[i, k] >= cutoff else 2)
            selected = set()

            for position, (_, j, discount) in enumerate(candidates[:keep]):
                selected.add(j)
                rank_factor = 1.0 + 0.08 * (keep - position)
                weight = -0.46 * float(support) * rank_factor / discount
                first, second = (i, j), (j, k)
                key = (first, second) if first < second else (second, first)
                quadratic[key] = quadratic.get(key, 0.0) + weight

            # Keep one pressure-favored backup relay, preserving a useful
            # alternative where the affinity ranking is imperfect.
            remaining = [item for item in candidates if item[1] not in selected]
            if remaining:
                _, j, discount = min(remaining, key=lambda item: (item[2], item[1]))
                first, second = (i, j), (j, k)
                key = (first, second) if first < second else (second, first)
                weight = -0.30 * float(support) / discount
                quadratic[key] = quadratic.get(key, 0.0) + weight

    return linear, quadratic, 1.25


# EVOLVE-BLOCK-END


def run_surrogate(demand):
    """Fixed entry point called by the evaluator, once per demand matrix."""
    demand = np.array(demand, dtype=float)
    start = time.perf_counter()
    linear, quadratic, penalty_weight = surrogate_terms(demand.copy())
    return {
        "n": len(demand),
        "linear": linear,
        "quadratic": quadratic,
        "penalty_weight": penalty_weight,
        "seconds": time.perf_counter() - start,
    }
