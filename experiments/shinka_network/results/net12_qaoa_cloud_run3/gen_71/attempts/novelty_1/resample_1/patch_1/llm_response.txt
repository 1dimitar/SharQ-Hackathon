<DESCRIPTION>
This keeps the recommended 0.85/0.15 direct-link calibration, but replaces the dense two-hop construction with a sparse relay-selection scheme. For each OD pair, it rewards only a few candidate two-hop routes, chosen using both demand and the usefulness of their constituent links. Relay pressure discounts routes through nodes with high incident traffic. The smaller interaction graph is intended to make QAOA partitions more focused while retaining plausible alternatives for routing traffic indirectly.

```python
"""Surrogate QUBO terms for QOBLIB 08 network design."""

import time

# EVOLVE-BLOCK-START
import math
import numpy as np


def surrogate_terms(demand):
    """Build direct-link rewards and sparse, relay-aware two-hop rewards."""
    traffic = np.asarray(demand, dtype=float)
    n = len(traffic)

    traffic = np.nan_to_num(traffic, nan=0.0, posinf=0.0, neginf=0.0)
    traffic = np.maximum(traffic, 0.0).copy()
    np.fill_diagonal(traffic, 0.0)

    scale = float(np.max(traffic)) if traffic.size else 0.0
    if scale <= 0.0:
        scale = 1.0
    q = traffic / scale

    # Keep direct traffic influential, while retaining a modest contribution
    # from medium-sized demands.
    linear = {}
    for i in range(n):
        for j in range(n):
            if i != j:
                value = float(q[i, j])
                linear[(i, j)] = -(0.85 * value + 0.15 * math.sqrt(value))

    incoming = np.sum(q, axis=0)
    outgoing = np.sum(q, axis=1)
    incident = np.maximum(incoming, outgoing)
    max_incident = float(np.max(incident)) if incident.size else 0.0
    if max_incident <= 0.0:
        max_incident = 1.0
    pressure = np.maximum(0.55, incident / max_incident)

    # Candidate relays are ranked separately for each OD pair. The balanced
    # leg affinity favors routes whose links both have useful traffic, rather
    # than routes supported by just one attractive leg.
    relay_budget = min(n - 2, max(2, int(math.sqrt(n))))
    quadratic = {}
    path_strength = 0.32

    for i in range(n):
        for k in range(n):
            if i == k:
                continue

            d = float(q[i, k])
            if d <= 0.0:
                continue
            support = 0.70 * d + 0.30 * math.sqrt(d)

            candidates = []
            for j in range(n):
                if j == i or j == k:
                    continue

                a = float(q[i, j])
                b = float(q[j, k])
                balanced_affinity = min(a, b) + 0.20 * max(a, b)
                relay_cost = 0.70 + float(pressure[j])
                leg_cost = 1.0 + 0.20 * (a + b)
                quality = (balanced_affinity + 0.015) / (relay_cost * leg_cost)
                candidates.append((quality, j))

            candidates.sort(key=lambda item: (-item[0], item[1]))
            for quality, j in candidates[:relay_budget]:
                first = (i, j)
                second = (j, k)
                key = (first, second) if first < second else (second, first)

                # Quality modulates the OD support gently: direct demand
                # remains the main signal, while candidate ranking controls
                # which relay alternatives enter the sparse interaction graph.
                modulation = 0.75 + 0.25 * min(1.0, quality * 2.0)
                quadratic[key] = -path_strength * support * modulation / (
                    0.70 + float(pressure[j])
                )

    # Degree penalties are supplied by the evaluator.
    penalty_weight = 1.5
    return linear, quadratic, penalty_weight


# EVOLVE-BLOCK-END


def run_surrogate(demand):
    """Fixed entry point called by the evaluator, once per demand matrix."""
    demand = np.array(demand, dtype=float)
    start = time.perf_counter()
    linear, quadratic, penalty_weight = surrogate_terms(demand.copy())
    return {
        "n": len(demand),
        "linear": linear,
        "quadratic": quadratic,
        "penalty_weight": penalty_weight,
        "seconds": time.perf_counter() - start,
    }
```
