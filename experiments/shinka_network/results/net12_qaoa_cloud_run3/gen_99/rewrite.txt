"""Surrogate QUBO terms for QOBLIB 08 network design."""

import time

# EVOLVE-BLOCK-START
import numpy as np


def _normalized_traffic(demand):
    traffic = np.maximum(np.asarray(demand, dtype=float), 0.0).copy()
    np.fill_diagonal(traffic, 0.0)
    maximum = float(np.max(traffic)) if traffic.size else 0.0
    if maximum > 0.0:
        traffic /= maximum
    return traffic


def _node_pressure(traffic):
    incident = np.maximum(traffic.sum(axis=0), traffic.sum(axis=1))
    maximum = float(np.max(incident)) if incident.size else 0.0
    if maximum <= 0.0:
        return np.ones(len(traffic), dtype=float)
    return np.maximum(0.65, incident / maximum)


def _rank_relays(traffic, pressure, relay_count):
    """Return the best relays for each ordered origin-destination pair."""
    n = len(traffic)

    # Axes are origin, destination, relay.
    first_leg = traffic[:, None, :]
    second_leg = traffic.T[None, :, :]
    affinity = np.sqrt((0.1 + first_leg) * (0.1 + second_leg))
    rank = affinity / (0.75 + pressure[None, None, :]) ** 1.25

    same_node = np.eye(n, dtype=bool)
    invalid = same_node[:, None, :] | same_node[None, :, :]
    rank = np.where(invalid, -np.inf, rank)

    # Stable sorting makes ties deterministic in node-index order.
    return np.argsort(-rank, axis=2, kind="stable")[:, :, :relay_count]


def surrogate_terms(demand):
    """Build direct-link costs and sparse pressure-aware path rewards."""
    traffic = _normalized_traffic(demand)
    n = len(traffic)
    support = 0.75 * traffic + 0.25 * np.sqrt(traffic)

    linear = {
        (i, j): -float(support[i, j])
        for i in range(n)
        for j in range(n)
        if i != j
    }

    pressure = _node_pressure(traffic)
    relay_count = min(n - 2, max(2, n // 4))
    relays = _rank_relays(traffic, pressure, relay_count)

    quadratic = {}
    origins, destinations = np.nonzero(traffic > 0.0)
    for origin, destination in zip(origins, destinations):
        i, k = int(origin), int(destination)
        if i == k:
            continue
        path_support = float(support[i, k])
        for relay in relays[i, k]:
            j = int(relay)
            first, second = (i, j), (j, k)
            key = (first, second) if first < second else (second, first)
            quadratic[key] = -0.25 * path_support / (0.75 + float(pressure[j]))

    # The evaluator supplies the degree penalties.
    return linear, quadratic, 1.5


# EVOLVE-BLOCK-END


def run_surrogate(demand):
    """Fixed entry point called by the evaluator, once per demand matrix."""
    demand = np.array(demand, dtype=float)
    start = time.perf_counter()
    linear, quadratic, penalty_weight = surrogate_terms(demand.copy())
    return {
        "n": len(demand),
        "linear": linear,
        "quadratic": quadratic,
        "penalty_weight": penalty_weight,
        "seconds": time.perf_counter() - start,
    }
