"""Surrogate QUBO terms for QOBLIB 08 network design."""

import time

# EVOLVE-BLOCK-START
import math
import numpy as np


def _prepare_demand(demand):
    """Sanitize traffic and compute the normalized matrix and node pressures."""
    traffic = np.asarray(demand, dtype=float)
    traffic = np.nan_to_num(
        traffic, nan=0.0, posinf=0.0, neginf=0.0
    )
    traffic = np.maximum(traffic, 0.0).copy()

    n = len(traffic)
    np.fill_diagonal(traffic, 0.0)

    scale = float(np.max(traffic)) if traffic.size else 0.0
    if scale <= 0.0:
        scale = 1.0

    normalized = traffic / scale
    incoming = np.sum(traffic, axis=0)
    outgoing = np.sum(traffic, axis=1)
    incident = np.maximum(incoming, outgoing)

    max_incident = float(np.max(incident)) if incident.size else 0.0
    if max_incident <= 0.0:
        max_incident = 1.0
    node_pressure = np.maximum(0.65, incident / max_incident)

    # Use all off-diagonal OD entries for a deterministic percentile cutoff.
    off_diagonal = traffic[~np.eye(n, dtype=bool)]
    cutoff = float(np.percentile(off_diagonal, 80.0)) if off_diagonal.size else 0.0

    return traffic, normalized, node_pressure, cutoff


def _direct_link_terms(normalized):
    """Build the existing concave direct-demand reward for each possible link."""
    n = len(normalized)
    linear = {}

    for i in range(n):
        for j in range(n):
            if i != j:
                value = float(normalized[i, j])
                linear[(i, j)] = -(0.75 * value + 0.25 * math.sqrt(value))

    return linear


def _relay_terms(traffic, normalized, node_pressure, cutoff):
    """Build sparse two-hop rewards, adjusting their strength by OD percentile."""
    n = len(normalized)
    relay_budget = min(n - 2, max(2, int(math.sqrt(n))))
    quadratic = {}

    for i in range(n):
        for k in range(n):
            if i == k:
                continue

            value = float(normalized[i, k])
            if value <= 0.0:
                continue

            support = 0.70 * value + 0.30 * math.sqrt(value)
            demand_factor = 1.1 if float(traffic[i, k]) >= cutoff else 0.9
            candidates = []

            for j in range(n):
                if j == i or j == k:
                    continue

                leg_a = float(normalized[i, j])
                leg_b = float(normalized[j, k])
                pressure = float(node_pressure[j]) * (
                    1.0 + 0.5 * (leg_a + leg_b)
                )
                affinity = min(leg_a, leg_b) + 0.25 * max(leg_a, leg_b) + 0.01
                candidates.append((affinity / pressure, j, pressure))

            candidates.sort(key=lambda item: (-item[0], item[1]))

            for _, j, pressure in candidates[:relay_budget]:
                first = (i, j)
                second = (j, k)
                key = (first, second) if first < second else (second, first)
                quadratic[key] = -0.30 * demand_factor * support / pressure

    return quadratic


def surrogate_terms(demand):
    """Return direct-link and percentile-adjusted sparse relay terms."""
    traffic, normalized, node_pressure, cutoff = _prepare_demand(demand)
    linear = _direct_link_terms(normalized)
    quadratic = _relay_terms(traffic, normalized, node_pressure, cutoff)

    # Degree penalties are supplied by the evaluator.
    penalty_weight = 1.5
    return linear, quadratic, penalty_weight


# EVOLVE-BLOCK-END


def run_surrogate(demand):
    """Fixed entry point called by the evaluator, once per demand matrix."""
    demand = np.array(demand, dtype=float)
    start = time.perf_counter()
    linear, quadratic, penalty_weight = surrogate_terms(demand.copy())
    return {
        "n": len(demand),
        "linear": linear,
        "quadratic": quadratic,
        "penalty_weight": penalty_weight,
        "seconds": time.perf_counter() - start,
    }
