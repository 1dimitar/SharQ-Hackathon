"""Sparse, high-demand relay surrogate for QOBLIB 08."""

import time

# EVOLVE-BLOCK-START
import math
import numpy as np


def surrogate_terms(demand):
    """Return direct-link and focused two-hop QUBO terms."""
    traffic = np.asarray(demand, dtype=float)
    n = len(traffic)

    traffic = np.nan_to_num(traffic, nan=0.0, posinf=0.0, neginf=0.0)
    traffic = np.maximum(traffic, 0.0).copy()
    np.fill_diagonal(traffic, 0.0)

    scale = float(np.max(traffic)) if traffic.size else 0.0
    if scale <= 0.0:
        scale = 1.0
    normalized = traffic / scale

    # Retain a direct-link incentive for every possible arc.
    linear = {}
    for i in range(n):
        for j in range(n):
            if i != j:
                value = float(normalized[i, j])
                linear[(i, j)] = -(0.78 * value + 0.22 * math.sqrt(value))

    # Select a deterministic top decile of positive-demand OD pairs.
    # Exact ranking avoids a large increase in term count when values tie
    # at the 90th-percentile cutoff.
    od_pairs = [
        (float(normalized[i, k]), i, k)
        for i in range(n)
        for k in range(n)
        if i != k and normalized[i, k] > 0.0
    ]
    od_pairs.sort(key=lambda item: (-item[0], item[1], item[2]))
    selected_count = int(math.ceil(0.10 * len(od_pairs)))
    selected_pairs = od_pairs[:selected_count]

    incoming = np.sum(normalized, axis=0)
    outgoing = np.sum(normalized, axis=1)
    incident = np.maximum(incoming, outgoing)
    max_incident = float(np.max(incident)) if incident.size else 0.0
    if max_incident > 0.0:
        pressure = incident / max_incident
    else:
        pressure = np.zeros(n, dtype=float)

    quadratic = {}
    relay_cap = min(2, max(0, n - 2))
    total_path_strength = 0.34

    for od_value, i, k in selected_pairs:
        if relay_cap == 0:
            continue

        support = 0.70 * od_value + 0.30 * math.sqrt(od_value)
        candidates = []

        for j in range(n):
            if j == i or j == k:
                continue

            leg_a = float(normalized[i, j])
            leg_b = float(normalized[j, k])

            # Prefer paths with two plausible legs, while discounting
            # already pressured relays and heavily demanded constituent arcs.
            leg_affinity = math.sqrt((0.04 + leg_a) * (0.04 + leg_b))
            congestion = (
                0.62 * float(pressure[j])
                + 0.24 * (leg_a + leg_b)
                + 0.14 * max(leg_a, leg_b)
            )
            quality = (0.05 + leg_affinity) / (0.55 + congestion)
            candidates.append((quality, j))

        candidates.sort(key=lambda item: (-item[0], item[1]))
        chosen = candidates[:relay_cap]
        quality_sum = sum(q for q, _ in chosen)
        if quality_sum <= 0.0:
            continue

        for quality, j in chosen:
            first, second = (i, j), (j, k)
            key = (first, second) if first < second else (second, first)
            weight = -total_path_strength * support * quality / quality_sum
            quadratic[key] = quadratic.get(key, 0.0) + float(weight)

    # Degree penalties are added by the evaluator, not encoded in these terms.
    penalty_weight = 2.0
    return linear, quadratic, penalty_weight


# EVOLVE-BLOCK-END


def run_surrogate(demand):
    """Fixed entry point called by the evaluator."""
    demand = np.array(demand, dtype=float)
    start = time.perf_counter()
    linear, quadratic, penalty_weight = surrogate_terms(demand.copy())
    return {
        "n": len(demand),
        "linear": linear,
        "quadratic": quadratic,
        "penalty_weight": penalty_weight,
        "seconds": time.perf_counter() - start,
    }
