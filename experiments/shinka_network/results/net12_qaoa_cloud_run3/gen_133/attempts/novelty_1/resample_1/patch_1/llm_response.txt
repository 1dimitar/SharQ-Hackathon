<DESCRIPTION>
This variant emphasizes large direct demands using the suggested 0.85/0.15 blend, while retaining a concave component so moderate demands still influence link selection. It adds sparse two-hop rewards for promising relay paths, ranking relays by the strength of both legs and discounting nodes with high incident traffic. The relay budget is kept small to limit interaction-graph density and preserve useful QAOA communities. Degree penalties remain the evaluator’s responsibility.

"""Surrogate QUBO terms for QOBLIB 08 network design. ShinkaEvolve edits the EVOLVE block."""

import time

# EVOLVE-BLOCK-START
import math
import numpy as np


def surrogate_terms(demand):
    """Build direct-link rewards and sparse, pressure-aware relay rewards."""
    traffic = np.maximum(np.asarray(demand, dtype=float), 0.0).copy()
    n = len(traffic)
    np.fill_diagonal(traffic, 0.0)

    scale = float(np.max(traffic)) if traffic.size else 0.0
    if scale <= 0.0:
        scale = 1.0
    normalized = traffic / scale

    # Emphasize the largest direct demands, without discarding moderate ones.
    linear = {}
    for i in range(n):
        for j in range(n):
            if i != j:
                value = float(normalized[i, j])
                linear[(i, j)] = -(0.85 * value + 0.15 * math.sqrt(value))

    if n < 3 or not np.any(traffic):
        return linear, {}, 1.5

    incoming = np.sum(traffic, axis=0)
    outgoing = np.sum(traffic, axis=1)
    incident = np.maximum(incoming, outgoing)
    pressure_scale = float(np.max(incident))
    if pressure_scale <= 0.0:
        pressure_scale = 1.0
    pressure = np.maximum(0.60, incident / pressure_scale)

    positive = normalized[normalized > 0.0]
    high_cutoff = float(np.quantile(positive, 0.85)) if positive.size else float("inf")
    base_relays = min(n - 2, max(1, n // 6))
    extra_relays = min(2, max(0, n - 2 - base_relays))

    quadratic = {}
    for i in range(n):
        for k in range(n):
            value = float(normalized[i, k])
            if i == k or value <= 0.0:
                continue

            # Stronger OD flows receive more relay support, but the square-root
            # component prevents smaller flows from disappearing entirely.
            support = 0.78 * value + 0.22 * math.sqrt(value)
            candidates = []

            for j in range(n):
                if j == i or j == k:
                    continue

                a = float(normalized[i, j])
                b = float(normalized[j, k])

                # A route is only attractive when both legs have support.
                # The small floor preserves alternative relays for sparse data.
                affinity = math.sqrt((min(a, b) + 1.0e-5) *
                                     (0.20 * max(a, b) + min(a, b) + 1.0e-5))
                relay_pressure = float(pressure[j]) * (1.0 + 0.35 * (a + b))
                rank = affinity / relay_pressure
                candidates.append((rank, j, relay_pressure, affinity))

            candidates.sort(key=lambda item: (-item[0], item[1]))
            keep = base_relays + (extra_relays if value >= high_cutoff else 0)

            for _, j, relay_pressure, affinity in candidates[:keep]:
                first = (i, j)
                second = (j, k)
                key = (first, second) if first < second else (second, first)
                weight = -0.34 * support * affinity / relay_pressure
                quadratic[key] = quadratic.get(key, 0.0) + weight

    return linear, quadratic, 1.5


# EVOLVE-BLOCK-END


def run_surrogate(demand):
    """Fixed entry point called by the evaluator, once per demand matrix."""
    demand = np.array(demand, dtype=float)
    start = time.perf_counter()
    linear, quadratic, penalty_weight = surrogate_terms(demand.copy())
    return {
        "n": len(demand),
        "linear": linear,
        "quadratic": quadratic,
        "penalty_weight": penalty_weight,
        "seconds": time.perf_counter() - start,
    }
