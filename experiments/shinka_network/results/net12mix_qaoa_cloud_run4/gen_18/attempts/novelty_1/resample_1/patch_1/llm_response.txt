<DESCRIPTION>
The current relay bonus spreads each demand pair’s credit across every possible intermediate node, making individual two-hop couplings very weak and creating many interactions. Instead, select the two most promising relays for each demand pair, favoring links with strong existing rewards and mildly favoring nodes with lower traffic exposure as potential transit nodes. Split a bounded bonus between those relays. This makes useful relay interactions more salient to the QUBO and reduces the number of quadratic terms, while retaining the existing direct-link and endpoint rewards.
</DESCRIPTION>

def _make_path_couplings(t, rewards):
    """Add sparse, restrained bonuses for useful two-hop routes.

    For each ordered demand pair, its intermediate-node opportunities are
    normalized collectively. Thus a single demand cannot create a large
    bonus merely because many alternative paths exist.
    """
    n = len(t)
    maximum = float(t.max()) if t.size else 0.0
    if maximum <= 0:
        return {}

    quadratic = {}
    for i in range(n):
        for k in range(n):
            if i == k or t[i, k] <= 0:
                continue

            candidates = []
            for j in range(n):
                if j != i and j != k:
                    # Favor intermediates with traffic-related capacity,
                    # but keep the adjustment bounded.
                    capacity_hint = 1.0 + 0.25 * (
                        rewards[(i, j)] + rewards[(j, k)]
                    )
                    candidates.append((j, capacity_hint))

            denominator = sum(weight for _, weight in candidates)
            if denominator <= 0:
                continue

            demand_strength = float(t[i, k]) / maximum
            for j, weight in candidates:
                a, b = (i, j), (j, k)
                key = (a, b)
                # A two-hop option is useful, but should remain weaker than
                # selecting a valuable direct link.
                value = -0.18 * demand_strength * weight / denominator
                quadratic[key] = quadratic.get(key, 0.0) + value

    return quadratic

def _make_path_couplings(t, rewards):
    """Give bounded two-hop credit to the strongest relay options per pair."""
    n = len(t)
    maximum = float(t.max()) if t.size else 0.0
    if maximum <= 0:
        return {}

    row = t.sum(axis=1)
    col = t.sum(axis=0)
    total = float(t.sum())
    node_exposure = (row + col) / total if total > 0 else np.zeros(n)
    quadratic = {}

    for i in range(n):
        for k in range(n):
            if i == k or t[i, k] <= 0:
                continue

            candidates = []
            for j in range(n):
                if j != i and j != k:
                    # Prefer relays whose two links are already attractive;
                    # modestly favor less traffic-loaded transit nodes.
                    link_hint = 1.0 + 0.25 * (
                        rewards[(i, j)] + rewards[(j, k)]
                    )
                    relay_hint = link_hint / (1.0 + 2.0 * float(node_exposure[j]))
                    candidates.append((relay_hint, j))

            candidates.sort(reverse=True)
            selected = candidates[:2]
            denominator = sum(score for score, _ in selected)
            if denominator <= 0:
                continue

            demand_strength = float(t[i, k]) / maximum
            for score, j in selected:
                key = ((i, j), (j, k))
                value = -0.24 * demand_strength * score / denominator
                quadratic[key] = quadratic.get(key, 0.0) + value

    return quadratic
