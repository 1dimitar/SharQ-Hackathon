"""Surrogate QUBO terms for QOBLIB 08 network design."""

import time

# EVOLVE-BLOCK-START
import numpy as np


def surrogate_terms(demand):
    """Build sparse endpoint-load balancing QUBO terms."""
    t = np.asarray(demand, dtype=float)
    n = len(t)
    if n == 0:
        return {}, {}, 3.0

    t = np.maximum(t, 0.0).copy()
    np.fill_diagonal(t, 0.0)

    maximum = float(np.max(t)) if t.size else 0.0
    if maximum > 0.0:
        t /= maximum

    row = t.sum(axis=1)
    col = t.sum(axis=0)
    links = [(i, j) for i in range(n) for j in range(n) if i != j]

    # Concavity prevents a few exceptionally large demands from overwhelming
    # the rest of the topology decisions.
    linear = {}
    for i, j in links:
        direct = float(t[i, j])
        endpoint = 0.0
        if row[i] > 0.0:
            endpoint += float(t[i, j]) / float(row[i])
        if col[j] > 0.0:
            endpoint += float(t[i, j]) / float(col[j])
        linear[(i, j)] = -(direct ** 0.55 + 0.10 * endpoint)

    quadratic = {}

    # Competing outgoing links: both consume the limited egress capacity of
    # the same origin. Pair weights reflect their relative demand burdens.
    for i in range(n):
        if row[i] <= 0.0:
            continue
        outgoing = [(i, j) for j in range(n) if j != i]
        for a in range(len(outgoing)):
            e1 = outgoing[a]
            p1 = float(t[e1]) / float(row[i])
            for b in range(a + 1, len(outgoing)):
                e2 = outgoing[b]
                p2 = float(t[e2]) / float(row[i])
                key = (e1, e2) if e1 <= e2 else (e2, e1)
                quadratic[key] = quadratic.get(key, 0.0) + 0.22 * np.sqrt(p1 * p2)

    # Competing incoming links: high-demand origins targeting the same
    # destination can concentrate traffic on its two ingress links.
    for j in range(n):
        if col[j] <= 0.0:
            continue
        incoming = [(i, j) for i in range(n) if i != j]
        for a in range(len(incoming)):
            e1 = incoming[a]
            p1 = float(t[e1]) / float(col[j])
            for b in range(a + 1, len(incoming)):
                e2 = incoming[b]
                p2 = float(t[e2]) / float(col[j])
                key = (e1, e2) if e1 <= e2 else (e2, e1)
                quadratic[key] = quadratic.get(key, 0.0) + 0.22 * np.sqrt(p1 * p2)

    largest = max(
        [abs(v) for v in linear.values()] +
        [abs(v) for v in quadratic.values()] +
        [1e-12]
    )
    inv = 1.0 / largest
    linear = {k: float(v * inv) for k, v in linear.items()}
    quadratic = {k: float(v * inv) for k, v in quadratic.items()}

    return linear, quadratic, 2.0


# EVOLVE-BLOCK-END


def run_surrogate(demand):
    """Fixed entry point called by the evaluator, once per demand matrix."""
    demand = np.array(demand, dtype=float)
    start = time.perf_counter()
    linear, quadratic, penalty_weight = surrogate_terms(demand.copy())
    return {
        "n": len(demand),
        "linear": linear,
        "quadratic": quadratic,
        "penalty_weight": penalty_weight,
        "seconds": time.perf_counter() - start,
    }
