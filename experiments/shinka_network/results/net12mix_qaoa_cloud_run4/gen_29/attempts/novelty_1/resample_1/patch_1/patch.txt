 def surrogate_terms(demand):
     """Build the surrogate cost terms of the topology QUBO from the demand matrix.
 
     Args:
         demand: (n, n) float array; demand[i, j] is the traffic from node i to node j
             (zero diagonal).
 
     Returns:
         linear: dict {(i, j): weight}. Weight on the binary variable x_ij
             (x_ij = 1 if the directed link i -> j is built), i != j.

         penalty_weight: float. Strength of the degree penalty that the evaluator adds
             after rescaling these terms so the largest |weight| is 1.
 
     Lower energy must mean a better topology. Baseline: reward each link by the demand
     it carries directly.
     """

     d = np.asarray(demand, dtype=float)
     scale = float(d.max()) or 1.0
 
     # A link's exposure includes its direct traffic and a modest estimate of
     # transit pressure at its endpoints.
     row = d.sum(axis=1)
     col = d.sum(axis=0)
     exposure = {}

             if i == j:
                 continue

+            pressure = 0.20 * (float(row[i]) + float(col[j])) / max(1, n - 1)
             exposure[i, j] = direct + pressure
             linear[i, j] = -direct / scale
 
     # Penalize pairs of selected links that concentrate estimated exposure
     # at the same sending or receiving node. Each pair is included once.
     quadratic = {}
     pair_scale = 0.18 / scale
     for center in range(n):
         outgoing = [(center, j) for j in range(n) if j != center]
         incoming = [(i, center) for i in range(n) if i != center]

                 for b in range(a + 1, len(links)):
                     e2 = links[b]
                     weight = pair_scale * exposure[e1] * exposure[e2] / scale
                     if weight:
                         quadratic[(e1, e2)] = weight
 
     penalty_weight = 4.0
     return linear, quadratic, penalty_weight

 def run_surrogate(demand):
     """Fixed entry point called by the evaluator, once per demand matrix."""
