<DESCRIPTION>
Replace energy-only batch selection with a small, mixed shortlist ranked by both QUBO energy and demand-weighted shortest-path distance. The distance proxy is independent of the pool’s Hamiltonian and costs no LP calls; combining its rank with energy should help reject moves that look good to the surrogate but make traffic routes longer. Keep energy and proxy selections mixed, and preserve some within-batch structural diversity so the new signal does not dominate the search. The proxy is computed only for a bounded shortlist, keeping its overhead adaptive in n.

    def screen_batch(moves, parent, batch_size=9):
        """Rank a bounded move shortlist by both QUBO energy and demand distances."""
        fresh = [
            m for m in moves
            if m not in seen and not tools.is_cached(m)
        ]
        if not fresh:
            return []

        fresh.sort(key=energy)
        # Keep the proxy calculation bounded while sampling beyond the energy head.
        head_count = min(len(fresh), 3 * n)
        shortlist = list(fresh[:head_count])
        rest = fresh[head_count:]
        if rest:
            sample_count = min(len(rest), n)
            ids = rng.choice(len(rest), size=sample_count, replace=False)
            shortlist.extend(rest[int(i)] for i in ids)

        D = np.asarray(demand, dtype=float)
        proxy = np.empty(len(shortlist), dtype=float)
        for i, net in enumerate(shortlist):
            dist = np.full((n, n), float(n + 1))
            np.fill_diagonal(dist, 0.0)
            for a, b in net:
                dist[a, b] = 1.0
            for k in range(n):
                np.minimum(
                    dist,
                    dist[:, k:k + 1] + dist[k:k + 1, :],
                    out=dist,
                )
            proxy[i] = float(np.sum(D * dist))

        energies = np.asarray([energy(net) for net in shortlist])
        order_e = np.argsort(energies, kind="stable")
        order_p = np.argsort(proxy, kind="stable")
        rank_e = np.empty(len(shortlist), dtype=float)
        rank_p = np.empty(len(shortlist), dtype=float)
        rank_e[order_e] = np.arange(len(shortlist))
        rank_p[order_p] = np.arange(len(shortlist))
        order_mix = np.argsort(rank_e + rank_p, kind="stable")

        selected = []
        selected_ids = set()

        def take(order, limit):
            for raw in order:
                if len(selected) >= limit:
                    break
                i = int(raw)
                if i in selected_ids:
                    continue
                if any(distance(shortlist[i], old) < 2 for old in selected):
                    continue
                selected.append(shortlist[i])
                selected_ids.add(i)

        # Reserve opportunities for each signal before filling by their consensus.
        take(order_e, min(2, batch_size))
        take(order_p, min(4, batch_size))
        take(order_mix, batch_size)

        # If diversity constraints prevented a full batch, fill with the best
        # remaining consensus moves.
        for raw in order_mix:
            if len(selected) >= batch_size:
                break
            i = int(raw)
            if i not in selected_ids:
                selected.append(shortlist[i])
                selected_ids.add(i)
        return selected
