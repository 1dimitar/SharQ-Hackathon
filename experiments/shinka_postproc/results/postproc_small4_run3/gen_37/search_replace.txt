<DESCRIPTION>
Replace the fixed energy/novelty screening mix with four complementary signals: QUBO energy, LP link pressure, structural novelty, and random exploration. Each signal gets a minimum slot when possible; remaining slots are allocated according to its exponentially smoothed rate of beating the current parent. This lets the screening portfolio adapt to the instance and pool while retaining exploration and avoiding dependence on any single noisy signal.

    def screen_batch(moves, parent, detail=None, batch_size=9):
        """Allocate LP trials adaptively across complementary screening signals."""
        screen_sources.clear()
        fresh = [
            m for m in moves
            if m not in seen and not tools.is_cached(m)
        ]
        if not fresh:
            return []

        target = min(batch_size, len(fresh))
        edges = {net: set(net) for net in fresh}
        orders = {
            "energy": sorted(fresh, key=energy),
        }

        # Use dual prices and loads to prioritize exchanges that remove
        # pressure from existing links and avoid high-demand replacement links.
        loads = {
            tuple(map(int, a)): float(v)
            for a, v in (detail.get("loads", {}) if detail else {}).items()
        }
        prices = {
            tuple(map(int, a)): float(v)
            for a, v in (detail.get("prices", {}) if detail else {}).items()
        }
        max_load = max(loads.values(), default=1.0) or 1.0
        max_price = max(prices.values(), default=1.0) or 1.0
        max_demand = float(np.max(demand)) if np.size(demand) else 1.0
        if max_demand <= 0:
            max_demand = 1.0
        parent_set = set(parent)
        pressure = []
        for net in fresh:
            net_set = edges[net]
            removed = parent_set - net_set
            added = net_set - parent_set
            value = sum(
                0.65 * loads.get(a, 0.0) / max_load
                + 0.35 * prices.get(a, 0.0) / max_price
                for a in removed
            )
            value -= sum(
                0.45 * float(demand[a[0], a[1]]) / max_demand
                for a in added
            )
            pressure.append((value, net))
        orders["pressure"] = [net for _, net in sorted(pressure, reverse=True)]

        # Novelty is measured against the archive so candidates from genuinely
        # different regions are considered, not merely different from siblings.
        references = [set(net) for net in archive()[:max(8, n)]]
        orders["novelty"] = sorted(
            fresh,
            key=lambda net: min(
                (len(edges[net] ^ ref) for ref in references),
                default=2 * n,
            ),
            reverse=True,
        )
        random_order = list(fresh)
        rng.shuffle(random_order)
        orders["random"] = random_order

        selected = []
        selected_set = set()
        allocated = {name: 0 for name in orders}

        def take(name):
            for net in orders[name]:
                if net not in selected_set:
                    selected.append(net)
                    selected_set.add(net)
                    screen_sources[net] = name
                    allocated[name] += 1
                    return True
            return False

        # Reserve one trial for each signal that has an unused candidate.
        for name in ("energy", "pressure", "novelty", "random"):
            if len(selected) < target:
                take(name)

        # Allocate remaining trials in proportion to recent success, with a
        # diminishing return for signals that already received more slots.
        weights = {
            name: (0.35 + signal_rate[name]) ** 2
            for name in orders
        }
        while len(selected) < target:
            available = [
                name for name in orders
                if any(net not in selected_set for net in orders[name])
            ]
            if not available:
                break
            name = max(
                available,
                key=lambda key: weights[key] / (1.0 + allocated[key]),
            )
            if not take(name):
                break

        return selected[:target]
