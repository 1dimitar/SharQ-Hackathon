<DESCRIPTION>
Select archive parents using both solution quality and search history, rather than choosing uniformly from the elite or the whole archive. Reward parents whose neighborhoods have produced improvements, while giving underexplored and structurally distinct networks a greater chance of selection. Update usage counts and accumulated improvement after each batch. This preserves elite exploitation but reduces repeated searches in unproductive regions and encourages exploration of diverse basins.
</DESCRIPTION>

    def select_parent(pop):
        """Favor quality and productive parents while reserving weight for exploration."""
        if not pop:
            return best_net

        total_uses = sum(parent_uses.values())
        weights = []
        for rank, net in enumerate(pop):
            uses = parent_uses.get(net, 0)
            nearest = min(
                (distance(net, other) for other in pop if other != net),
                default=26,
            )
            novelty = min(1.0, nearest / 26.0)
            quality = 1.0 / (1.0 + 0.04 * rank)
            productivity = parent_gain.get(net, 0.0) / max(1, uses)
            productivity_factor = 0.75 + min(1.5, 500.0 * productivity)
            exploration = 0.35 * math.sqrt(
                math.log(total_uses + 2.0) / (uses + 1.0)
            )
            weights.append(
                quality * (0.55 + novelty) * productivity_factor
                * (1.0 + exploration)
            )

        pick = float(rng.random()) * sum(weights)
        for net, weight in zip(pop, weights):
            pick -= weight
            if pick <= 0:
                return net
        return pop[-1]
