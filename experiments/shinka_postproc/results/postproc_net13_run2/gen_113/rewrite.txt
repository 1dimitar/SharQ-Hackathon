"""Classical post-processing of QAOA output for QOBLIB 08."""

import json
import time

import numpy as np

# EVOLVE-BLOCK-START
def improve(demand, candidates, tools, rng):
    """Search repaired candidates using an adaptive, bottleneck-focused archive."""
    scored = {}
    best_net = None
    best_score = float("inf")
    stalls = {}
    op_trials = {"switch": 1, "rotation": 1, "kick": 1}
    op_wins = {"switch": 0, "rotation": 0, "kick": 0}

    def canon(arcs):
        return tuple(sorted(tuple(map(int, edge)) for edge in arcs))

    def dist(a, b):
        return len(set(a) ^ set(b))

    def energy(net):
        try:
            return float(tools.energy(net))
        except Exception:
            return 0.0

    def evaluate(net):
        nonlocal best_net, best_score
        net = canon(net)
        if net not in scored:
            scored[net] = tools.lp(net)
        value = scored[net]
        if value < best_score:
            best_net, best_score = net, value
        return value

    repaired = set()
    for candidate in candidates:
        try:
            net = canon(tools.repair(candidate))
            if tools.is_valid(net):
                repaired.add(net)
        except Exception:
            continue

    if not repaired:
        try:
            net = canon(tools.repair([]))
            if tools.is_valid(net):
                repaired.add(net)
        except Exception:
            return []

    ordered = sorted(repaired, key=energy)
    seeds = []
    for net in ordered:
        if all(dist(net, old) >= 4 for old in seeds):
            seeds.append(net)
        if len(seeds) >= min(30, len(ordered)):
            break
    for net in ordered:
        if len(seeds) >= min(30, len(ordered)):
            break
        if net not in seeds:
            seeds.append(net)

    for net in seeds:
        if tools.lp_calls_left <= 0:
            break
        try:
            evaluate(net)
        except Exception as exc:
            if type(exc).__name__ == "BudgetExhausted":
                break
            continue
    if not scored:
        return list(seeds[0])

    def archive():
        ranked = sorted(scored, key=scored.get)
        out = []
        for net in ranked:
            if all(dist(net, old) >= 4 for old in out):
                out.append(net)
            if len(out) >= 24:
                break
        for net in ranked:
            if len(out) >= 32:
                break
            if net not in out:
                out.append(net)
        return out

    def choose_parent(pop):
        weights = []
        for net in pop:
            quality = 1.0 / (
                1.0 + max(0.0, scored[net] - best_score) / max(1.0, best_score)
            )
            exploration = 1.0 / (1.0 + stalls.get(net, 0))
            weights.append(quality * (0.35 + 0.65 * exploration))
        total = sum(weights)
        r = float(rng.random()) * total
        for net, weight in zip(pop, weights):
            r -= weight
            if r <= 0:
                return net
        return pop[-1]

    def switch(current, e1, e2):
        a, b = e1
        c, d = e2
        if a == d or c == b or (a, d) in current or (c, b) in current:
            return None
        return (current - {e1, e2}) | {(a, d), (c, b)}

    def make_proposals(parent, detail):
        current = set(parent)
        arcs = list(parent)
        hot = {
            tuple(map(int, edge))
            for edge, price in (detail.get("prices", {}) if detail else {}).items()
            if price > 0
        }
        if not hot and detail:
            loads = detail.get("loads", {})
            hot = {
                edge for _, edge in sorted(
                    ((float(v), tuple(map(int, e))) for e, v in loads.items()),
                    reverse=True,
                )[:6]
            }

        rates = {
            op: (op_wins[op] + 1.0) / (op_trials[op] + 4.0)
            for op in op_trials
        }
        avg = sum(rates.values()) / len(rates)
        factor = {
            op: min(1.8, max(0.55, rates[op] / max(avg, 1e-12)))
            for op in rates
        }
        proposals = {}

        # Exhaustive switches touching at least one priced bottleneck.
        for i in range(len(arcs)):
            for j in range(i + 1, len(arcs)):
                if hot and arcs[i] not in hot and arcs[j] not in hot:
                    continue
                changed = switch(current, arcs[i], arcs[j])
                if changed:
                    net = tuple(sorted(changed))
                    if net != parent:
                        proposals[net] = "switch"

        # Unrestricted switches preserve exploration under dual degeneracy.
        for _ in range(int(70 * factor["switch"])):
            i = int(rng.integers(len(arcs)))
            j = int(rng.integers(len(arcs) - 1))
            if j >= i:
                j += 1
            changed = switch(current, arcs[i], arcs[j])
            if changed:
                net = tuple(sorted(changed))
                if net != parent:
                    proposals[net] = "switch"

        stall = stalls.get(parent, 0)
        rotation_count = min(
            300, int(70 * (1 + stall // 3) * factor["rotation"])
        )
        for _ in range(rotation_count):
            ids = rng.choice(len(arcs), size=3, replace=False)
            e1, e2, e3 = (arcs[int(i)] for i in ids)
            old = {e1, e2, e3}
            if hot and not (old & hot):
                continue
            a, b = e1
            c, d = e2
            e, f = e3
            for additions in (
                {(a, d), (c, f), (e, b)},
                {(a, f), (c, b), (e, d)},
            ):
                if len(additions) != 3 or any(x == y for x, y in additions):
                    continue
                if additions & (current - old):
                    continue
                net = tuple(sorted((current - old) | additions))
                if net != parent:
                    proposals[net] = "rotation"

        # Random short walks provide larger moves across local-search basins.
        for _ in range(int((18 + min(stall, 8)) * factor["kick"])):
            walk = set(parent)
            for _ in range(2 + int(rng.integers(4))):
                now = list(walk)
                i, j = rng.choice(len(now), size=2, replace=False)
                changed = switch(walk, now[int(i)], now[int(j)])
                if changed:
                    walk = changed
            net = tuple(sorted(walk))
            if net != parent:
                proposals[net] = "kick"
        return proposals

    def screen(proposals, parent, detail, batch_size):
        fresh = [
            net for net in proposals
            if net not in scored and not tools.is_cached(net) and tools.is_valid(net)
        ]
        if not fresh:
            return []
        base_e = energy(parent)
        ranked = sorted(
            ((energy(net) - base_e, net) for net in fresh),
            key=lambda item: item[0],
        )

        # Compare more candidates when the surrogate is indecisive or a basin stalls.
        stall = stalls.get(parent, 0)
        if len(ranked) >= 8:
            values = [delta for delta, _ in ranked[:8]]
            spread = max(values) - min(values)
            if spread <= 0.02 * max(1.0, abs(base_e), abs(values[0])):
                batch_size = max(batch_size, 16)
        if stall >= 4:
            batch_size = max(batch_size, 14)
        if stall >= 9:
            batch_size = max(batch_size, 22)
        batch_size = min(24, batch_size)

        hot = {
            tuple(map(int, edge))
            for edge, price in (detail.get("prices", {}) if detail else {}).items()
            if price > 0
        }
        parent_edges = set(parent)
        selected = []
        if hot:
            focused = sorted(
                (
                    (delta, -len((parent_edges ^ set(net)) & hot), net)
                    for delta, net in ranked
                    if (parent_edges ^ set(net)) & hot
                ),
                key=lambda item: (item[0], item[1]),
            )
            for _, _, net in focused:
                if all(dist(net, old) >= 2 for old in selected):
                    selected.append(net)
                if len(selected) >= min(3, batch_size):
                    break

        for _, net in ranked:
            if len(selected) >= min(6, batch_size):
                break
            if net not in selected and all(dist(net, old) >= 2 for old in selected):
                selected.append(net)
        for _, net in ranked:
            if len(selected) >= batch_size:
                break
            if net not in selected:
                selected.append(net)
        return selected

    rounds = 0
    while tools.lp_calls_left > 0:
        rounds += 1
        pop = archive()
        if not pop:
            break
        parent = choose_parent(pop)
        try:
            detail = tools.lp_detail(parent)
        except Exception:
            detail = None

        proposals = make_proposals(parent, detail)
        batch_size = min(12, 6 + stalls.get(parent, 0) // 3)
        batch = screen(proposals, parent, detail, batch_size)

        if not batch:
            alternatives = [net for net in pop if net != parent]
            rng.shuffle(alternatives)
            for alternative in alternatives[:6]:
                try:
                    alt_detail = tools.lp_detail(alternative)
                except Exception:
                    alt_detail = None
                alt_props = make_proposals(alternative, alt_detail)
                batch = screen(alt_props, alternative, alt_detail, batch_size)
                if batch:
                    parent, detail, proposals = alternative, alt_detail, alt_props
                    break

        if not batch:
            if rounds > 10000:
                break
            continue

        parent_score = scored.get(parent, float("inf"))
        batch_best = float("inf")
        for net in batch:
            if tools.lp_calls_left <= 0:
                break
            try:
                value = evaluate(net)
            except Exception as exc:
                if type(exc).__name__ == "BudgetExhausted":
                    return list(best_net) if best_net is not None else []
                continue
            batch_best = min(batch_best, value)
            changed = dist(net, parent)
            kind = "switch" if changed <= 4 else (
                "rotation" if changed <= 6 else proposals.get(net, "kick")
            )
            op_trials[kind] += 1
            if value < parent_score:
                op_wins[kind] += 1

        if batch_best < parent_score:
            stalls[parent] = 0
        else:
            stalls[parent] = stalls.get(parent, 0) + 1

    return list(best_net) if best_net is not None else []
# EVOLVE-BLOCK-END


def run_postproc(pool_path, lp_budget, seed, control=False, time_limit=150):
    """Fixed entry point called by the evaluator: one run on one candidate pool."""
    import importlib.util
    import signal
    from pathlib import Path

    import oracle

    pool = json.load(open(pool_path))
    import qoblib_tools as qt

    if "hamiltonian_file" in pool:
        spec = importlib.util.spec_from_file_location(
            f"hamiltonian_{pool['hamiltonian_tag']}",
            Path(pool_path).resolve().parent / pool["hamiltonian_file"],
        )
        ham = importlib.util.module_from_spec(spec)
        spec.loader.exec_module(ham)
        surrogate_terms = ham.surrogate_terms
    else:
        from hamiltonian import surrogate_terms

    n = int(pool["instance"][-2:])
    demand = qt.load_network_demand(n)
    linear, quadratic, _ = surrogate_terms(np.array(demand, dtype=float))
    tools = oracle.Tools(
        demand,
        lp_budget,
        (
            {tuple(map(int, a)): float(w) for a, w in linear.items()},
            {
                (tuple(map(int, a)), tuple(map(int, b)): float(w)
                for (a, b), w in quadratic.items()
            },
        ),
    )
    candidates = [
        [tuple(map(int, a)) for a in c["arcs"]]
        for c in pool["candidates"]
    ]
    if control:
        crng = np.random.default_rng(10_000 + seed)
        candidates = [qt.random_topology(n, crng) for _ in candidates]

    def _timeout(signum, frame):
        raise oracle.RunTimeout()

    start = time.perf_counter()
    returned, error, timed_out = None, None, False
    signal.signal(signal.SIGALRM, _timeout)
    signal.alarm(int(time_limit))
    try:
        returned = improve(demand.copy(), candidates, tools, np.random.default_rng(seed))
    except oracle.BudgetExhausted:
        pass
    except oracle.RunTimeout:
        timed_out = True
    except Exception as exc:
        error = f"{type(exc).__name__}: {exc}"
    finally:
        signal.alarm(0)

    best_score, best_arcs = tools.certified_best()
    if returned is not None:
        try:
            ret = [tuple(map(int, a)) for a in returned]
            if qt.is_valid_topology(n, ret):
                score = qt.route_min_congestion(demand, ret).objective
                if score < best_score:
                    best_score, best_arcs = score, tuple(sorted(ret))
        except Exception:
            pass
    return {
        "pool": pool_path,
        "control": control,
        "n": n,
        "error": error,
        "best": float(best_score),
        "arcs": [list(a) for a in best_arcs] if best_arcs else None,
        "lp_calls": tools.lp_calls_used,
        "seconds": time.perf_counter() - start,
        "timed_out": timed_out,
    }
