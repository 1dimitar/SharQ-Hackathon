<DESCRIPTION>
Add a modest recency bonus to the existing quality-, novelty-, productivity-, and usage-based parent selection. The current usage bonus discourages repeatedly selecting a parent, but does not distinguish a recently explored parent from one that has been neglected for many rounds; likewise, `parent_stalls` is tracked but does not affect selection. The added factors gently favor underexplored or persistently stalled archive members without overriding the existing quality bias. Record the selected round so the recency signal updates consistently.
</DESCRIPTION>

    def select_parent(pop):
        """Balance quality with novelty, productivity, and underexploration."""
        if not pop:
            return best_net
        total_uses = sum(parent_uses.values())
        weights = []
        for rank, net in enumerate(pop):
            uses = parent_uses.get(net, 0)
            nearest = min(
                (distance(net, other) for other in pop if other != net),
                default=26,
            )
            novelty = min(1.0, nearest / 26.0)
            quality = 1.0 / (1.0 + 0.04 * rank)
            productivity = parent_gain.get(net, 0.0) / max(1, uses)
            productivity_factor = 0.75 + min(1.5, 500.0 * productivity)
            exploration = 0.35 * math.sqrt(
                math.log(total_uses + 2.0) / (uses + 1.0)
            )
            age = rounds - parent_last_selected.get(net, 0)
            recency_factor = (
                1.0 + 0.30 * min(1.0, age / 12.0)
                + 0.025 * min(6, parent_stalls.get(net, 0))
            )
            weights.append(
                quality * (0.55 + novelty) * productivity_factor
                * (1.0 + exploration) * recency_factor
            )

        pick = float(rng.random()) * sum(weights)
        chosen = pop[-1]
        for net, weight in zip(pop, weights):
            pick -= weight
            if pick <= 0:
                chosen = net
                break
        parent_last_selected[chosen] = rounds
        return chosen
