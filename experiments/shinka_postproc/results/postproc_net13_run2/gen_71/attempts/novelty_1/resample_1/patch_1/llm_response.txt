<DESCRIPTION>
Reorganizes the search around a bounded score-and-diversity archive, with separate proposal and batch-selection stages. It seeds that archive from repaired candidates, then uses LP bottleneck links to guide switches and rotations while reserving a few proposals for consensus edges and exploration. Each batch mixes surrogate-promising moves with novel moves, and parent selection adapts to search history to reduce repeated work.
</DESCRIPTION>

# EVOLVE-BLOCK-START
def improve(demand, candidates, tools, rng):
    """Improve repaired QAOA networks with an adaptive, diverse LP-scored archive."""
    n = tools.n
    scored = {}
    best_net = None
    best_score = float("inf")
    visits = {}
    stalls = {}
    repaired = set()

    def canon(arcs):
        return tuple(sorted(tuple(map(int, a)) for a in arcs))

    def energy(net):
        try:
            return float(tools.energy(net))
        except Exception:
            return 0.0

    def distance(a, b):
        return len(set(a) ^ set(b))

    def evaluate(net):
        nonlocal best_net, best_score
        net = canon(net)
        if net not in scored:
            value = tools.lp(net)
            scored[net] = value
        value = scored[net]
        if value < best_score:
            best_net, best_score = net, value
        return value

    def score_order(nets):
        return sorted(nets, key=lambda x: scored.get(x, float("inf")))

    # Repair is free; retain valid, unique topologies for both seeding and restarts.
    for raw in candidates:
        try:
            net = canon(tools.repair(raw))
            if tools.is_valid(net):
                repaired.add(net)
        except Exception:
            continue

    if not repaired:
        try:
            net = canon(tools.repair([]))
            if tools.is_valid(net):
                repaired.add(net)
        except Exception:
            return []

    energy_ranked = sorted(repaired, key=energy)

    # Build a modest seed set: prefer good predicted quality, but avoid near clones.
    starts = []
    for net in energy_ranked:
        if all(distance(net, old) >= 4 for old in starts):
            starts.append(net)
        if len(starts) >= min(22, len(repaired)):
            break
    for net in energy_ranked:
        if len(starts) >= min(22, len(repaired)):
            break
        if net not in starts:
            starts.append(net)

    # Consensus is only a proposal bias. The search remains free to leave this region.
    consensus_count = {}
    for net in energy_ranked[:min(40, len(energy_ranked))]:
        for edge in net:
            consensus_count[edge] = consensus_count.get(edge, 0) + 1
    consensus = {
        edge for edge, count in consensus_count.items()
        if count >= max(3, int(0.18 * min(40, len(energy_ranked))))
    }

    for net in starts:
        if tools.lp_calls_left <= 0:
            break
        try:
            evaluate(net)
        except Exception as exc:
            if type(exc).__name__ == "BudgetExhausted":
                break
            continue

    if not scored:
        return list(starts[0])

    def archive():
        """Keep top solutions and representatives of separated search regions."""
        ranked = score_order(scored)
        out = ranked[:8]
        for net in ranked[8:]:
            if all(distance(net, old) >= 4 for old in out):
                out.append(net)
            if len(out) >= 26:
                break
        for net in ranked:
            if len(out) >= 36:
                break
            if net not in out:
                out.append(net)
        return out

    def select_parent(pop):
        if not pop:
            return best_net
        r = rng.random()
        if r < 0.58:
            pool = pop[:min(7, len(pop))]
        elif r < 0.88:
            pool = pop[:min(20, len(pop))]
        else:
            pool = pop
        # Usually choose a strong parent; occasionally favor an under-visited one.
        weights = []
        for i, net in enumerate(pool):
            quality = 1.0 / (1.0 + 0.16 * i)
            exploration = 1.0 + 1.5 / (1.0 + visits.get(net, 0)) ** 0.5
            stale = 1.0 + min(1.5, 0.08 * stalls.get(net, 0))
            weights.append(quality * exploration * stale)
        total = sum(weights)
        return pool[int(rng.choice(len(pool), p=[w / total for w in weights]))]

    def switch(net, p, q):
        arcs = list(net)
        (a, b), (c, d) = arcs[p], arcs[q]
        if a == d or c == b:
            return None
        cur = set(net)
        if (a, d) in cur or (c, b) in cur:
            return None
        cur.remove((a, b))
        cur.remove((c, d))
        cur.add((a, d))
        cur.add((c, b))
        return tuple(sorted(cur))

    def add_consensus_moves(net, moves, current):
        # Insert a recurring target arc with one legal degree-preserving 2-switch.
        targets = [e for e in consensus if e not in current]
        if not targets:
            return
        rng.shuffle(targets)
        for a, b in targets[:min(10, len(targets))]:
            outgoing = [edge for edge in net if edge[0] == a]
            incoming = [edge for edge in net if edge[1] == b]
            for _, x in outgoing:
                for y, _ in incoming:
                    old = {(a, x), (y, b)}
                    if y == a or x == b or (y, x) in current - old:
                        continue
                    proposal = tuple(sorted((current - old) | {(a, b), (y, x)}))
                    if proposal != net and tools.is_valid(proposal):
                        moves.add(proposal)

    def propose(net, detail):
        arcs = list(net)
        current = set(net)
        moves = set()

        prices = detail.get("prices", {}) if detail else {}
        hot = {
            tuple(map(int, edge)) for edge, price in prices.items()
            if float(price) > 0
        }
        loads = detail.get("loads", {}) if detail else {}
        if not hot and loads:
            ranked_loads = sorted(
                ((float(value), tuple(map(int, edge))) for edge, value in loads.items()),
                reverse=True,
            )
            hot = {edge for _, edge in ranked_loads[:6]}

        # Critical-link switches form the main focused neighborhood.
        for p in range(len(arcs)):
            for q in range(p + 1, len(arcs)):
                if hot and arcs[p] not in hot and arcs[q] not in hot:
                    continue
                proposal = switch(net, p, q)
                if proposal is not None:
                    moves.add(proposal)

        # Keep a small exploration floor, including when prices are degenerate.
        for _ in range(90):
            if len(arcs) < 2:
                break
            p = int(rng.integers(len(arcs)))
            q = int(rng.integers(len(arcs) - 1))
            if q >= p:
                q += 1
            proposal = switch(net, p, q)
            if proposal is not None:
                moves.add(proposal)

        # Three-link rotations explore beyond the two-switch neighborhood.
        trials = 70 + min(180, 35 * (stalls.get(net, 0) // 3))
        for _ in range(trials):
            if len(arcs) < 3:
                break
            ids = rng.choice(len(arcs), size=3, replace=False)
            (a, b), (c, d), (e, f) = (arcs[int(i)] for i in ids)
            old = {(a, b), (c, d), (e, f)}
            if hot and not (old & hot):
                continue
            for new in (
                {(a, d), (c, f), (e, b)},
                {(a, f), (c, b), (e, d)},
            ):
                if any(x == y for x, y in new) or len(new) != 3:
                    continue
                if new & (current - old):
                    continue
                proposal = tuple(sorted((current - old) | new))
                moves.add(proposal)

        add_consensus_moves(net, moves, current)

        # A few short random kicks can cross local barriers.
        for _ in range(10):
            cur = set(net)
            for _ in range(2 + int(rng.integers(4))):
                now = list(cur)
                if len(now) < 2:
                    break
                p, q = rng.choice(len(now), size=2, replace=False)
                (a, b), (c, d) = now[int(p)], now[int(q)]
                if a == d or c == b or (a, d) in cur or (c, b) in cur:
                    continue
                cur.remove((a, b))
                cur.remove((c, d))
                cur.add((a, d))
                cur.add((c, b))
            if cur != current:
                moves.add(tuple(sorted(cur)))

        return [m for m in moves if m != net and tools.is_valid(m)]

    def select_batch(moves, parent, batch_size=5):
        fresh = [net for net in moves if net not in scored]
        if not fresh:
            return []
        base_energy = energy(parent)
        ranked = sorted(
            ((energy(net) - base_energy, net) for net in fresh),
            key=lambda item: item[0],
        )
        selected = []

        # Exploitation: take several low-energy proposals, spread across topologies.
        for _, net in ranked:
            if all(distance(net, old) >= 2 for old in selected):
                selected.append(net)
            if len(selected) >= min(3, batch_size):
                break

        remaining = [net for _, net in ranked if net not in selected]
        refs = archive()

        # Novelty slot: probe a topology far from already scored regions.
        if remaining and len(selected) < batch_size:
            novel = max(
                remaining,
                key=lambda net: (
                    min((distance(net, old) for old in refs), default=len(net)),
                    -energy(net),
                ),
            )
            selected.append(novel)
            remaining.remove(novel)

        # Exploration slot: randomize among moderately promising proposals.
        if remaining and len(selected) < batch_size:
            window = remaining[:max(1, min(len(remaining), 30))]
            selected.append(window[int(rng.integers(len(window)))])
        for _, net in ranked:
            if len(selected) >= batch_size:
                break
            if net not in selected:
                selected.append(net)
        return selected[:batch_size]

    rounds = 0
    while tools.lp_calls_left > 0:
        rounds += 1
        pop = archive()
        parent = select_parent(pop)
        if parent is None:
            break

        try:
            detail = tools.lp_detail(parent)
        except Exception:
            detail = None

        moves = propose(parent, detail)
        batch = select_batch(moves, parent)

        # If the selected parent has an exhausted neighborhood, restart from another
        # archived topology rather than spending time repeatedly regenerating it.
        if not batch:
            alternatives = [x for x in pop if x != parent]
            if alternatives:
                parent = alternatives[int(rng.integers(len(alternatives)))]
                try:
                    detail = tools.lp_detail(parent)
                except Exception:
                    detail = None
                batch = select_batch(propose(parent, detail), parent)

        if not batch:
            # Try fresh kicks before terminating on a fully explored neighborhood.
            for _ in range(60):
                cur = set(parent)
                arcs = list(cur)
                for _ in range(3 + int(rng.integers(5))):
                    p, q = rng.choice(len(arcs), size=2, replace=False)
                    (a, b), (c, d) = arcs[int(p)], arcs[int(q)]
                    if a == d or c == b or (a, d) in cur or (c, b) in cur:
                        continue
                    cur.remove((a, b))
                    cur.remove((c, d))
                    cur.add((a, d))
                    cur.add((c, b))
                kicked = tuple(sorted(cur))
                if kicked != parent and kicked not in scored and tools.is_valid(kicked):
                    batch = [kicked]
                    break

        if not batch:
            if rounds > 10000:
                break
            continue

        visits[parent] = visits.get(parent, 0) + 1
        old_score = scored.get(parent, float("inf"))
        try:
            for net in batch:
                if tools.lp_calls_left <= 0:
                    break
                evaluate(net)
        except Exception as exc:
            if type(exc).__name__ == "BudgetExhausted":
                return list(best_net) if best_net is not None else []
            # Other individual scoring failures should not discard the search state.

        new_best = min((scored.get(net, float("inf")) for net in batch), default=float("inf"))
        stalls[parent] = 0 if new_best < old_score else stalls.get(parent, 0) + 1

    return list(best_net) if best_net is not None else []
# EVOLVE-BLOCK-END
