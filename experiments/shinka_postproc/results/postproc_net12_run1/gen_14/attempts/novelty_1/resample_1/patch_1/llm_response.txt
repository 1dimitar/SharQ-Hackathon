    def kick(net, count):
        """Apply up to count random degree-preserving two-switches."""
        for _ in range(count):
            arcs = list(net)
            if len(arcs) < 2:
                break
            order = rng.permutation(len(arcs))
            changed = False
            for x in range(len(order)):
                p = int(order[x])
                a, b = arcs[p]
                for y in range(x + 1, len(order)):
                    q = int(order[y])
                    c, d = arcs[q]
                    new1, new2 = (a, d), (c, b)
                    arc_set = set(net)
                    if a == c or a == d or c == b or new1 == new2:
                        continue
                    if new1 in arc_set or new2 in arc_set:
                        continue
                    net = tuple(sorted((arc_set - {(a, b), (c, d)}) | {new1, new2}))
                    changed = True
                    break
                if changed:
                    break
            if not changed:
                break
        return net
