"""Classical post-processing of QAOA output for QOBLIB 08."""

import json
import time

import numpy as np

# EVOLVE-BLOCK-START
import math


def improve(demand, candidates, tools, rng):
    """Search a diverse portfolio with adaptive surrogate/dual screening."""
    scores = {}
    detail_cache = {}
    archive = []
    best_net = None
    best_score = float("inf")
    n = int(tools.n)

    def canon(arcs):
        return tuple(sorted((int(a), int(b)) for a, b in arcs))

    def topo_distance(a, b):
        return len(set(a).symmetric_difference(b))

    def evaluate(net):
        nonlocal best_net, best_score
        net = canon(net)
        if net in scores:
            return scores[net]
        if not tools.is_valid(net):
            return float("inf")
        value = float(tools.lp(net))
        scores[net] = value
        if value < best_score:
            best_net, best_score = net, value
        return value

    def add_archive(net, score):
        if any(net == old for old, _ in archive):
            return
        # Keep a quality/diversity portfolio, rather than only the best neighbors.
        if len(archive) >= 16:
            nearest = min(range(len(archive)),
                          key=lambda i: topo_distance(net, archive[i][0]))
            old_net, old_score = archive[nearest]
            if topo_distance(net, old_net) <= 2 and score >= old_score:
                return
        archive.append((net, score))
        archive.sort(key=lambda x: x[1])
        if len(archive) > 16:
            archive.pop()

    def get_detail(net):
        if net not in detail_cache:
            try:
                detail_cache[net] = tools.lp_detail(net)
            except Exception:
                detail_cache[net] = {}
        return detail_cache[net]

    def make_moves(net, hot, trials=100):
        arcs = list(net)
        cur = set(net)
        moves = set()

        # Degree-preserving two-switches.
        for i, (a, b) in enumerate(arcs):
            for c, d in arcs[i + 1:]:
                if a == c or ((a, b) not in hot and (c, d) not in hot):
                    continue
                x, y = (a, d), (c, b)
                if x[0] == x[1] or y[0] == y[1] or x == y:
                    continue
                if x in cur or y in cur:
                    continue
                moves.add(canon((cur - {(a, b), (c, d)}) | {x, y}))

        # Rotations provide additional moves beyond two-switch descent.
        for _ in range(trials):
            ids = rng.choice(len(arcs), size=3, replace=False)
            (a, b), (c, d), (e, f) = (arcs[int(i)] for i in ids)
            old = {(a, b), (c, d), (e, f)}
            if hot and not (old & hot):
                continue
            if rng.random() < 0.5:
                repl = {(a, d), (c, f), (e, b)}
            else:
                repl = {(a, f), (c, b), (e, d)}
            if len(repl) != 3 or any(x == y for x, y in repl):
                continue
            if repl & (cur - old):
                continue
            moves.add(canon((cur - old) | repl))
        return moves

    # Repair once, then seed LP scoring with low-energy, topologically diverse starts.
    repaired = set()
    for cand in candidates:
        try:
            net = canon(tools.repair(cand))
            if tools.is_valid(net):
                repaired.add(net)
        except Exception:
            continue

    ordered = sorted(repaired, key=lambda x: tools.energy(x))
    starts = []
    pool = ordered[:]
    while pool and len(starts) < 28:
        if not starts:
            chosen = pool[0]
        else:
            # Blend energy rank with distance to already chosen starts.
            chosen = max(pool, key=lambda x:
                         min(topo_distance(x, y) for y in starts)
                         - 0.10 * ordered.index(x))
        starts.append(chosen)
        pool.remove(chosen)
        if len(starts) >= 12 and pool:
            # After establishing diversity, fill by surrogate rank.
            if len(starts) >= 20:
                break
    for net in ordered:
        if len(starts) >= 28:
            break
        if net not in starts:
            starts.append(net)

    for net in starts:
        if tools.lp_calls_left <= 0:
            break
        try:
            value = evaluate(net)
            add_archive(net, value)
        except Exception as exc:
            if type(exc).__name__ == "BudgetExhausted":
                break
            raise

    if not scores:
        return []

    iteration = 0
    state = best_net
    state_score = best_score

    while tools.lp_calls_left > 0:
        iteration += 1

        # Use an elite-biased portfolio, with occasional broader restart.
        if archive:
            if rng.random() < 0.68:
                limit = min(6, len(archive))
                state, state_score = archive[int(rng.integers(limit))]
            else:
                state, state_score = archive[int(rng.integers(len(archive)))]

        detail = get_detail(state)
        prices = {}
        try:
            prices = {tuple(map(int, link)): float(p)
                      for link, p in detail.get("prices", {}).items()}
        except Exception:
            prices = {}

        hot = {link for link, price in prices.items() if price > 1e-10}
        if not hot:
            loads = detail.get("loads", {})
            try:
                vals = [float(v) for v in loads.values()]
                cutoff = max(vals) * 0.93 if vals else 0.0
                hot = {tuple(map(int, link)) for link, val in loads.items()
                       if float(val) >= cutoff}
            except Exception:
                hot = set(state)

        moves = make_moves(state, hot, trials=120)
        fresh = [m for m in moves
                 if m not in scores and not tools.is_cached(m)]
        if not fresh:
            moves = make_moves(state, set(), trials=180)
            fresh = [m for m in moves
                     if m not in scores and not tools.is_cached(m)]
        if not fresh:
            continue

        old = set(state)

        def dual_benefit(net):
            removed, added = old - set(net), set(net) - old
            return (sum(prices.get(x, 0.0) for x in removed)
                    - sum(prices.get(x, 0.0) for x in added))

        # Rank all proposals cheaply using both the Hamiltonian and LP duals.
        energy_order = sorted(fresh, key=tools.energy)
        energy_rank = {m: i for i, m in enumerate(energy_order)}
        dual_order = sorted(fresh, key=dual_benefit, reverse=True)
        dual_rank = {m: i for i, m in enumerate(dual_order)}
        ranked = sorted(fresh, key=lambda m:
                        energy_rank[m] + 0.65 * dual_rank[m])

        shortlist = ranked[:min(64, len(ranked))]
        top = shortlist[:min(16, len(shortlist))]
        if len(top) > 1:
            distances = [
                topo_distance(top[i], top[j])
                for i in range(len(top)) for j in range(i)
            ]
            mean_distance = sum(distances) / max(1, len(distances))
        else:
            mean_distance = 0.0

        # Clustered top proposals: increase diversity pressure.
        diversity_weight = 0.12 if mean_distance >= 5.0 else 0.34
        rank = {m: i for i, m in enumerate(shortlist)}
        batch = []
        while shortlist and len(batch) < 9:
            if not batch:
                chosen = shortlist[0]
            else:
                chosen = max(
                    shortlist,
                    key=lambda m: (
                        min(topo_distance(m, old_net) for old_net in batch)
                        - diversity_weight * rank[m]
                    ),
                )
            batch.append(chosen)
            shortlist.remove(chosen)

        # Reserve a few evaluations for high dual-benefit moves and exploration.
        chosen_set = set(batch)
        positive_dual = sorted(
            (m for m in fresh if m not in chosen_set and dual_benefit(m) > 1e-10),
            key=dual_benefit, reverse=True)
        batch.extend(positive_dual[:3])
        tail = [m for m in fresh if m not in set(batch)]
        if tail:
            count = min(2, len(tail))
            batch.extend(tail[int(i)] for i in
                         rng.choice(len(tail), size=count, replace=False))

        evaluated = []
        for net in batch:
            if tools.lp_calls_left <= 0:
                break
            try:
                value = evaluate(net)
                evaluated.append((net, value))
                add_archive(net, value)
            except Exception as exc:
                if type(exc).__name__ == "BudgetExhausted":
                    break
                raise

        if evaluated:
            winner, value = min(evaluated, key=lambda x: x[1])
            if value <= state_score:
                state, state_score = winner, value
            else:
                # Mild annealing prevents the portfolio from becoming greedy.
                temp = max(500.0, min(4500.0, 0.012 * state_score))
                temp *= 0.85 ** (iteration % 10)
                if rng.random() < math.exp(-min(50.0, (value - state_score) / temp)):
                    state, state_score = winner, value

    return list(best_net) if best_net is not None else []

def run_postproc(pool_path, lp_budget, seed, control=False, time_limit=150):
    """Fixed entry point called by the evaluator: one run on one candidate pool."""
    import signal

    import oracle
    from hamiltonian import surrogate_terms

    pool = json.load(open(pool_path))
    import qoblib_tools as qt

    n = int(pool["instance"][-2:])
    demand = qt.load_network_demand(n)
    linear, quadratic, _ = surrogate_terms(np.array(demand, dtype=float))
    tools = oracle.Tools(demand, lp_budget, (
        {tuple(map(int, a)): float(w) for a, w in linear.items()},
        {(tuple(map(int, a)), tuple(map(int, b))): float(w)
         for (a, b), w in quadratic.items()},
    ))
    candidates = [[tuple(map(int, a)) for a in c["arcs"]]
                  for c in pool["candidates"]]
    if control:
        crng = np.random.default_rng(10_000 + seed)
        candidates = [qt.random_topology(n, crng) for _ in candidates]

    def _timeout(signum, frame):
        raise oracle.RunTimeout()

    start = time.perf_counter()
    returned, error, timed_out = None, None, False
    signal.signal(signal.SIGALRM, _timeout)
    signal.alarm(int(time_limit))
    try:
        returned = improve(demand.copy(), candidates, tools,
                           np.random.default_rng(seed))
    except oracle.BudgetExhausted:
        pass
    except oracle.RunTimeout:
        timed_out = True
    except Exception as exc:
        error = f"{type(exc).__name__}: {exc}"
    finally:
        signal.alarm(0)

    best_score, best_arcs = tools.certified_best()
    if returned is not None:
        try:
            ret = [tuple(map(int, a)) for a in returned]
            if qt.is_valid_topology(n, ret):
                s = qt.route_min_congestion(demand, ret).objective
                if s < best_score:
                    best_score, best_arcs = s, tuple(sorted(ret))
        except Exception:
            pass
    return {
        "pool": pool_path, "control": control, "n": n, "error": error,
        "best": float(best_score),
        "arcs": [list(a) for a in best_arcs] if best_arcs else None,
        "lp_calls": tools.lp_calls_used,
        "seconds": time.perf_counter() - start,
        "timed_out": timed_out,
    }
