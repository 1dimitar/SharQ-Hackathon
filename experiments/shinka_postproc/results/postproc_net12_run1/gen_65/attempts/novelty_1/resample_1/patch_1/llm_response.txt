<DESCRIPTION>
Track whether focused switches, random switches, or three-edge rotations produce improvements. Periodically adjust their proposal counts modestly according to recent improvement rates, while enforcing a minimum number of proposals for each move type. This lets the search spend more generation effort on productive neighborhoods without eliminating exploration of the others. Path-relink moves are grouped with random switches for this lightweight adaptation.

    def make_moves(net, hot, kick_size, effort):
        """Generate separately trackable focused, random, and rotation moves."""
        arcs = list(net)
        current = set(net)
        moves = {}
        hot_ids = [i for i, arc in enumerate(arcs) if arc in hot]

        # Keep minimum exploration in every neighborhood while adapting effort.
        focused_attempts = max(20, int(max(100, 35 * kick_size) * 0.72 * effort["focused"]))
        random_attempts = max(20, int(max(100, 35 * kick_size) * 0.28 * effort["random"]))

        def add_switch(p, q, source):
            a, b = arcs[p]
            c, d = arcs[q]
            if a == d or c == b or (a, d) in current or (c, b) in current:
                return
            proposal = canon((current - {(a, b), (c, d)}) | {(a, d), (c, b)})
            if proposal != net:
                moves.setdefault(proposal, source)

        for _ in range(focused_attempts):
            if len(arcs) < 2:
                break
            p = int(rng.choice(hot_ids)) if hot_ids else int(rng.integers(len(arcs)))
            q = int(rng.integers(len(arcs) - 1))
            if q >= p:
                q += 1
            add_switch(p, q, "focused")

        for _ in range(random_attempts):
            if len(arcs) < 2:
                break
            p = int(rng.integers(len(arcs)))
            q = int(rng.integers(len(arcs) - 1))
            if q >= p:
                q += 1
            add_switch(p, q, "random")

        rotations = max(8, int(max(10, 8 * kick_size) * effort["rotation"]))
        for _ in range(rotations):
            if len(arcs) < 3:
                break
            ids = rng.choice(len(arcs), size=3, replace=False)

            if new & (current - old):
                continue
            proposal = canon((current - old) | new)
            if proposal != net:
                moves.setdefault(proposal, "rotation")
